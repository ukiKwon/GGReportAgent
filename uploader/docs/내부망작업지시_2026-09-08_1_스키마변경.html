테이블 생성 후 ③/uploader/upload 로 파일 1건 C:\uploader-local\unclassified\ 에 생성 + 목록 표시
테이블 생성 후 ④DB 에서 SELECT 업로드일시 FROM TSKGIAF01 20260908103000 형태 14자 = 일시 TypeHandler 정상
분류등록한 기관명과 똑같이 적어 업로드 파일이 classified\ 아래로 이동
증상별 원인
  • ORA-00942 → 테이블 이름. 소스가 옛 이름을 부르거나 DDL 미적용.
  • ORA-00904 → 컬럼 이름. 2절 표와 대조하십시오.
  • ORA-02289: sequence does not exist → 시퀀스 2개 누락. 테이블만 만들면 조회는 되고 등록만 실패합니다.
  • ORA-02290: check constraint violated → 분류상태구분 에 '01'~'04' 밖의 값이 들어갔습니다.
  • 코드로 옮길 수 없는 값이다 — '지자체' → 카테고리 목록이 옛 4종입니다. 3절의 수정 8개 중 컨트롤러·템플릿이 반영되지 않았습니다.
  • 일시 형식이 YYYYMMDDHH24MISS 가 아니다 → DB 에 다른 형식의 값이 이미 들어 있습니다(수기 입력 등).
  • 오류는 없는데 분류가 하나도 안 됨 → InstitutionMapper.findByName 을 의심하십시오. 그다음은 기관명 오타입니다.

6DDL 전문 (DBA 전달용)

-- ── 업로드된 파일 ───────────────────────────────────
CREATE TABLE TSKGIAF01 (
    업로드파일일련번호  DECIMAL(16)        NOT NULL,
    기관명              VARCHAR2(40 CHAR),
    기관영업분류구분    VARCHAR2(2 CHAR),                          -- 01~05
    문서년              VARCHAR2(4 CHAR),
    분류상태구분        VARCHAR2(2 CHAR)   DEFAULT '01' NOT NULL,  -- 01~04
    원본파일명          VARCHAR2(260 CHAR) NOT NULL,
    업로드일시          VARCHAR2(14 CHAR)  NOT NULL,               -- YYYYMMDDHH24MISS · 업무 시각
    분류일시            VARCHAR2(14 CHAR),
    저장경로내용        VARCHAR2(600 CHAR) NOT NULL,
    시스템사용일시      VARCHAR2(14 CHAR),                         -- 감사 시각
    시스템사용자번호    CHAR(7),                                   -- GI11100 / BATCH01
    CONSTRAINT PK_TSKGIAF01 PRIMARY KEY (업로드파일일련번호),
    CONSTRAINT CK_TSKGIAF01_분류상태 CHECK (분류상태구분 IN ('01','02','03','04'))
);

-- ── 기관 ────────────────────────────────────────────
CREATE TABLE TSKGIAF02 (
    기관일련번호        DECIMAL(16)       NOT NULL,
    기관명              VARCHAR2(40 CHAR) NOT NULL,
    기관영업분류구분    VARCHAR2(2 CHAR)  NOT NULL,
    협약기관상태구분    VARCHAR2(2 CHAR),              -- 01 미협약 / 02 협약     (앱 미사용)
    입찰공고상태구분    VARCHAR2(2 CHAR),              -- 01 공고전 / 02 공고중 / 03 공고완료
    입찰공고시작년월일  VARCHAR2(8 CHAR),              -- YYYYMMDD               (앱 미사용)
    입찰공고마감년월일  VARCHAR2(8 CHAR),              -- YYYYMMDD               (앱 미사용)
    시스템사용일시      VARCHAR2(14 CHAR) NOT NULL,    -- 옛 수정일시가 이 이름으로
    시스템사용자번호    CHAR(7),
    CONSTRAINT PK_TSKGIAF02 PRIMARY KEY (기관일련번호),
    CONSTRAINT UK_TSKGIAF02_기관명 UNIQUE (기관명)
);

-- ── 시퀀스 (이름 변경 불가 — 매퍼가 그대로 부른다) ──
CREATE SEQUENCE SQ_KGI_AGENTFILETABLE  START WITH 1 INCREMENT BY 1 NOCACHE;  -- TSKGIAF01 용
CREATE SEQUENCE SQ_KGI_AGENTGIGANTABLE START WITH 1 INCREMENT BY 1 NOCACHE;  -- TSKGIAF02 용

-- ── 인덱스 2개 (25·28바이트 — 11g 상한 30) ──────────

CREATE INDEX IX_TSKGIAF01_분류상태   ON TSKGIAF01 (분류상태구분, 업로드파일일련번호);

SELECT table_name    FROM user_tables    WHERE table_name    IN ('TSKGIAF01', 'TSKGIAF02');

⚠️ 시퀀스는 NOCACHE 를 유지해 주십시오

목록 화면의 최신순 정렬을 업로드일시가 아니라 PK(일련번호)로 합니다. 업로드일시가 문자열 초 단위라 같은 초에 여러 건이 들어오면 순서가 뒤섞이는데, 일련번호는 그 경우에도 정확하기 때문입니다. 이 전제는 시퀀스가 증가순 = 삽입순일 때 성립합니다 — 캐시를 켜면 재기동 때 번호가 튀어 깨집니다.
