            + "\\u2606\\u2605\\u3147\\u110B\\u2217\\u00D7]";
    /** 구분자 0~3자. 소유 한정자라 되돌아가지 않는다(상-2). */
    private static final String SP = "[\\s\\-./]{0,3}+";
    /** 국가번호(+82·0082·82, 뒤에 (0)·0 선택) 또는 국내 0. +82-02 도 잡는다(상-3). */

            + "(?:2|3[1-3]|4[1-4]|5[1-5]|6[1-4]|70|50\\d)\\)?" + SP

    /** scanExternal 이 마감 시각을 넘기면 던진다. 호출자는 그 check 를 TIMEOUT 으로 제외한다. */
    public static final class ScanTimeoutException extends RuntimeException {
        private static final long serialVersionUID = 1L;
        ScanTimeoutException() { super("scan deadline exceeded"); }
    }

    /**
     * 외부 확인 반입용 안전망 검사(기본 10초). 종류·건수만 알려 주고 value 는 항상 "[가림]" 이다.
     * 휴대폰은 가렸든 아니든 있으면 건수에 든다. 유선은 국번 뒤 두 묶음이 모두 가림표일 때만 통과한다.
     * 이메일은 로컬 부분이 {@code 첫글자+*...} 일 때만 통과한다.
     */
    public static List<Finding> scanExternal(String text) {
        return scanExternal(text, System.nanoTime() + DEFAULT_BUDGET_NANOS);
    }

    /** @param deadlineNanos {@link System#nanoTime()} 기준 절대 마감 시각. 넘기면 {@link ScanTimeoutException}. */
    public static List<Finding> scanExternal(String text, long deadlineNanos) {
        List<Finding> out = new ArrayList<Finding>();
        if (text == null || text.isEmpty()) {
            return out;
        }
        checkDeadline(deadlineNanos);
        String f = collapseSpace(fold(text, true));            // 공백 연속은 하나로(상-2)
        Matcher m = MOBILE_ANY.matcher(new Budget(f, deadlineNanos));
        while (m.find()) {
            out.add(new Finding("휴대폰", REDACTED));
        }
        Matcher l = LANDLINE.matcher(new Budget(f, deadlineNanos));
        while (l.find()) {
            if (hasDigit(l.group(1)) || hasDigit(l.group(2))) {
                out.add(new Finding("유선전화", REDACTED));
            }
        }
        checkDeadline(deadlineNanos);
        for (Finding x : scanLoose(text)) {                    // 주민·외국인·카드(기존 규칙). 휴대폰·이메일은 위에서 따로
            String k = x.getKind();
            if (!"휴대폰".equals(k) && !EMAIL_KIND.equals(k)) {
                out.add(new Finding(k, REDACTED));
            }
        }
        Matcher e = EMAIL_ANY.matcher(new Budget(fold(text, false), deadlineNanos));   // '.' 을 살린 접힘
        while (e.find()) {
            if (!EMAIL_MASKED_OK.matcher(e.group(1)).matches()) {
                out.add(new Finding(EMAIL_KIND, REDACTED));
            }
        }
        return out;
    }

    private static void checkDeadline(long deadlineNanos) {
        if (System.nanoTime() - deadlineNanos > 0) {
            throw new ScanTimeoutException();
        }
    }

    private static boolean hasDigit(String s) {
        for (int i = 0; i < s.length(); i++) {
            if (Character.isDigit(s.charAt(i))) {
                return true;
            }
        }
        return false;
    }

    /** 공백류(스페이스·탭·줄바꿈·VT·FF·CR) 연속을 스페이스 하나로. 정규식을 쓰지 않는다. */
    private static String collapseSpace(String s) {
        StringBuilder b = new StringBuilder(s.length());
        boolean inWs = false;
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            boolean ws = c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\f' || c == (char) 0x0B;
            if (ws) {
                if (!inWs) {
                    b.append(' ');
                }
                inWs = true;
            } else {
                b.append(c);
                inWs = false;
            }
        }
        return b.toString();
    }

    /** charAt 마다(1024번에 한 번 시각 확인) 마감을 보는 감싸개 — 정규식은 인터럽트에 반응하지 않기 때문이다(상-2). */
    private static final class Budget implements CharSequence {
        private final CharSequence s;
        private final long deadline;
        private int n;

        Budget(CharSequence s, long deadline) { this.s = s; this.deadline = deadline; }

        @Override public char charAt(int i) {
            if ((n++ & 0x3FF) == 0 && System.nanoTime() - deadline > 0) {
                throw new ScanTimeoutException();
            }
            return s.charAt(i);
        }
        @Override public int length() { return s.length(); }
        @Override public CharSequence subSequence(int a, int b) { return s.subSequence(a, b); }
        @Override public String toString() { return s.toString(); }
    }

RESERVED_NAME_RE = re.compile(r"con|prn|aux|nul|com[1-9]|lpt[1-9]", re.IGNORECASE)
TOOL_RE = re.compile(r"[A-Za-z0-9._ -]{1,40}")
_BAD_CATEGORIES = ("Cc", "Cf", "Zl", "Zp", "Cs")


def _text_ok(value, max_len: int) -> bool:
    if not isinstance(value, str) or not value.strip() or len(value) > max_len:
        return False
    return not any(unicodedata.category(c) in _BAD_CATEGORIES for c in value)


def _opt_text(value, max_len: int) -> bool:
    return value is None or _text_ok(value, max_len)


def _opt_match(pattern, value) -> bool:
    return value is None or (isinstance(value, str) and pattern.fullmatch(value) is not None)


def _is_int(value, lo: int, hi: int) -> bool:
    return isinstance(value, int) and not isinstance(value, bool) and lo <= value <= hi


def _validate_checks(manifest: dict, batch_dir: Path) -> list[str]:
    checks = manifest.get("checks")
    if checks is None:
        return []
    if not isinstance(checks, list):
        return ["checks가 배열이 아닙니다"]
    if len(checks) > MAX_CHECKS:
        return [f"checks 건수가 상한({MAX_CHECKS})을 넘습니다"]
    errors: list[str] = []
    ids: set[str] = set()
    paths: set[str] = set()
    total = [0]
    for index, check in enumerate(checks):
        errors.extend(_validate_check(check, index, batch_dir, ids, paths, total))
    if total[0] > MAX_BATCH_BYTES:
        errors.append("checks 첨부 합계가 상한(200 MiB)을 넘습니다")
    return errors


def _validate_check(check, index: int, batch_dir: Path, ids: set, paths: set, total: list) -> list[str]:
    where = f"checks[{index}]"
    if not isinstance(check, dict):
        return [f"{where}: 객체가 아닙니다"]
    errors: list[str] = []

    check_id = check.get("check_id")
    if not isinstance(check_id, str) or not CHECK_ID_RE.fullmatch(check_id):
        errors.append(f"{where}.check_id: 형식이 잘못됐습니다")
    elif check_id.lower() in ids:       # Windows 는 대소문자를 구분하지 않아 저장 이름이 겹친다
        errors.append(f"{where}.check_id: 중복입니다")
    else:
        ids.add(check_id.lower())

    inst = check.get("institution")
    if not isinstance(inst, dict):
        errors.append(f"{where}.institution: 객체가 아닙니다")
    else:
        if not _text_ok(inst.get("name_ko"), 100):
            errors.append(f"{where}.institution.name_ko: 형식이 잘못됐습니다")
        if not _opt_text(inst.get("type"), 30):
            errors.append(f"{where}.institution.type: 형식이 잘못됐습니다")
        for key in ("region_code", "sub_region_code"):
            if not _opt_match(REGION_RE, inst.get(key)):
                errors.append(f"{where}.institution.{key}: 형식이 잘못됐습니다")

    status = check.get("status")
    if status not in CHECK_STATUSES:
        errors.append(f"{where}.status: 허용값이 아닙니다")
    if not _opt_match(DATETIME_RE, check.get("checked_at")) or check.get("checked_at") is None:
        errors.append(f"{where}.checked_at: 형식이 잘못됐습니다")
    kind = check.get("source_kind")
    if kind not in CHECK_KINDS:
        errors.append(f"{where}.source_kind: 허용값이 아닙니다")
    slug = check.get("source_slug")
    if not (isinstance(slug, str) and SOURCE_SLUG_RE.fullmatch(slug)):
        errors.append(f"{where}.source_slug: 형식이 잘못됐습니다")

    code = check.get("error_code")
    if status == "error":
        if not (isinstance(code, str) and CODE_RE.fullmatch(code)):
            errors.append(f"{where}.error_code: 형식이 잘못됐습니다")
    elif not _opt_match(CODE_RE, code):
        errors.append(f"{where}.error_code: 형식이 잘못됐습니다")

    if status == "found":
        errors.extend(_validate_found(check, where, kind))

    attachments = check.get("attachments")
    if attachments is not None:
        if not isinstance(attachments, list) or len(attachments) > MAX_CHECK_ATTACHMENTS:
            errors.append(f"{where}.attachments: 배열이 아니거나 건수 상한({MAX_CHECK_ATTACHMENTS})을 넘습니다")
        else:
            for j, att in enumerate(attachments):
                errors.extend(_validate_check_attachment(att, f"{where}.attachments[{j}]", batch_dir, paths, total))
    return errors


def _validate_found(check: dict, where: str, kind) -> list[str]:
    errors: list[str] = []
    match = check.get("match")
    if not isinstance(match, dict):
        errors.append(f"{where}.match: 객체가 아닙니다")
    else:
        if match.get("confidence") not in CONFIDENCE_LEVELS:
            errors.append(f"{where}.match.confidence: 허용값이 아닙니다")
        basis = match.get("basis")
        if not (isinstance(basis, list) and 1 <= len(basis) <= 6
                and all(isinstance(b, str) and CODE_RE.fullmatch(b) for b in basis)):
            errors.append(f"{where}.match.basis: 형식이 잘못됐습니다")
    if kind == "notice":
        n = check.get("notice")
        if not isinstance(n, dict):
            errors.append(f"{where}.notice: 객체가 아닙니다")
        else:
            if not _text_ok(n.get("title"), 300):
                errors.append(f"{where}.notice.title: 형식이 잘못됐습니다")
            if not _opt_text(n.get("number"), 60):
                errors.append(f"{where}.notice.number: 형식이 잘못됐습니다")
            for key in ("posted_at", "deadline_at"):
                if not _opt_match(DATE_STRICT_RE, n.get(key)):
                    errors.append(f"{where}.notice.{key}: YYYY-MM-DD가 아닙니다")
            if not _opt_match(URL_RE, n.get("url")):
                errors.append(f"{where}.notice.url: 형식이 잘못됐습니다")
    elif kind == "press":
        p = check.get("press")
        if not isinstance(p, dict):
            errors.append(f"{where}.press: 객체가 아닙니다")
        else:
            if not _text_ok(p.get("title"), 200):
                errors.append(f"{where}.press.title: 형식이 잘못됐습니다")
            if not _opt_text(p.get("publisher"), 60):
                errors.append(f"{where}.press.publisher: 형식이 잘못됐습니다")
            if not _opt_match(DATE_STRICT_RE, p.get("published_at")):
                errors.append(f"{where}.press.published_at: YYYY-MM-DD가 아닙니다")
            if not _opt_match(URL_RE, p.get("url")):
                errors.append(f"{where}.press.url: 형식이 잘못됐습니다")
    return errors


def _validate_check_attachment(att, w: str, batch_dir: Path, paths: set, total: list) -> list[str]:
    if not isinstance(att, dict):
        return [f"{w}: 객체가 아닙니다"]
    errors: list[str] = []
    path = att.get("path")
    path_ok = (isinstance(path, str) and len(path) <= MAX_CHECK_PATH_LEN
               and CHECK_PATH_RE.fullmatch(path) is not None
               and RESERVED_NAME_RE.fullmatch(path[6:].split(".", 1)[0]) is None)
    if not path_ok:
        errors.append(f"{w}.path: 허용되지 않는 경로 형식입니다")     # 받은 값은 싣지 않는다
    elif path.lower() in paths:
        errors.append(f"{w}.path: 중복입니다")
    else:
        paths.add(path.lower())
        target = batch_dir / path
        if target.is_symlink() or not target.is_file():
            errors.append(f"{w}.path: 파일이 없습니다")
    if not (isinstance(att.get("sha256"), str) and SHA256_RE.fullmatch(att["sha256"])):
        errors.append(f"{w}.sha256: 형식이 잘못됐습니다")
    size = att.get("size")
    if _is_int(size, 1, MAX_FILE_BYTES):
        total[0] += size
    else:
        errors.append(f"{w}.size: 1바이트~20 MiB 정수여야 합니다")
    for key in ("reported_tool", "reported_version"):
        if not _opt_match(TOOL_RE, att.get(key)):
            errors.append(f"{w}.{key}: 형식이 잘못됐습니다")
    for key in ("reported_phones_removed", "reported_landlines_masked", "reported_emails_masked"):
        v = att.get(key)
        if v is not None and not _is_int(v, 0, MAX_REPORTED_COUNT):
            errors.append(f"{w}.{key}: 0 이상 정수여야 합니다")
    scope = att.get("reported_scope")
    if scope is not None and not (isinstance(scope, list) and len(scope) <= len(REPORTED_SCOPES)
                                  and all(s in REPORTED_SCOPES for s in scope)):
        errors.append(f"{w}.reported_scope: 허용값이 아닙니다")
    return errors

    private static boolean absent(JsonNode n) { return n.isMissingNode() || n.isNull(); }
    private static boolean fullMatch(Pattern p, JsonNode n) { return n.isTextual() && p.matcher(n.asText()).matches(); }
    private static boolean optMatch(Pattern p, JsonNode n) { return absent(n) || fullMatch(p, n); }

    private static boolean textOk(JsonNode n, int max) {
        if (!n.isTextual()) {
            return false;
        }
        String s = n.asText();
        if (s.trim().isEmpty() || s.codePointCount(0, s.length()) > max) {
            return false;
        }
        for (int i = 0; i < s.length(); ) {
            int cp = s.codePointAt(i);
            i += Character.charCount(cp);
            int t = Character.getType(cp);
            if (t == Character.CONTROL || t == Character.FORMAT || t == Character.LINE_SEPARATOR
                    || t == Character.PARAGRAPH_SEPARATOR || t == Character.SURROGATE) {
                return false;
            }
        }
        return true;
    }

    private static List<String> validateChecks(JsonNode manifest, Path batchDir) {

            return errors;

        if (!checks.isArray()) {

    private static List<String> validateCheck(JsonNode check, int index, Path batchDir,
                                              Set<String> ids, Set<String> paths, long[] total) {
        String where = "checks[" + index + "]";
        List<String> errors = new ArrayList<>();
        if (!check.isObject()) {
            errors.add(where + ": 객체가 아닙니다");
            return errors;
        }
        JsonNode id = check.path("check_id");
        if (!fullMatch(CHECK_ID_RE, id)) {
            errors.add(where + ".check_id: 형식이 잘못됐습니다");
        } else if (!ids.add(id.asText().toLowerCase(Locale.ROOT))) {   // Windows 는 대소문자를 구분하지 않는다
            errors.add(where + ".check_id: 중복입니다");
        }

        JsonNode inst = check.path("institution");
        if (!inst.isObject()) {
            errors.add(where + ".institution: 객체가 아닙니다");
        } else {
            if (!textOk(inst.path("name_ko"), 100)) {
                errors.add(where + ".institution.name_ko: 형식이 잘못됐습니다");
            }
            if (!optText(inst.path("type"), 30)) {
                errors.add(where + ".institution.type: 형식이 잘못됐습니다");
            }
            for (String key : new String[]{"region_code", "sub_region_code"}) {
                if (!optMatch(REGION_RE, inst.path(key))) {
                    errors.add(where + ".institution." + key + ": 형식이 잘못됐습니다");
                }
            }
        }

        JsonNode statusNode = check.path("status");
        if (!inSet(statusNode, CHECK_STATUSES)) {
            errors.add(where + ".status: 허용값이 아닙니다");
        }
        String status = statusNode.isTextual() ? statusNode.asText() : "";
        if (!fullMatch(DATETIME_RE, check.path("checked_at"))) {
            errors.add(where + ".checked_at: 형식이 잘못됐습니다");
        }
        JsonNode kindNode = check.path("source_kind");
        if (!inSet(kindNode, CHECK_KINDS)) {
            errors.add(where + ".source_kind: 허용값이 아닙니다");
        }
        String kind = kindNode.isTextual() ? kindNode.asText() : "";
        if (!fullMatch(SOURCE_SLUG_RE, check.path("source_slug"))) {
            errors.add(where + ".source_slug: 형식이 잘못됐습니다");
        }

        JsonNode code = check.path("error_code");
        boolean codeOk = "error".equals(status) ? fullMatch(CODE_RE, code) : optMatch(CODE_RE, code);
        if (!codeOk) {
            errors.add(where + ".error_code: 형식이 잘못됐습니다");
        }
        if ("found".equals(status)) {
            errors.addAll(validateFound(check, where, kind));
        }

        JsonNode attachments = check.path("attachments");
        if (!absent(attachments)) {
            if (!attachments.isArray() || attachments.size() > MAX_CHECK_ATTACHMENTS) {
                errors.add(where + ".attachments: 배열이 아니거나 건수 상한(" + MAX_CHECK_ATTACHMENTS + ")을 넘습니다");
            } else {
                for (int j = 0; j < attachments.size(); j++) {
                    errors.addAll(validateCheckAttachment(attachments.get(j),
                            where + ".attachments[" + j + "]", batchDir, paths, total));
                }
            }
        }
        return errors;
    }

    private static List<String> validateFound(JsonNode check, String where, String kind) {
        List<String> out = new ArrayList<>();
        JsonNode match = check.path("match");
        if (!match.isObject()) {
            out.add(where + ".match: 객체가 아닙니다");

            }
            JsonNode basis = match.path("basis");
            boolean ok = basis.isArray() && basis.size() >= 1 && basis.size() <= 6;
            if (ok) {
                for (JsonNode b : basis) {
                    ok = ok && fullMatch(CODE_RE, b);
                }
            }
            if (!ok) {
                out.add(where + ".match.basis: 형식이 잘못됐습니다");
            }
        }
        if ("notice".equals(kind)) {
            JsonNode n = check.path("notice");
            if (!n.isObject()) {
                out.add(where + ".notice: 객체가 아닙니다");
            } else {
                if (!textOk(n.path("title"), 300)) {
                    out.add(where + ".notice.title: 형식이 잘못됐습니다");
                }
                if (!optText(n.path("number"), 60)) {
                    out.add(where + ".notice.number: 형식이 잘못됐습니다");
                }
                for (String key : new String[]{"posted_at", "deadline_at"}) {
                    if (!optMatch(DATE_STRICT_RE, n.path(key))) {
                        out.add(where + ".notice." + key + ": YYYY-MM-DD가 아닙니다");
                    }
                }
                if (!optMatch(URL_RE, n.path("url"))) {
                    out.add(where + ".notice.url: 형식이 잘못됐습니다");
                }
            }
        } else if ("press".equals(kind)) {
            JsonNode p = check.path("press");
            if (!p.isObject()) {
                out.add(where + ".press: 객체가 아닙니다");
            } else {
                if (!textOk(p.path("title"), 200)) {
                    out.add(where + ".press.title: 형식이 잘못됐습니다");
                }
                if (!optText(p.path("publisher"), 60)) {
                    out.add(where + ".press.publisher: 형식이 잘못됐습니다");
                }
                if (!optMatch(DATE_STRICT_RE, p.path("published_at"))) {
                    out.add(where + ".press.published_at: YYYY-MM-DD가 아닙니다");
                }
                if (!optMatch(URL_RE, p.path("url"))) {
                    out.add(where + ".press.url: 형식이 잘못됐습니다");
                }
            }
        }
        return out;
    }

    private static List<String> validateCheckAttachment(JsonNode att, String w, Path batchDir,
                                                        Set<String> paths, long[] total) {
        List<String> out = new ArrayList<>();
        if (!att.isObject()) {
            out.add(w + ": 객체가 아닙니다");
            return out;
        }
        JsonNode pathNode = att.path("path");
        boolean pathOk = false;
        if (pathNode.isTextual()) {
            String p = pathNode.asText();
            pathOk = p.length() <= MAX_CHECK_PATH_LEN && CHECK_PATH_RE.matcher(p).matches()
                    && !RESERVED_NAME_RE.matcher(p.substring(6).split("\\.", 2)[0]).matches();
        }
        if (!pathOk) {
            out.add(w + ".path: 허용되지 않는 경로 형식입니다");           // 받은 값은 싣지 않는다
        } else if (!paths.add(pathNode.asText().toLowerCase(Locale.ROOT))) {
            out.add(w + ".path: 중복입니다");
        } else if (!Files.isRegularFile(batchDir.resolve(pathNode.asText()), LinkOption.NOFOLLOW_LINKS)) {
            out.add(w + ".path: 파일이 없습니다");
        }
        if (!fullMatch(SHA256_RE, att.path("sha256"))) {
            out.add(w + ".sha256: 형식이 잘못됐습니다");
        }
        JsonNode size = att.path("size");
        if (intInRange(size, 1, MAX_FILE_BYTES)) {
            total[0] += size.asLong();
        } else {
            out.add(w + ".size: 1바이트~20 MiB 정수여야 합니다");
        }
        for (String key : new String[]{"reported_tool", "reported_version"}) {
            if (!optMatch(TOOL_RE, att.path(key))) {
                out.add(w + "." + key + ": 형식이 잘못됐습니다");
            }
        }
        for (String key : new String[]{"reported_phones_removed", "reported_landlines_masked", "reported_emails_masked"}) {
            JsonNode v = att.path(key);
            if (!absent(v) && !intInRange(v, 0, MAX_REPORTED_COUNT)) {
                out.add(w + "." + key + ": 0 이상 정수여야 합니다");
            }
        }
        JsonNode scope = att.path("reported_scope");
        if (!absent(scope)) {
            boolean ok = scope.isArray() && scope.size() <= REPORTED_SCOPES.length;
            if (ok) {
                for (JsonNode s : scope) {
                    ok = ok && inSet(s, REPORTED_SCOPES);
                }
            }
            if (!ok) {
                out.add(w + ".reported_scope: 허용값이 아닙니다");
            }
        }
        return out;
    }

### ④-3 첨부 전처리 규칙 (망 밖에서 반입 전에 끝낸다)

서버는 안전망으로 다시 검사하고, 걸린 check는 **그 check만 제외**하며 파일은 즉시 삭제한다. 제외 사유에는 종류·위치·건수만 남고 값은 남지 않는다.

1. **휴대폰**(010·011·016~019, `+82`·`0082`·괄호·사선·점 구분 포함)은 **삭제**한다. `***`로 바꾸지 않는다 — 서버는 가려진 휴대폰 모양도 "있음"으로 본다.
2. **유선**(02·031~064·070·050x)은 국번 뒤를 전부 가린다: `02-****-****`. 대표번호 15xx·16xx·18xx는 그대로 둔다.
3. **이메일**은 `첫글자***@도메인`. 로컬 부분이 `첫글자 + *`가 아니면(`ho***@`, `ab***z@`, `***@`) 서버가 거부한다.
4. 처리 위치: 본문, 문서 속성(작성자·제목·XMP·요약정보), 주석, 양식 필드 값, 책갈피, 미리보기 텍스트(HWP·HWPX), 파일명.
5. **덮개 사각형만 씌우지 않는다.** 아래 글자가 남는다. 글자 층에서 지운 파일로 새로 쓴다.
6. **증분 저장·문서 이력 없이 새 파일로 저장한다.** PDF는 "다른 이름으로 새로 저장"(증분 저장 금지 — 이전 판의 객체가 파일에 남는다), HWP는 "문서 이력 저장" 해제 후 새 파일로 저장한다. 서버는 PDF의 `/Prev`·`startxref` 2개 이상, HWP 이력·XML 템플릿·전자서명 비트가 있으면 `INCREMENTAL`/`HISTORY`로 제외한다.
7. **이미지 속 글자는 서버가 읽지 못한다.** 이미지 위주 문서(이미지가 있고 추출 글자가 거의 없음)는 서버가 `IMAGE_ONLY`로 제외한다. 첨부하지 말고 `notice.url`만 보낸다.
8. 첨부마다 `reported_tool`, `reported_version`, `reported_phones_removed`, `reported_landlines_masked`, `reported_emails_masked`, `reported_scope[]`(`body`·`meta`·`annot`·`form`·`bookmark`·`filename`)를 쓸 수 있다. 개수만 쓰고 값은 쓰지 않는다. **자기신고다.** 서버는 믿지 않고 판정에 쓰지 않으며 이력에도 남기지 않는다.

## 4. 2~4단계 — v3 설계 문안 (패치 아님)

**공통 순서** (security 중-5·상-4): 반입 호출마다 batchDir 하나를 아래 순서로 처리합니다.
1. `batchDir.toRealPath()`가 inbox 루트의 realPath 아래인지 확인합니다. `files/`가 symlink·reparse point(`isSymbolicLink()` 또는 `readAttributes(NOFOLLOW_LINKS).isOther()`)이면 배치를 거부합니다. `isOther()`가 junction에서 true인지는 Windows에서 `mklink /J`로 만든 시험 한 건이 있어야 합니다. 이는 security도 기억 의존이라 했습니다.
2. 선언된 파일마다 `NOFOLLOW_LINKS`로 열어 **서버 전용 임시 폴더로 복사**합니다. 이때 바이트를 세고(20 MiB 상한) sha256을 계산하며, 기본 데이터 스트림만 복사되므로 ADS가 떨어집니다. 이후 **검사·저장은 모두 사본에서** 합니다(TOCTOU 제거). sha256이 불일치하면 배치 422입니다.
3. 처리가 끝나면(422 포함) **checks가 있는 배치의 원본 폴더를 삭제**합니다. `archiveBatch`가 옮기는 보관본에는 **원본 파일이 아니라 다시 쓴 manifest**만 둡니다. 제외 check는 `{"check_id":..,"excluded":true}`로 다시 쓰고, 통과 check의 제목은 서버가 `scanExternal`을 통과시킨 값만 둡니다. records만 있는 v1 배치(`InboxApiTest` 경로)의 동작은 바꾸지 않습니다.
4. 저장 이름은 서버가 만듭니다: `corpus/ext_check/<CHECK_KEY>/<check_id>_<순번>.<매직 넘버로 정한 확장자>`. `check_id`는 형식 검사를 이미 통과한 값만 씁니다(중-1). 기관 id를 못 찾으면 그 check만 제외합니다.

**제외 사유 코드 (check 단위, 응답 `excluded[]`에는 `{check_id, reason, locator, count}`만 싣습니다)**: `PII_MOBILE`, `PII_LANDLINE`, `PII_EMAIL`, `PII_ID`, `ENCRYPTED`, `SCRIPT`, `EMBEDDED`, `INCREMENTAL`, `HISTORY`, `IMAGE_ONLY`, `PAGES_LIMIT`, `TIMEOUT`, `FORMAT_MISMATCH`, `PARSE_FAILED`, `DELETE_FAILED`, `XXE_REJECTED`, `ZIP_LIMIT`.

**PDF (pdfbox 2.0.29, `setupMixed`)**
- 거부: 암호화, `/JavaScript`·`/EmbeddedFiles`·XFA, 쪽수 200 초과, FileAttachment·RichMedia·3D 주석.
- `/OpenAction`은 `/S`가 JavaScript·Launch·URI·SubmitForm·ImportData·GoToR·GoToE일 때만 거부합니다(중-3 과잉 완화). 페이지·필드의 `/AA`도 같은 기준으로 거부합니다.
- 상-4: 트레일러에 `/Prev`가 있거나 `startxref`가 2개 이상이면 `INCREMENTAL`입니다.
- 추출 대상: 쪽별 본문, 문서 정보 전 항목·XMP, 주석 `/Contents`·`/T`·`/Subj`, AcroForm 값·기본값, 책갈피.

**HWPX (JDK zip·javax.xml)**
- zip 상한: 항목 500개, 항목당 50 MiB, 합계 100 MiB, 압축비 100:1. 바이트를 읽으며 세고 헤더 값은 믿지 않습니다.
- 항목명에 `..`·절대경로·`\`·`:`가 있으면 거부합니다. `mimetype`은 `application/hwp+zip`이어야 하고, `Scripts/*`·이미지가 아닌 `BinData/*`·중첩 zip·암호화 표시는 거부합니다.

**HWP (hwplib — 메인이 Maven Central에서 버전·전이 의존성·클래스 버전·poi 충돌 확인 뒤 반입, 하-3)**

- FileHeader 속성 비트(읽기 전에 거부): 1 암호, 2 배포용, 3 스크립트 저장, 4 DRM, 5 XMLTemplate, **6 문서 이력**, **7 전자서명**, 8 인증서 암호, 10 인증서 DRM. 비트 번호는 security가 기억 의존이라 했으므로 HWP 5.0 규격서와 대조한 뒤 상수로 확정합니다.

**DDL — `db/mysql/041_external_check.sql` / `db/oracle/041_external_check.sql`** (v2 §3-5에서 아래만 바뀝니다. 나머지는 v2와 같음)

CREATE TABLE EXT_CHECK_EXCL (
    BATCH_ID    VARCHAR(80) NOT NULL,
