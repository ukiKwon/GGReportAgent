### 3-3. `collector/httpclient.py` (신규, ⑦)
```python
"""표준 라이브러리 HTTP 클라이언트. TLS는 기본 검증 ON, 끄는 것은 호스트별 설정으로만."""
from __future__ import annotations

import socket
import ssl
import time
import urllib.error
import urllib.parse
import urllib.request

UA = ("Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
      "(KHTML, like Gecko) Chrome/124.0 Safari/537.36")


class FetchError(Exception):
    pass


class FetchHttpError(FetchError):
    """서버가 응답한 오류(4xx/5xx) — 스킴을 바꿔 재시도해도 소용없다."""


class _SameHostRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        if urllib.parse.urlsplit(newurl).hostname != urllib.parse.urlsplit(req.full_url).hostname:
            raise urllib.error.HTTPError(req.full_url, code, "다른 호스트로의 리다이렉트 차단", headers, fp)
        return super().redirect_request(req, fp, code, msg, headers, newurl)


class HttpClient:
    def __init__(self, timeout: float = 20.0, max_bytes: int = 20 * 1024 * 1024,
                 delay: float = 1.0, sleep=time.sleep) -> None:
        self.timeout, self.max_bytes, self.delay, self._sleep = timeout, max_bytes, delay, sleep
        self._first = True

    def get(self, url: str, verify: bool = True) -> tuple[bytes, str | None]:
        if urllib.parse.urlsplit(url).scheme not in ("http", "https"):
            raise FetchError("http/https만 허용합니다")
        if not self._first:
            self._sleep(self.delay)        # 호스트에 부담 주지 않는 간격
        self._first = False
        ctx = ssl.create_default_context() if verify else ssl._create_unverified_context()
        opener = urllib.request.build_opener(_SameHostRedirect, urllib.request.HTTPSHandler(context=ctx))
        req = urllib.request.Request(url, headers={
            "User-Agent": UA, "Accept": "*/*", "Accept-Language": "ko-KR,ko;q=0.9"})
        try:
            with opener.open(req, timeout=self.timeout) as resp:   # http.client = HTTP/1.1
                data = resp.read(self.max_bytes + 1)
                charset = resp.headers.get_content_charset()
        except urllib.error.HTTPError as exc:          # URLError보다 먼저
            raise FetchHttpError(f"HTTP {exc.code}") from exc
        except (urllib.error.URLError, ssl.SSLError, socket.timeout, OSError) as exc:
            raise FetchError(f"연결 실패: {type(exc).__name__} {str(getattr(exc, 'reason', exc))[:80]}") from exc
        if len(data) > self.max_bytes:
            raise FetchError("응답이 너무 큽니다")
        return data, charset
```

### 3-4. `collector/state.py` (② 상태)
위치 `data/collector/_state/state.json`. 형식은 JSON이고 임시파일에 쓴 뒤 `os.replace`한다(원자적, 사람이 읽고 diff 가능). `seen` 키는 `"{host}|{공고번호}"`다.
```json
{"version":1,"sources":{"eminwon":{
  "seen":{"gangnam.eminwon.seoul.kr|제2026-2459호":
     {"title":"…","posted_at":"2026-10-23","first_seen":"2026-11-11T10:30:00+09:00",
      "raw":"eminwon/gangnam.eminwon.seoul.kr/제2026-2459호","batch_id":null}},
  "runs":[{"started_at":"…","finished_at":"…","new":2,"skipped":1,"failed":[],"hosts_ok":72,"hosts_failed":{"x.go.kr":"연결 실패: …"}}]}}}
```
- **seen 표시는 원문 보관이 성공한 뒤에만** 한다(실패한 글은 다음 실행에 다시 시도).
- `batch_id`는 포장 때 채운다. null이면 아직 포장 전이다.
- `runs`는 최근 50개만 둔다.

```python
from __future__ import annotations

import json
import os
from pathlib import Path

STATE_VERSION = 1
MAX_RUNS = 50


class StateError(Exception):
    pass


class _Index:
    def __init__(self, state: "State", slug: str) -> None:
        self._state, self._slug = state, slug

    def is_seen(self, key: str) -> bool:
        return self._state.is_seen(self._slug, key)


class State:
    def __init__(self, path: Path | str) -> None:
        self.path = Path(path)
        self.data: dict = {"version": STATE_VERSION, "sources": {}}
        if self.path.is_file():
            try:
                data = json.loads(self.path.read_text(encoding="utf-8"))
            except (json.JSONDecodeError, UnicodeDecodeError) as exc:
                raise StateError(f"상태 파일을 읽을 수 없습니다: {self.path.name}") from exc
            if data.get("version") != STATE_VERSION:
                raise StateError(f"지원하지 않는 상태 버전: {data.get('version')!r}")
            self.data = data

    def _src(self, slug: str) -> dict:
        return self.data["sources"].setdefault(slug, {"seen": {}, "runs": []})

    def index(self, slug: str) -> _Index:
        return _Index(self, slug)

    def is_seen(self, slug: str, key: str) -> bool:
        return key in self._src(slug)["seen"]

    def mark_seen(self, slug: str, key: str, **info) -> None:
        self._src(slug)["seen"][key] = info

    def set_batch(self, slug: str, key: str, batch_id: str) -> None:
        self._src(slug)["seen"][key]["batch_id"] = batch_id

    def with_raw(self, slug: str, include_packed: bool) -> list[tuple[str, dict]]:
        return sorted((k, v) for k, v in self._src(slug)["seen"].items()
                      if v.get("raw") and (include_packed or not v.get("batch_id")))

    def add_run(self, slug: str, run: dict) -> None:
        runs = self._src(slug)["runs"]
        runs.append(run)
        del runs[:-MAX_RUNS]

    def save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        tmp = self.path.with_suffix(".tmp")
        tmp.write_text(json.dumps(self.data, ensure_ascii=False, indent=1) + "\n", encoding="utf-8")
        os.replace(tmp, self.path)
```

### 3-5. `collector/store.py` (③ 원문 보관)
규칙: `_raw/<source>/<host>/<공고번호>/`에 `detail.html`(받은 바이트 그대로), `meta.json`(목록 행·인코딩·sha256·기관 매핑·첨부 목록·첨부 오류·fetched_at), `attachments/NN_<정제된 이름>`을 둔다. 임시 폴더에 쓰고 rename한다. 같은 공고를 다시 받으면 원문 최신본으로 덮어쓴다(배치가 아니라 원문 저장소이므로).
```python
from __future__ import annotations

import hashlib
import json
import re
import shutil
from pathlib import Path

_UNSAFE = re.compile(r'[\x00-\x1f/\\:*?"<>|]')


def safe_name(name: str, limit: int = 100) -> str:
    base = _UNSAFE.sub("_", name.replace("\\", "/").split("/")[-1]).strip(". ")
    if not base:
        return "file"
    stem, dot, ext = base.rpartition(".")
    if dot and len(base) > limit:
        base = stem[: max(1, limit - len(ext) - 1)] + "." + ext
    return base[:limit]


class RawStore:
    def __init__(self, root: Path | str) -> None:
        self.root = Path(root)

    def dir_for(self, source: str, host: str, number: str) -> Path:
        path = (self.root / safe_name(source) / safe_name(host) / safe_name(number)).resolve()
        if not path.is_relative_to(self.root.resolve()):
            raise ValueError("원문 경로가 루트 밖입니다")
        return path

    def rel(self, path: Path) -> str:
        return path.resolve().relative_to(self.root.resolve()).as_posix()

    def write(self, source: str, host: str, number: str, detail: bytes,
              files: list[tuple[str, bytes]], meta: dict) -> Path:
        final = self.dir_for(source, host, number)
        tmp = final.parent / f".tmp-{final.name}"
        shutil.rmtree(tmp, ignore_errors=True)
        (tmp / "attachments").mkdir(parents=True)
        (tmp / "detail.html").write_bytes(detail)
        stored = []
        for i, (name, blob) in enumerate(files, 1):
            fname = f"{i:02d}_{safe_name(name)}"
            (tmp / "attachments" / fname).write_bytes(blob)
            stored.append({"name": name, "stored": f"attachments/{fname}",
                           "size": len(blob), "sha256": hashlib.sha256(blob).hexdigest()})
        meta = dict(meta, attachments=stored, detail_sha256=hashlib.sha256(detail).hexdigest())
        (tmp / "meta.json").write_text(json.dumps(meta, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
        if final.exists():
            shutil.rmtree(final)
        tmp.rename(final)
        return final
```

```python
    if "금고" not in t:
```

`extract/hwpx.py`
```python
import re
import zipfile
import io
import html

MAX_UNCOMPRESSED = 30 * 1024 * 1024
_SEC = re.compile(r"^Contents/section(\d+)\.xml$")
_T = re.compile(r"<hp:t\b[^>]*>(.*?)</hp:t>", re.S)
_TAG = re.compile(r"<[^>]+>")


class ExtractError(Exception):
    pass


def hwpx_text(data: bytes) -> str:
    """section*.xml의 <hp:t> 텍스트를 문단 단위 줄로 합친다. XML 파서를 쓰지 않는다(외부 엔티티 위험 회피)."""
    try:
        zf = zipfile.ZipFile(io.BytesIO(data))
    except zipfile.BadZipFile as exc:
        raise ExtractError("hwpx(zip)가 아닙니다") from exc
    secs = sorted((int(m.group(1)), i) for i in zf.infolist() if (m := _SEC.match(i.filename)))
    if not secs:
        raise ExtractError("Contents/section*.xml이 없습니다")
    lines: list[str] = []
    for _, info in secs:
        if info.file_size > MAX_UNCOMPRESSED:
            raise ExtractError("hwpx 구간이 너무 큽니다")
        xml = zf.read(info).decode("utf-8", errors="replace")
        for para in xml.split("</hp:p>"):
            text = "".join(html.unescape(_TAG.sub("", t)) for t in _T.findall(para)).strip()
            if text:
                lines.append(text)
    return "\n".join(lines)
```

```python
_D = r"(\d{4})\s*[.\-년]\s*(\d{1,2})\s*[.\-월]\s*(\d{1,2})\s*(?:일|\.)?"
PERIOD_RE = re.compile(r"(?:약정|계약|운영)\s*기간[^0-9]{0,15}" + _D + r"\s*[~∼～–\-]\s*" + _D)
DATE_RE = re.compile(_D)
DEADLINE_ANCHOR = re.compile(r"(?:접수|제출)\s*기간")


def iso(y, m, d) -> str | None:
    try:
        return datetime.date(int(y), int(m), int(d)).isoformat()
    except ValueError:
        return None


def find_period(text: str):
    """(시작, 끝, 근거문장) 또는 None. 날짜가 달력에 없으면 버린다."""
    m = PERIOD_RE.search(text)
    if not m:
        return None
    start, end = iso(*m.group(1, 2, 3)), iso(*m.group(4, 5, 6))
    if not (start and end) or end < start:
        return None
    return start, end, m.group(0)[:120]


def find_deadline(text: str):
    m = DEADLINE_ANCHOR.search(text)
    if not m:
        return None
    window = text[m.start(): m.end() + 80]
    dates = [iso(*d.groups()) for d in DATE_RE.finditer(window)]
    dates = [d for d in dates if d]
    return (dates[1], window[:120]) if len(dates) >= 2 else None


def years_between(start: str, end: str) -> int:
    days = (datetime.date.fromisoformat(end) - datetime.date.fromisoformat(start)).days + 1
    return max(1, round(days / 365.25))
```

```python
def _html_text(raw: bytes, encoding: str) -> str:
    import html
    enc = "utf-8" if encoding.endswith("-replace") else encoding
    text = raw.decode(enc, errors="replace")
    return re.sub(r"\s+", " ", html.unescape(_TAG.sub(" ", _STRIP_BLOCK.sub(" ", text)))).strip()


def build_candidate(kind: str, posted_at: str | None, texts: list[tuple[str, str]]) -> dict:
    fields: dict = {}
    basis: list[dict] = []
    for origin, text in texts:                       # hwpx 우선, 상세 본문은 폴백
        found = find_period(text)
        if found:
            start, end, sentence = found
            fields.update(contract_start=start, contract_end=end, term=years_between(start, end))
            basis += [{"field": "contract_start", "text": sentence, "from": origin},
                      {"field": "contract_end", "text": sentence, "from": origin},
                      {"field": "term", "text": sentence, "from": origin, "derived": True}]
            break
    for origin, text in texts:
        found = find_deadline(text)
        if found:
            fields["deadline_at"] = found[0]
            basis.append({"field": "deadline_at", "text": found[1], "from": origin})
            break
    if ROLE[kind] == "회차종료" and posted_at:
        fields["last_bid"] = posted_at
        basis.append({"field": "last_bid", "text": f"공고 게시일 {posted_at}", "from": "목록 게시일"})
    return {"status": "후보", "kind": kind, "role": ROLE[kind], "fields": fields, "basis": basis}


def notice_from_raw(raw_dir: Path, rel: str) -> CollectedNotice:
    meta = json.loads((raw_dir / "meta.json").read_text(encoding="utf-8"))
    kind = classify(meta["title"])
    texts: list[tuple[str, str]] = []
    atts: list[AttachmentRef] = []
    notes: list[str] = []
    for a in meta.get("attachments", []):
        blob = (raw_dir / a["stored"]).read_bytes()
        atts.append(AttachmentRef(filename=Path(a["stored"]).name, data=blob))
        if a["name"].lower().endswith(".hwpx"):
            try:
                texts.append((f"hwpx:{a['name']}", hwpx_text(blob)))
            except ExtractError as exc:
                notes.append(f"{a['name']}: {exc}")
    texts.append(("상세 본문", _html_text((raw_dir / "detail.html").read_bytes(), meta.get("encoding", "utf-8"))))
    inst = meta["institution"]
    candidate = build_candidate(kind, meta.get("posted_at"), texts) if is_candidate_kind(kind) else None
    return CollectedNotice(
        notice_id=f"{meta['host']}#{meta['number']}",
        title=meta["title"],
        institution_name_ko=inst["name_ko"],
        evidence_url=meta["detail_url"],
        posted_at=meta.get("posted_at"),
        confirmed=False,                       # 자동 추출값은 항상 '예상' — 사람이 확인해야 확정
        institution_type=inst.get("type"),
        region_code=inst.get("region_code"),
        attachments=tuple(atts),
        notice_info={"kind": kind, "number": meta["number"], "host": meta["host"],
                     "mgt_no": meta.get("native_id"), "department": meta.get("department"),
                     "raw_path": rel},
        candidate=candidate,
        evidence_note="; ".join(notes) or None,
    )
```

설계 결정 3가지(근거는 SCHEMA ④ "추측 금지"):
1. `notice_id`는 `"{host}#{공고번호}"`다. eminwon 공고번호는 호스트마다 겹치는데 검증기가 배치 안 `notice_id` 중복을 막기 때문이다. 원문 번호는 `notice.number`에 둔다. `#`을 쓴 것은 `files/<notice_id>_…` 파일명이 Windows에서 안전하고(`:`·`/`는 안 됨) `UNSAFE_FILENAME`에 안 걸리기 때문이다.
2. `schedule`에는 `posted_at`과 `confidence:"예상"`만 넣는다. 약정기간·회차 후보는 `candidate`에만 둔다. 자동 추출값이 CSV(`입찰예상일`)를 거쳐 서버 upsert로 가면 확정 데이터를 덮어쓸 수 있다. 이 매핑은 기획이 정할 일이다(§6).
3. `term`은 약정기간 길이에서 계산한 값이라 `basis.derived=true`로 표시한다.

```python
NOTICE_KINDS = ("계획", "재공고", "결과", "지정", "기타")
CANDIDATE_DATE_KEYS = ("contract_start", "contract_end", "last_bid", "deadline_at")


def _validate_notice_candidate(record: dict, where: str) -> list[str]:
    """notice·candidate는 선택 칸(schema_version 1 유지). 있으면 형식을 지켜야 한다."""
    errors: list[str] = []
    notice = record.get("notice")
    if notice is not None:
        if not isinstance(notice, dict):
            errors.append(f"{where}.notice: 객체가 아닙니다")
        elif notice.get("kind") is not None and notice["kind"] not in NOTICE_KINDS:
            errors.append(f"{where}.notice.kind: 허용값이 아닙니다 ({notice['kind']!r})")
    cand = record.get("candidate")
    if cand is None:
        return errors
    if not isinstance(cand, dict):
        return errors + [f"{where}.candidate: 객체가 아닙니다"]
    if cand.get("status") != "후보":
        errors.append(f"{where}.candidate.status: '후보'여야 합니다 ({cand.get('status')!r})")
    fields = cand.get("fields")
    basis = cand.get("basis")
    if not isinstance(fields, dict) or not isinstance(basis, list):
        return errors + [f"{where}.candidate: fields(객체)·basis(배열)가 필요합니다"]
    for key in CANDIDATE_DATE_KEYS:
        if key in fields and not DATE_RE.match(str(fields[key])):
            errors.append(f"{where}.candidate.fields.{key}: YYYY-MM-DD가 아닙니다 ({fields[key]!r})")
    if "term" in fields and not (isinstance(fields["term"], int) and fields["term"] > 0):
        errors.append(f"{where}.candidate.fields.term: 양의 정수여야 합니다")
    based = {b.get("field") for b in basis if isinstance(b, dict) and isinstance(b.get("text"), str)}
    for key in fields:                       # 근거 없는 값 금지
        if key not in based:
            errors.append(f"{where}.candidate.fields.{key}: basis(근거)가 없습니다")
    return errors
```

```python
from __future__ import annotations

import html
import re
from dataclasses import dataclass

from collector.extract.terms import iso

ROW_RE = re.compile(r"<tr\b[^>]*>(.*?)</tr>", re.S | re.I)
CELL_RE = re.compile(r"<td\b[^>]*>(.*?)</td>", re.S | re.I)
ANCHOR_RE = re.compile(
    r"<a\b[^>]*?(?:searchDetail\(\s*['\"]?(\d+)['\"]?\s*\)|not_ancmt_mgt_no=(\d+))[^>]*>(.*?)</a>",
    re.S | re.I)
NUMBER_RE = re.compile(r"제\s*(\d{4})\s*-\s*(\d+)\s*호")
DATE_RE = re.compile(r"(\d{4})\s*[-./]\s*(\d{1,2})\s*[-./]\s*(\d{1,2})")
DOWNLOAD_RE = re.compile(
    r"goDownLoad\(\s*'([^']*)'\s*,\s*'([^']*)'\s*,\s*'([^']*)'\s*\)")
_TAG = re.compile(r"<[^>]+>")


@dataclass(frozen=True)
class ListRow:
    mgt_no: str
    title: str
    number: str | None
    department: str | None
    posted_at: str | None


@dataclass(frozen=True)
class AttachmentLink:
    user_file_nm: str
    sys_file_nm: str
    file_path: str


def clean(fragment: str) -> str:
    return re.sub(r"\s+", " ", html.unescape(_TAG.sub(" ", fragment))).strip()


def decode_body(data: bytes, header_charset: str | None = None) -> tuple[str, str]:
    """'공고'가 디코딩되는 인코딩을 쓴다. 못 찾으면 utf-8 대체문자."""
    for enc in ([header_charset] if header_charset else []) + ["utf-8", "euc-kr", "cp949"]:
        try:
            text = data.decode(enc)
        except (UnicodeDecodeError, LookupError):
            continue
        if "공고" in text:
            return text, enc
    return data.decode("utf-8", errors="replace"), "utf-8-replace"


def parse_list(text: str) -> list[ListRow]:
    rows: list[ListRow] = []
    for row in ROW_RE.findall(text):
        anchor = ANCHOR_RE.search(row)
        if not anchor:
            continue                                   # 헤더·빈 행
        title = clean(anchor.group(3))
        cells = [clean(c) for c in CELL_RE.findall(row)]
        others = [c for c in cells if c and title not in c]
        joined = " ".join(others)
        num = NUMBER_RE.search(joined)
        date = DATE_RE.search(joined)
        dept = next((c for c in others if not NUMBER_RE.search(c) and not DATE_RE.search(c)
                     and not c.isdigit()), None)
        rows.append(ListRow(
            mgt_no=anchor.group(1) or anchor.group(2),
            title=title,
            number=f"제{num.group(1)}-{num.group(2)}호" if num else None,
            department=dept,
            posted_at=iso(*date.groups()) if date else None,
        ))
    return rows


def parse_attachments(text: str) -> list[AttachmentLink]:
    return [AttachmentLink(*m.groups()) for m in DOWNLOAD_RE.finditer(text)]
```

```python
from __future__ import annotations

import datetime

from typing import Iterator
from urllib.parse import quote

from collector.extract.classify import classify, is_candidate_kind
from collector.httpclient import FetchError, FetchHttpError, HttpClient
from collector.sources.base import NoticeRef, RawNotice, RunReport, SourceError, register
from collector.sources.eminwon_parse import decode_body, parse_attachments, parse_list

LIST_PATH = "/emwp/gov/mogaha/ntis/web/ofr/action/OfrAction.do"
FILE_PATH = "/emwp/jsp/ofr/FileDown.jsp"
DEFAULT_HOSTS_FILE = Path(__file__).resolve().parents[1] / "config" / "eminwon_hosts.json"
HOST_RE = re.compile(r"^[a-z0-9]([a-z0-9.-]*[a-z0-9])?$")
ALLOWED_EXT = (".hwpx", ".pdf", ".hwp")


@dataclass(frozen=True)
class HostEntry:
    host: str
    key: str                       # "region|name" — institutions.js 기관 key
    name_ko: str
    region_code: str | None = None
    type: str | None = None
    tls_verify: bool = True        # 끄는 것은 호스트 단위 명시 설정으로만
    schemes: tuple[str, ...] = ("https", "http")


@dataclass(frozen=True)
class EminwonConfig:
    hosts_file: Path = DEFAULT_HOSTS_FILE
    keyword: str = "금고"
    page_size: int = 50
    max_pages: int = 5
    max_attachments: int = 5
    delay_sec: float = 1.0
    timeout: float = 20.0


def load_hosts(path: Path) -> tuple[HostEntry, ...]:
    try:
        raw = json.loads(Path(path).read_text(encoding="utf-8"))
        items = raw["hosts"]
    except (OSError, ValueError, KeyError, TypeError) as exc:
        raise SourceError(f"호스트 설정을 읽을 수 없습니다: {Path(path).name}") from exc
    out: dict[str, HostEntry] = {}
    for item in items:
        host = str(item.get("host", "")).strip().lower()
        schemes = tuple(item.get("schemes") or ("https", "http"))
        if not HOST_RE.match(host) or not set(schemes) <= {"https", "http"}:
            raise SourceError(f"호스트 설정이 올바르지 않습니다: {host!r}")
        out[host] = HostEntry(host, item.get("key") or host, item.get("name_ko") or host,
                              item.get("region_code"), item.get("type"),
                              bool(item.get("tls_verify", True)), schemes)
    return tuple(out.values())


def _query(params) -> str:
    return "&".join(f"{k}={quote(v, safe=',')}" for k, v in params)


class EminwonSource:
    slug = "eminwon"
    name_ko = "eminwon 고시공고(금고)"
    base_url = "urn:eminwon:multi-host"

    def __init__(self, config: EminwonConfig | None = None, http=None,
                 hosts: tuple[HostEntry, ...] | None = None) -> None:
        self.config = config or EminwonConfig()
        self.http = http or HttpClient(timeout=self.config.timeout, delay=self.config.delay_sec)
        self._hosts = hosts
        self._scheme: dict[str, str] = {}

    # --- 설정 주입 ---
    def with_config(self, hosts_file=None, http=None, hosts=None, **config) -> "EminwonSource":
        cfg = replace(self.config, **config)
        if hosts_file:
            cfg = replace(cfg, hosts_file=Path(hosts_file))
        return EminwonSource(cfg, http, hosts)

    def hosts(self) -> tuple[HostEntry, ...]:
        if self._hosts is None:
            self._hosts = load_hosts(self.config.hosts_file)
        return self._hosts

    def _select(self, wanted: list[str] | None) -> list[HostEntry]:
        if not wanted:
            return list(self.hosts())
        by = {}
        for e in self.hosts():
            by[e.host] = e
            by[e.key] = e
        unknown = [w for w in wanted if w not in by]
        if unknown:
            raise SourceError(f"설정에 없는 호스트/기관: {', '.join(unknown)}")
        return list(dict.fromkeys(by[w] for w in wanted))

    def _entry(self, host: str) -> HostEntry:
        return self._select([host])[0]

    # --- 기존 Source 계약: 이 소스는 CLI 전용 ---
    def fetch(self):
        raise SourceError("eminwon은 CLI로 실행합니다: python -m collector run --source eminwon")

    # --- HTTP ---
    def _get(self, entry: HostEntry, path_query: str):
        order = list(entry.schemes)
        pref = self._scheme.get(entry.host)
        if pref in order:
            order.remove(pref)
            order.insert(0, pref)
        last: FetchError | None = None
        for scheme in order:
            url = f"{scheme}://{entry.host}{path_query}"
            try:
                data, charset = self.http.get(url, verify=entry.tls_verify)
            except FetchHttpError:
                raise                      # 서버가 응답함 — 스킴 재시도 무의미
            except FetchError as exc:
                last = exc
                continue
            self._scheme[entry.host] = scheme
            return url, data, charset
        raise last or FetchError("연결 실패")

    def _list_query(self, page: int) -> str:
        c = self.config
        return _query([
            ("context", "NTIS"), ("jndinm", "OfrNotAncmtEJB"), ("method", "selectListOfrNotAncmt"),
            ("methodnm", "selectListOfrNotAncmtHomepage"), ("homepage_pbs_yn", "Y"), ("subCheck", "Y"),
            ("not_ancmt_se_code", "01,02,03,04,05,06,07"), ("not_ancmt_sj", c.keyword),
            ("s_gubun", "글제목"), ("ofr_pageSize", str(c.page_size)), ("pageIndex", str(page)),
            ("initValue", "Y"), ("countYn", "Y"), ("dep_se", "%")])

    @staticmethod
    def _detail_query(mgt_no: str) -> str:
        return _query([("context", "NTIS"), ("jndinm", "OfrNotAncmtEJB"),
                       ("method", "selectOfrNotAncmt"), ("methodnm", "selectOfrNotAncmtRegst"),
                       ("not_ancmt_mgt_no", mgt_no)])

    def _list_host(self, entry: HostEntry):
        rows = []
        for page in range(1, self.config.max_pages + 1):
            _, data, charset = self._get(entry, f"{LIST_PATH}?{self._list_query(page)}")
            page_rows = parse_list(decode_body(data, charset)[0])
            rows.extend(page_rows)
            if len(page_rows) < self.config.page_size:
                break
        kw = self.config.keyword
        return [r for r in rows if kw in r.title.replace(" ", "")]

    # --- IncrementalSource ---
    def list_new(self, seen, since=None, report=None, hosts=None) -> Iterator[NoticeRef]:
        report = report if report is not None else RunReport()
        for entry in self._select(hosts):
            try:
                rows = self._list_host(entry)
            except FetchError as exc:
                report.hosts_failed[entry.host] = str(exc)[:200]
                continue
            report.hosts_ok.append(entry.host)
            scheme = self._scheme.get(entry.host, "https")
            local: set[str] = set()
            for r in rows:
                if since and r.posted_at and r.posted_at < since:
                    continue
                number = r.number or f"mgt{r.mgt_no}"
                ref = NoticeRef(
                    source=self.slug, host=entry.host, number=number, title=r.title,
                    detail_url=f"{scheme}://{entry.host}{LIST_PATH}?{self._detail_query(r.mgt_no)}",
                    posted_at=r.posted_at, department=r.department, native_id=r.mgt_no)
                if ref.key in local or seen.is_seen(ref.key):
                    continue
                local.add(ref.key)
                yield ref

    def accepts(self, ref: NoticeRef) -> bool:
        return is_candidate_kind(classify(ref.title))

    def fetch_raw(self, ref: NoticeRef, store) -> RawNotice:
        entry = self._entry(ref.host)
        try:
            _, data, charset = self._get(entry, f"{LIST_PATH}?{self._detail_query(ref.native_id or '')}")
        except FetchError as exc:
            raise SourceError(f"상세 조회 실패({ref.host}): {exc}") from exc
        text, encoding = decode_body(data, charset)
        files: list[tuple[str, bytes]] = []
        errors: list[dict] = []
        for link in parse_attachments(text)[: self.config.max_attachments]:
            name = link.user_file_nm
            if not name.lower().endswith(ALLOWED_EXT):
                errors.append({"name": name, "error": "허용하지 않는 확장자"})
                continue
            path_q = (f"{FILE_PATH}?user_file_nm={quote(link.user_file_nm, safe='')}"
                      f"&sys_file_nm={quote(link.sys_file_nm, safe='')}"
                      f"&file_path={quote(link.file_path, safe='')}")
            try:
                _, blob, _ = self._get(entry, path_q)
            except FetchError as exc:
                errors.append({"name": name, "error": str(exc)[:120]})
                continue
            files.append((name, blob))
        meta = {
            "source": self.slug, "host": ref.host, "number": ref.number, "title": ref.title,
            "posted_at": ref.posted_at, "department": ref.department, "native_id": ref.native_id,
            "detail_url": ref.detail_url, "encoding": encoding,
            "institution": {"key": entry.key, "name_ko": entry.name_ko,
                            "region_code": entry.region_code, "type": entry.type},
            "fetched_at": datetime.datetime.now().astimezone().isoformat(timespec="seconds"),
            "attachment_errors": errors,
        }
        raw_dir = store.write(self.slug, ref.host, ref.number, data, files, meta)
        return RawNotice(ref=ref, raw_dir=raw_dir, rel=store.rel(raw_dir))


register(EminwonSource())
```

- 인코딩: `decode_body`가 헤더 charset → utf-8 → euc-kr → cp949 순으로 '공고'가 나오는 첫 인코딩을 쓴다.
- 기관명은 설정의 `name_ko`를 쓴다. 공고 본문에서 뽑지 않는다.

### 3-11. `collector/pipeline.py`
```python
from __future__ import annotations

import datetime
from pathlib import Path

from collector.batch import write_batch
from collector.extract.build import notice_from_raw
from collector.sources.base import RunReport, SourceError


def _now_iso() -> str:
    return datetime.datetime.now().astimezone().isoformat(timespec="seconds")


def collect(source, state, store, *, since=None, hosts=None, limit=None, baseline=False) -> dict:
    slug, report, started = source.slug, RunReport(), _now_iso()
    new: list[str] = []
    skipped = 0
    failed: list[dict] = []
    for ref in source.list_new(state.index(slug), since, report, hosts):
        info = dict(title=ref.title, posted_at=ref.posted_at, first_seen=_now_iso(), batch_id=None)
        if baseline:               # 현재 목록을 '본 것'으로만 표시 — 첫 실행의 과거 글 폭주 방지
            state.mark_seen(slug, ref.key, baseline=True, **info)
            continue
        if not source.accepts(ref):
            state.mark_seen(slug, ref.key, skipped=True, **info)
            skipped += 1
            continue
        if limit is not None and len(new) >= limit:
            break
        try:
            raw = source.fetch_raw(ref, store)
        except SourceError as exc:
            failed.append({"key": ref.key, "error": str(exc)[:200]})
            continue
        state.mark_seen(slug, ref.key, raw=raw.rel, **info)
        state.save()
        new.append(ref.key)
    summary = {"source": slug, "new": len(new), "skipped": skipped, "failed": failed,
               "hosts_ok": len(report.hosts_ok), "hosts_failed": report.hosts_failed}
    state.add_run(slug, dict(summary, started_at=started, finished_at=_now_iso(),
                             baseline=baseline, since=since))
    state.save()
    return summary


def extract_and_pack(source, state, store, out_root, *, redo_all=False, now=None) -> dict:
    slug = source.slug
    notices, keys, errors = [], [], []
    for key, info in state.with_raw(slug, include_packed=redo_all):
        try:
            notices.append(notice_from_raw(store.root / info["raw"], info["raw"]))
            keys.append(key)
        except (OSError, ValueError, KeyError, SourceError) as exc:
            errors.append({"key": key, "error": f"{type(exc).__name__}: {str(exc)[:150]}"})
    if not notices:
        return {"batch_id": None, "records": 0, "extract_errors": errors}
    result = write_batch(source, notices, out_root, now=now)   # 자기검사 실패 시 배치를 남기지 않고 예외
    for key in keys:
        state.set_batch(slug, key, result.batch_id)
    state.save()
    return {"batch_id": result.batch_id, "records": result.record_count, "extract_errors": errors}
```
재추출은 `extract --all`이다. 원문(`_raw`)만 읽으므로 네트워크가 필요 없다. 새 배치가 만들어지고 SCHEMA의 "나중 배치가 이긴다"를 따른다.

```python
def _lock(path: Path):
    path.parent.mkdir(parents=True, exist_ok=True)
    try:
        return os.open(path, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
    except FileExistsError:
        return None


def main(argv=None, http=None) -> int:
    args = _parser().parse_args(argv)
    if args.cmd == "sources":
        print(json.dumps(list_sources(), ensure_ascii=False))
        return 0
    if getattr(args, "since", None) and not DATE_RE.match(args.since):
        print("--since는 YYYY-MM-DD여야 합니다", file=sys.stderr)
        return 2
    try:
        source = get_source(args.source)
    except KeyError:
        print(f"등록되지 않은 소스: {args.source}", file=sys.stderr)
        return 2
    if not hasattr(source, "list_new"):
        print("증분 소스가 아닙니다(웹 API /collect 사용)", file=sys.stderr)
        return 2
    opts = {k: v for k, v in (("hosts_file", getattr(args, "hosts_file", None)), ("http", http)) if v}
    source = source.with_config(**opts)

    root = Path(args.out_root)
    lock_path = root / "_state" / "run.lock"
    fd = _lock(lock_path)
    if fd is None:
        print(f"다른 실행이 진행 중입니다(비정상 종료였다면 {lock_path} 삭제)", file=sys.stderr)
        return 4
    try:
        state, store = State(root / "_state" / "state.json"), RawStore(root / "_raw")
        out: dict = {}
        code = 0
        if args.cmd in ("run", "collect"):
            hosts = [h.strip() for h in args.hosts.split(",") if h.strip()] if args.hosts else None
            out["collect"] = collect(source, state, store, since=args.since, hosts=hosts,
                                     limit=args.limit, baseline=args.baseline)
            if out["collect"]["hosts_failed"]:

            out["pack"] = extract_and_pack(source, state, store, root, redo_all=args.all)
        print(json.dumps(out, ensure_ascii=False))
        return code
    except (SourceError, StateError, BatchError) as exc:
        print(f"실패: {exc}", file=sys.stderr)
        return 1
    finally:
        os.close(fd)
        lock_path.unlink(missing_ok=True)
```

### 3-13. 호스트 목록 (⑨) `tools/extract_eminwon_hosts.py`
```python
"""frontend/data/institutions.js의 eminwon(/emwp/) URL에서 호스트 설정을 만든다. 휴리스틱이라 결과를 사람이 본다."""
import json
import re
import sys
from pathlib import Path

URL = re.compile(r"(https?)://([A-Za-z0-9.-]+)/emwp/")
NAME = re.compile(r'name\s*:\s*"([^"]+)"')
TYPE = re.compile(r'type\s*:\s*"([^"]+)"')
REGION = re.compile(r'region\s*:\s*"(\d+)"')
KEYED = re.compile(r'^\s*"(\d{5})"\s*:\s*\{\s*//\s*([^\s—-]+)')


def main(src: str, dst: str | None) -> int:
    text = Path(src).read_text(encoding="utf-8")
    current = None                       # (name, type, region, approx)
    hosts: dict[str, dict] = {}
    for line in text.splitlines():
        k = KEYED.match(line)
        if k:
            current = (k.group(2), "지자체", k.group(1)[:2], True)
        n = NAME.search(line)
        if n:
            rest = text[text.find(line): text.find(line) + 400]
            t, r = TYPE.search(rest), REGION.search(rest)
            current = (n.group(1), t.group(1) if t else None, r.group(1) if r else None, False)
        for scheme, host in URL.findall(line):
            host = host.lower()
            entry = hosts.setdefault(host, {"host": host, "schemes": [], "tls_verify": True})
            if scheme not in entry["schemes"]:
                entry["schemes"].append(scheme)
            if current and ("name_ko" not in entry or (entry.get("approx") and not current[3])):
                entry.update(name_ko=current[0], type=current[1], region_code=current[2],
                             key=f"{current[2] or ''}|{current[0]}", approx=current[3])
    out = []
    for host in sorted(hosts):
        e = hosts[host]
        e["schemes"] = sorted(e["schemes"], key=lambda s: s != "https")   # https 우선, http 폴백
        if "http" in e["schemes"] and "https" not in e["schemes"]:
            e["schemes"] = ["http"]
        e.pop("approx", None)
        out.append(e)
    unresolved = [h["host"] for h in out if "key" not in h]
    print(f"호스트 {len(out)}개(기대 73), 기관 미확정 {len(unresolved)}: {unresolved}", file=sys.stderr)
    payload = json.dumps({"hosts": out}, ensure_ascii=False, indent=1) + "\n"
    if dst:
        Path(dst).write_text(payload, encoding="utf-8")
    else:
        sys.stdout.write(payload)
    return 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1] if len(sys.argv) > 1 else "frontend/data/institutions.js",
                  sys.argv[2] if len(sys.argv) > 2 else None))
```
실행 명령: `python tools/extract_eminwon_hosts.py frontend/data/institutions.js collector/config/eminwon_hosts.json`. 개수가 73과 다르거나 `key`가 없는 호스트가 나오면 수동으로 보정한다. `name`/`region`이 institutions.js에서 같은 줄·근처 400자 안에 있다는 가정이고, 노원처럼 주석 속 URL은 잘못 잡힐 수 있다. 의심 호스트는 `tls_verify:false`를 손으로 켠다(실패 시 `hosts_failed` 메시지가 SSL이면 켠다).

호출자가 실행할 명령(레포 루트):
- `python -m pytest collector -q`
- `python -m collector sources` → fixture·eminwon이 보여야 한다.
- 실측(네트워크, **쓰기: `data/collector/` 생성**, 승인 뒤): `python -m collector run --source eminwon --hosts gangnam.eminwon.seoul.kr --since 2026-06-01`

## 5. 보안·위험·되돌리는 법
- 접근 제어: 이 변경은 DMZ 수집기(Python) CLI이고 망 안 server·`/menus`·골든·`roles.js`는 건드리지 않는다. 웹 API `/collect`에서 eminwon을 부르면 `fetch()`가 `SourceError`를 내 422로 응답한다(외부 수집이 API로 트리거되는 경로를 새로 열지 않는다). 인증은 수집기에 없다(기존과 동일, 포트·네트워크 경계가 방어).
- 아웃바운드(SSRF 방지): 호스트는 `HOST_RE` 검증을 통과한 설정 파일 값뿐이고 URL은 항상 설정 호스트+고정 경로로 만든다. 다운로드 링크는 페이지에서 인자만 가져와 같은 호스트에 붙인다. 리다이렉트는 같은 호스트만 허용한다. 확장자는 hwpx·pdf·hwp만, 크기는 20MB 상한, 첨부는 5개 상한이다.
- TLS: 검증이 기본이고 호스트별 `tls_verify:false` 명시 설정으로만 끈다(조용한 다운그레이드 없음). http 폴백은 스킴 목록에 있는 호스트만 하며 공개 공고 조회라 비밀값이 오가지 않는다.
- 파일 안전: `safe_name`이 경로 구분자·제어문자를 제거하고 `dir_for`가 루트 이탈을 막는다. hwpx는 XML 파서를 쓰지 않고(외부 엔티티·billion laughs 회피) 정규식으로 읽으며 30MB 상한이 있다.
- 로그·산출물: 요청 쿼리·본문은 로그에 남기지 않고, `meta.json`에는 공개 공고 메타만 있다. 개인정보는 다루지 않는다(공고에 담당자 연락처가 있으면 `detail.html` 원문에만 있고 candidate에는 안 싣는다).
- 값은 후보다: 자동 추출은 항상 `confidence:"예상"`·`candidate.status:"후보"`다. 망 안으로 확정 데이터를 덮어쓰지 않으며, 사람(data-auditor)이 원문(`raw_path`·첨부)을 열어 반영한다.
- 운영 위험: ① 호스트 73개를 1초 간격으로 순차 호출하면 한 번에 수 분이 걸린다. ② `run.lock`은 비정상 종료 시 남으므로 메시지대로 삭제한다. ③ 상태 파일이 손상되면 `StateError`로 멈춘다(자동 복구 안 함, 파일을 지우면 `--baseline`부터 다시). ④ 분 단위 `batch_id` 충돌은 기존 동작(`BatchError`)이며 같은 분에 `run`을 두 번 돌리면 두 번째가 실패한다.
- 되돌리는 법: 신규 파일을 삭제하고 `base.py`·`batch.py`·`batch_schema.py`·`sources/__init__.py` 변경을 되돌리면 된다. 기존 계약은 선택 필드 추가뿐이라 이미 만든 배치는 영향이 없고 `data/collector/_raw`·`_state`는 삭제해도 된다(원문은 다시 받을 수 있지만, 게재 종료로 사라진 계획 공고는 못 받는다). 그러니 `_raw`는 삭제하지 말고 백업하라고 권한다.

## 6. 이관 메모
- `[이관: 기획]` candidate를 `schedule`/CSV(`입찰예상일`·`지난입찰일`)에 올릴지, 망 안 `bid_cases`로 어떻게 반영할지(계획·재공고→진행 중 contractEnd, 결과·지정→회차 종료 lastBid로 옮기는 매핑)와 `confidence`를 언제 `확정`으로 바꿀지(사람 확인 시점) — 기다리는 것은 기획의 정책 결정 — 근거: `SOURCES.md` P1·P4와 SCHEMA ④ "추측 금지". 이 패치는 기본값을 "schedule에는 올리지 않음"으로 두었다.
- `[이관: 데이터]` 제목 규칙 `EXCLUDE`(금리·협력사업비·공시송달·이율·출연)와 종류 판정(`지정공고`=회차 종료)이 실제 공고 제목과 맞는지, 호스트 설정의 기관 key(region|name)가 `institutions.js`와 맞는지 검수 — 기다리는 것은 최근 1년치 실측 목록 — 근거: `institutions.js` 272·281행 서술.
- `[이관: 메인]` `SCHEMA.md` ④에 `notice`·`candidate` 칸 문서화, `.gitignore`에 `data/collector/` 추가, `SOURCES.md` §3의 "아직 만들지 않음" 갱신, `collector/README`(있다면)에 CLI 사용법 추가 — 규칙급·공통 문서라 메인 담당이다.
