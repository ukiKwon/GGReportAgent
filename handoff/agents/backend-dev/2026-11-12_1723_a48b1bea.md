```python
def _reclaim(path: Path, seen: bytes) -> bool:
    """오래된 잠금은 '회수 표식'(O_EXCL)을 쥔 한 실행만, 다시 읽은 내용이 처음 본 것과 같을 때만 지운다(qa P3-D).
    True = 지웠거나 이미 없음, False = 다른 실행이 회수 중·이미 새 잠금·지울 수 없음('다른 실행 중' 4)."""
    mark = path.with_name(path.name + ".reclaim")
    try:
        fd = os.open(mark, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
    except FileExistsError:
        try:
            if time.time() - mark.stat().st_mtime > STALE_RECLAIM_SEC:
                mark.unlink(missing_ok=True)       # 회수 도중 죽은 표식 — 이번은 넘기고 다음 실행이 회수
        except OSError:
            pass
        return False
    except OSError:
        return False
    try:
        os.close(fd)
        try:
            now = path.read_bytes()
        except FileNotFoundError:
            return True                            # 그새 주인이 정상 해제
        except OSError:
            return False
        if now != seen:
            return False                           # 다른 실행이 먼저 회수해 새 잠금을 만들었다 — 지우지 않는다
        try:
            path.unlink(missing_ok=True)
        except OSError:                            # 백신·색인·열린 핸들(Windows)
            return False
        return True
    finally:
        try:
            mark.unlink(missing_ok=True)
        except OSError:
            pass
```

```python
def _stale(lock, text):
    lock.parent.mkdir(parents=True, exist_ok=True)
    lock.write_text(text)
    os.utime(lock, (1, 1))


def test_reclaim_race_keeps_other_runs_new_lock(tmp_path, monkeypatch):
    lock = tmp_path / "_state" / "run.lock"
    _stale(lock, "pid=1 token=old at=1\n")
    orig = cli._reclaim

    def racing(path, seen):
        path.write_text("pid=2 token=bbbb at=2\n")      # 다른 실행이 먼저 회수하고 새 잠금을 만듦
        return orig(path, seen)
    monkeypatch.setattr(cli, "_reclaim", racing)
    assert cli._lock(lock, "aaaa") is None
    assert "token=bbbb" in lock.read_text()
    assert not lock.with_name("run.lock.reclaim").exists()
```

## 5. 보안·위험·되돌리는 법
- 이번 패치는 알려진 실패로 분류되는 범위를 좁히는 쪽이다. 첫 실행에서 서울 3곳이 tls_verify가 아닌 원인으로 실패하면 종료 코드가 0에서 3으로 바뀐다. 의도한 동작이다.
- 오류 메시지와 실패 종류 값에는 사용자 입력이나 비밀값이 들어가지 않는다. 로더 오류에는 호스트 이름만 싣는다.
- ⓒ에 남는 위험(POSIX만): 회수 표식 자체가 10분 넘게 남은 상태에서 3개 이상의 실행이 겹치면 이론상 경쟁이 남는다. Windows에서는 살아 있는 주인이 잠금 fd를 열어 두고, CRT `_wopen`은 FILE_SHARE_DELETE를 주지 않으므로 남의 활성 잠금 삭제가 PermissionError로 막힐 것으로 본다(추정, 실측하지 않음). 이 패치는 그 동작에 기대지 않는다.
- ⓑ의 `exit /b`가 스케줄러 "마지막 결과"에 반영되지 않으면 `exit %FINAL%`(/b 없이)로 바꾼다. 다만 그렇게 하면 콘솔에서 직접 돌릴 때 창이 닫힌다.
- **되돌리는 법**: 파일 단위로 되돌리면 된다. JSON에 추가한 두 필드는 이전 코드가 무시하므로 코드만 되돌려도 깨지지 않는다. `run.lock.reclaim`은 남아도 해가 없다(10분 뒤 자동 삭제).
