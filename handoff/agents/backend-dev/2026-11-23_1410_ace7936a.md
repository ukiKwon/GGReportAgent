    public FirstStage firstStageAuthority(String taskId, SysUser me) {
        boolean legacyOk = me != null && "ACTIVE".equals(me.getStatus()) && LEAD_SALES_ROLE.equals(roleOf(me));
        FirstStage legacy = legacyOk ? FirstStage.LEGACY_ROLE : FirstStage.NONE;
        int n = mapper.selectMaxRound(taskId);
        if (n <= 0) {
            return legacy;                                   // 회차 없는 옛 작업
        }
        DesignRound r = mapper.selectRound(taskId, n);
        List<String> rs = mapper.selectReviewer(taskId, n, REASON_APPROVER);
        if (r == null || rs.isEmpty()) {
            return legacy;                                   // 옛 규칙 회차(받는 사람 여럿)
        }
        SysUser reviewer = users.selectByEmpNo(rs.get(0));

     * S4(팀 결정, 되돌리기 쉬움): 디자인 1차는 영업팀 내부 확인이라 대결자를 영업팀원·영업팀장으로 한정한다 —

     */
    private boolean isDeputyOf(SysUser reviewer, SysUser me, DesignRound r) {
        if (!"ACTIVE".equals(reviewer.getStatus()) || !"ACTIVE".equals(me.getStatus())
                || reviewer.getEmpNo().equals(me.getEmpNo())) {
            return false;
        }
        boolean salesOrg = MEMBER_SALES_ROLE.equals(roleOf(me)) || LEAD_SALES_ROLE.equals(roleOf(me));
        if (!salesOrg || !deputies.deputiesOf(reviewer.getEmpNo(), Times.todayKst()).contains(me.getEmpNo())) {
            return false;
        }
        SysUser designer = users.selectByEmpNo(r.getOpenedBy());
        return designer != null && Ranks.mayDecide(designer, me);
    }

    /** 디자이너 1차 결재 직전(TASKS 잠금 아래) — 회차가 있는데 마지막 회차가 닫혀 있으면 1차완료와 어긋난 상태라 409(fail-closed, S5). 회차 없는 옛 작업은 통과. */
    public void requireOpenFirstStageRound(String taskId) {
        int n = mapper.selectMaxRound(taskId);
        if (n <= 0) {
            return;
        }
        DesignRound r = mapper.selectRound(taskId, n);
        if (r == null || !r.isOpen()) {
            throw coded(409, "ROUND_STATE", "확인 요청 회차가 이미 닫혀 있습니다. 새로 고친 뒤 다시 확인해 주세요.");
        }
    }

    /** 디자이너 최종 결재 직전 — 마지막 회차를 부장께 올린(FORWARD) 사람은 최종 결재할 수 없다(S3, 역할이 바뀐 경우 대비). 회차 없는 옛 작업은 통과. */
    public void requireNotFirstStageActor(String taskId, SysUser approver) {
        int n = mapper.selectMaxRound(taskId);
        if (n <= 0) {
            return;
        }
        DesignRound r = mapper.selectRound(taskId, n);
        if (r != null && CLOSE_FORWARD.equals(r.getCloseCd()) && approver.getEmpNo().equals(r.getClosedBy())) {
            throw new ApiException(403, "1차를 처리한 사람은 최종 결재할 수 없습니다.");
        }
    }

## 5. 보안·위험·되돌리는 법
- **권한 범위**
  - 디자이너 1차 결재는 "영업팀장 누구나"에서 **REVIEWER·DEPUTY·LEGACY_ROLE**로 좁아집니다.
  - 넓어지는 쪽은 영업 조직 대결자뿐입니다.
  - 결재와 결재함이 같은 함수를 쓰므로 "보이는데 못 누름" 또는 "안 보이는데 누를 수 있음"이 생기지 않습니다. 예외는 S5 409뿐이고, 이건 상태가 어긋난 경우입니다.
- **LEGACY_ROLE 범위**: 회차 팀장이 잠기거나 퇴직하거나 역할이 바뀌면 모든 활성 영업팀장이 처리할 수 있습니다. 건이 묻히지 않게 하려는 장치인데, 같은 날 대결자가 지정돼 있어도 대결자(영업팀원) 대신 팀장 전원에게 열립니다. security가 이 정책을 확인해야 합니다.
- **403·감사**: 403은 서비스에서 던지고 TaskController.java:248-250에서 DSN_* DENIED와 ACCESS_LOG DENIED로 남습니다. 성공은 `DSN_FORWARD`/`DSN_FORWARD_DEP` 등에 ACTOR·SCREEN·BUTTON이 함께 남습니다(원안과 같음). 본문·사유는 남기지 않습니다.
- **자기신고의 한계**: 사번은 `X-Emp-No` 자기신고입니다. 대결·회차 팀장 판정도 그 위에서 하므로 사칭을 막지 못합니다(NEXT 35).
- **위험**
  - 영업팀원 대결자는 `approvals` 메뉴 권한이 없으면 MenuGuard에서 403이 납니다.
  - 결재함 숫자는 portal.js:27 때문에 영업팀원 대결자에게 잡히지 않습니다(원안 이관 그대로).
  - `firstStageAuthority`는 카드 한 장마다 조회를 3~5번 합니다. 카드 수가 적어 무시할 만합니다.
- **되돌리는 법**
  - 대결자를 영업 조직으로 한정한 것(S4)만 풀려면 `isDeputyOf`의 `salesOrg` 조건을 지웁니다.
  - "다른 영업팀장 차단"만 풀려면 `firstStageAuthority`의 마지막 `NONE`을 `legacy`로 바꿉니다.
  - 전체를 되돌리면 원안 상태가 되고, DB 정리는 필요 없습니다.
