## 요약
- 고정 경로 `path.join(ROOT,'..','handoff','questions.json')`만 읽는다. 요청 값으로 경로를 만들지 않는다.
- GET/HEAD만 허용하고, 그 외 메서드는 405 + `Allow: GET, HEAD`. Host 검사는 기존 순서대로 가장 먼저 한다.
- 응답은 200 + `{"questions":[...]}`. 파일 없음·깨진 JSON·형식 불일치·256KB 초과·읽기 실패는 200 + `{"questions":[],"error":"짧은 일반 문구"}`. 오류 문구에 경로나 예외 내용은 싣지 않는다.
- 헤더는 `Content-Type: application/json; charset=utf-8`, `Cache-Control: no-store`.
- 형식 검증은 최소: 최상위가 객체이고 `questions`가 배열인지만 보고, 항목 중 객체가 아닌 것은 걸러낸다. 필드·status 값은 검증하지 않고 그대로 통과시킨다.
- `/api/version`은 변경 없음. 화면 파일 4개의 mtime만 보므로 questions.json 변경으로 새로고침되지 않는다.
- 프런트 계약: 같은 출처(8090)로 `GET /api/questions` → `{"questions":[...],"error"?}`.

```js
const QFILE = path.join(ROOT, '..', 'handoff', 'questions.json'), MAX_Q = 256 * 1024; // 고정 경로(요청 값으로 경로를 만들지 않는다)
const readQuestions = () => { // 어떤 실패도 {questions:[],error}로 — 대시보드가 죽지 않게. 오류 문구에는 경로·예외 내용을 싣지 않는다

    if (!st.isFile()) return { questions: [], error: 'questions.json not found' };
    if (st.size > MAX_Q) return { questions: [], error: 'questions.json too large' };
```

## 2. serve.py 패치

(a) 상수와 함수 추가.
```python
# old
MAX_BODY = 16 * 1024
# new
MAX_BODY = 16 * 1024
QFILE = ROOT.parent / "handoff" / "questions.json"  # 고정 경로(요청 값으로 경로를 만들지 않는다)
MAX_Q = 256 * 1024


def read_questions():
    """어떤 실패도 {questions:[],error}로 돌려 대시보드가 죽지 않게 한다. 오류 문구에 경로·예외 내용을 싣지 않는다."""
    try:
        if not QFILE.is_file():
            return {"questions": [], "error": "questions.json not found"}
        if QFILE.stat().st_size > MAX_Q:
            return {"questions": [], "error": "questions.json too large"}
        j = json.loads(QFILE.read_bytes().decode("utf-8-sig"))
        if not isinstance(j, dict) or not isinstance(j.get("questions"), list):
            return {"questions": [], "error": "questions.json bad format"}
        return {"questions": [q for q in j["questions"] if isinstance(q, dict)]}
    except (OSError, ValueError):
        return {"questions": [], "error": "questions.json unreadable"}
```

(b) send_head 안에서 Host 검사 `return None` 다음, `try: name = unquote` 앞에 삽입한다.
```python
# old
        try:
            name = unquote(self.path.split("?")[0].split("#")[0], errors="strict")
# new
        if self.path.split("?")[0].split("#")[0] == "/api/questions":  # 읽기 전용 API
            body = json.dumps(read_questions(), ensure_ascii=False).encode("utf-8")
            self.send_response(200)
            self.send_header("Content-Type", "application/json; charset=utf-8")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            if self.command != "HEAD":

            name = unquote(self.path.split("?")[0].split("#")[0], errors="strict")
```

## 3. 테스트 — test/server.test.js 맨 아래에 추가

기존 시험은 `os.tmpdir()/agdash-*`에 복사해 띄우므로 `ROOT/..`가 공용 임시 폴더가 된다. 그래서 `base/agent-dashboard` + `base/handoff` 구조의 별도 서버를 띄운다. 기존 before·FILES는 그대로 두며, 깨지는 기존 시험은 없다.
```js
// ── /api/questions: handoff/questions.json(ROOT/../handoff)을 읽는 읽기 전용 API. 부모 폴더가 달라야 해서 base/agent-dashboard + base/handoff 구조로 따로 띄운다 ──
describe('/api/questions', () => {
  let base, qport, qchild;
  const qfile = () => path.join(base, 'handoff', 'questions.json');
  const qreq = (method, p, headers = {}) => new Promise((resolve, reject) => {
    const r = http.request({ host: '127.0.0.1', port: qport, method, path: p, headers }, res => {
      let b = ''; res.setEncoding('utf8'); res.on('data', d => b += d); res.on('end', () => resolve({ status: res.statusCode, headers: res.headers, body: b }));
    });
    r.on('error', reject); r.end();
  });
  before(async () => {
    base = fs.mkdtempSync(path.join(os.tmpdir(), 'agdash-q-'));
    const dash = path.join(base, 'agent-dashboard'); fs.mkdirSync(dash); fs.mkdirSync(path.join(base, 'handoff'));
    for (const f of FILES) fs.copyFileSync(path.join(SRC, f), path.join(dash, f));
    qport = await freePort();
    qchild = spawn(process.execPath, [path.join(dash, 'serve.js'), String(qport)], { cwd: dash, stdio: ['ignore', 'pipe', 'inherit'] });
    await new Promise((resolve, reject) => {
      qchild.on('error', reject);
      qchild.stdout.on('data', d => { if (String(d).includes('agent-dashboard:')) resolve(); });
      setTimeout(() => reject(new Error('server start timeout')), 10000);
    });
  });
  after(async () => {
    if (qchild) { const done = new Promise(r => qchild.once('exit', r)); qchild.kill(); await done; }
    try { fs.rmSync(base, { recursive: true, force: true }); } catch (_) {}
  });

  it('정상 파일은 200·JSON·no-store로 questions를 돌려준다', async () => {
    fs.writeFileSync(qfile(), JSON.stringify({ questions: [{ id: 'Q-1', summary: '요약', status: 'open', next: [72] }, 'x', null] }));
    const r = await qreq('GET', '/api/questions?x=1');
    assert.strictEqual(r.status, 200);
    assert.match(r.headers['content-type'], /application\/json/);
    assert.strictEqual(r.headers['cache-control'], 'no-store');
    const j = JSON.parse(r.body);
    assert.strictEqual(j.error, undefined);
    assert.deepStrictEqual(j.questions, [{ id: 'Q-1', summary: '요약', status: 'open', next: [72] }]); // 객체 아닌 항목은 걸러진다
  });
  it('파일이 없으면 200 + 빈 목록 + error', async () => {
    fs.rmSync(qfile(), { force: true });
    const r = await qreq('GET', '/api/questions');
    assert.strictEqual(r.status, 200);
    const j = JSON.parse(r.body);
    assert.deepStrictEqual(j.questions, []); assert.ok(j.error);
    assert.ok(!r.body.includes(base), '오류에 경로가 실리면 안 된다');
  });
  it('깨진 JSON·형식 불일치는 200 + 빈 목록 + error', async () => {
    for (const txt of ['{broken', '[]', '{"questions":"x"}', '']) {
      fs.writeFileSync(qfile(), txt);
      const r = await qreq('GET', '/api/questions');
      assert.strictEqual(r.status, 200, txt);
      const j = JSON.parse(r.body); assert.deepStrictEqual(j.questions, [], txt); assert.ok(j.error, txt);
    }
  });
  it('256KB 초과는 200 + 빈 목록 + error(본문을 읽지 않는다)', async () => {
    fs.writeFileSync(qfile(), '{"questions":[{"summary":"' + 'a'.repeat(260 * 1024) + '"}]}');
    const j = JSON.parse((await qreq('GET', '/api/questions')).body);
    assert.deepStrictEqual(j.questions, []); assert.ok(j.error);
  });
  it('UTF-8 BOM이 있어도 읽힌다(Windows 편집기 대비)', async () => {
    fs.writeFileSync(qfile(), '\uFEFF{"questions":[{"id":"Q-2"}]}');
    assert.strictEqual(JSON.parse((await qreq('GET', '/api/questions')).body).questions.length, 1);
  });
  it('POST·PUT·DELETE는 405이고 파일은 그대로다', async () => {
    fs.writeFileSync(qfile(), '{"questions":[]}');
    for (const m of ['POST', 'PUT', 'DELETE']) {
      const r = await qreq(m, '/api/questions', { 'Content-Type': 'application/json' });
      assert.strictEqual(r.status, 405, m); assert.match(r.headers.allow, /GET/);
    }
    assert.strictEqual(fs.readFileSync(qfile(), 'utf8'), '{"questions":[]}');
  });
  it('Host 변조는 403, HEAD는 200이고 본문 없음', async () => {
    assert.strictEqual((await qreq('GET', '/api/questions', { Host: 'evil.com' })).status, 403);
    const h = await qreq('HEAD', '/api/questions');
    assert.strictEqual(h.status, 200); assert.strictEqual(h.body, '');
  });
  it('/handoff/questions.json 직접 접근은 404(정적 허용 목록 밖)', async () => {
    assert.strictEqual((await qreq('GET', '/handoff/questions.json')).status, 404);
    assert.strictEqual((await qreq('GET', '/../handoff/questions.json')).status, 404);
  });
});
```

## 4. README.md(agent-dashboard) 문안 3줄
> **사용자에게 물은 질문 패널.** 정본은 저장소 루트 `handoff/questions.json`(git 추적, 메인이 손으로 기록)이고, 서버는 `GET /api/questions`로 그 파일 하나만 읽어 `{"questions":[...]}`로 돌려줍니다(읽기 전용 — POST 등은 405, 256KB 초과·파일 없음·깨진 JSON은 200 + 빈 목록 + `error`).
> 화면은 이 주소를 폴링해 갱신하므로 파일을 고치면 새로고침 없이 반영됩니다(`/api/version`은 화면 코드 변경만 감지).
> 항목 형식: `id, summary, askedAt, next[], options, recommend, status(open|answered|done|hold), answer, answeredAt, appliedRef`.

## 5. 보안·위험·되돌리는 법
- 경로는 상수로 고정되어 사용자 입력이 경로에 들어가지 않는다. 정적 서빙 허용 목록(ALLOW)은 그대로라 handoff/ 파일이 직접 노출되지 않는다(위 마지막 시험으로 확인).
- 127.0.0.1 바인딩 + Host 검사로 DNS 리바인딩을 막지만, 이 API는 questions.json 내용을 그대로 노출한다. 질문·답변에 비밀값·개인정보를 적지 않는 것은 기록하는 쪽(메인·pm)의 몫이다.
- 읽기 전용이라 쓰기 경로가 늘지 않는다. 되돌리기는 삽입 블록을 지우면 되고 DB·데이터 변경은 없다.
- 편집 중 깨진 JSON이 읽히면 그 폴링 한 번만 error + 빈 목록이 온다.

## 6. 이관 메모
- [이관: 프런트] error가 있으면 목록을 비우지 말고 직전 성공본을 유지하며 작은 오류 표시만 한다. 폴링 주기는 /state.json 폴링과 맞춘다. 근거: 편집 중 깨진 JSON 응답.
- [이관: 기획] status 4값(open|answered|done|hold)의 표시 용어·정렬 우선순위. 서버는 status를 검증하지 않고 그대로 통과시킨다.
