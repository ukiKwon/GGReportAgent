package com.kb.uploader.parse;

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.zip.*;

public final class ZipFixtures {
    private ZipFixtures() {}

    public static byte[] zip(String name, byte[] data) throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        try (ZipOutputStream z = new ZipOutputStream(out)) {
            z.putNextEntry(new ZipEntry(name));
            z.write(data);
            z.closeEntry();
        }
        return out.toByteArray();
    }

    /** 내용 0 으로 채운 entry(압축 후 작다). */
    public static byte[] zeros(String name, int size) throws IOException {
        return zip(name, new byte[size]);
    }

    /**
     * 기존 zip 에 같은 이름(existing)의 entry 를 하나 더 만든다. ZipOutputStream 은 중복을 거부하므로
     * 이름 끝 글자를 '~' 로 바꿔 쓴 뒤, 바이트에서 그 이름을 원래 이름으로 되돌린다(같은 길이, CRC 무관).
     */
    public static byte[] duplicateEntry(byte[] zip, String existing) throws IOException {
        String fake = existing.substring(0, existing.length() - 1) + "~";
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        try (ZipInputStream in = new ZipInputStream(new ByteArrayInputStream(zip));
             ZipOutputStream z = new ZipOutputStream(out)) {
            ZipEntry e;
            byte[] buf = new byte[8192];
            while ((e = in.getNextEntry()) != null) {
                z.putNextEntry(new ZipEntry(e.getName()));
                int n;
                while ((n = in.read(buf)) > 0) z.write(buf, 0, n);
                z.closeEntry();
            }
            z.putNextEntry(new ZipEntry(fake));
            z.write(new byte[]{'x'});
            z.closeEntry();
        }
        byte[] b = out.toByteArray();
        byte[] f = fake.getBytes(StandardCharsets.UTF_8);
        byte[] r = existing.getBytes(StandardCharsets.UTF_8);
        for (int i = 0; i + f.length <= b.length; i++) {
            boolean m = true;
            for (int k = 0; k < f.length && m; k++) m = b[i + k] == f[k];
            if (m) System.arraycopy(r, 0, b, i, r.length);
        }
        return b;
    }

    /** 기존 zip 에 entry 하나를 덧붙인다. */
    public static byte[] addEntry(byte[] zip, String name, byte[] data) throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        try (ZipInputStream in = new ZipInputStream(new ByteArrayInputStream(zip));
             ZipOutputStream z = new ZipOutputStream(out)) {
            ZipEntry e;
            byte[] buf = new byte[8192];
            while ((e = in.getNextEntry()) != null) {
                z.putNextEntry(new ZipEntry(e.getName()));
                int n;
                while ((n = in.read(buf)) > 0) z.write(buf, 0, n);
                z.closeEntry();
            }
            z.putNextEntry(new ZipEntry(name));
            z.write(data);
            z.closeEntry();

package com.kb.uploader.parse;

import org.apache.poi.xssf.usermodel.XSSFWorkbook;
import org.junit.Rule;
import org.junit.Test;
import org.junit.rules.TemporaryFolder;
import java.io.ByteArrayOutputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import static org.junit.Assert.*;

public class ZipGuardTest {
    private static final long MB = 1024L * 1024;
    @Rule public TemporaryFolder tmp = new TemporaryFolder();

    private static byte[] smallXlsx() throws Exception {
        try (XSSFWorkbook wb = new XSSFWorkbook()) {
            wb.createSheet("s").createRow(0).createCell(0).setCellValue("가");
            ByteArrayOutputStream o = new ByteArrayOutputStream();
            wb.write(o);
            return o.toByteArray();
        }
    }

    private static String code(byte[] zip, long maxEntry, long maxTotal) {
        try { ZipGuard.check(zip, maxEntry, maxTotal); fail("거부되어야 한다"); return null; }
        catch (ZipGuard.Rejected e) { return e.code; }
    }

    @Test public void entry_하나가_한도를_넘으면_TOO_LARGE_UNPACKED() throws Exception {
        byte[] z = ZipFixtures.zeros("a.bin", (int) (2 * MB));
        assertTrue(z.length < MB);
        assertEquals("TOO_LARGE_UNPACKED", code(z, MB, 10 * MB));
    }

    @Test public void 같은_이름_entry는_DUPLICATE_ENTRY() throws Exception {
        byte[] dup = ZipFixtures.duplicateEntry(smallXlsx(), "xl/workbook.xml");
        assertEquals("DUPLICATE_ENTRY", code(dup, 20 * MB, 40 * MB));
    }

    @Test public void 대소문자만_다른_entry도_DUPLICATE_ENTRY() throws Exception {
        byte[] dup = ZipFixtures.addEntry(smallXlsx(), "XL/Workbook.XML", "<x/>".getBytes("UTF-8"));
        assertEquals("DUPLICATE_ENTRY", code(dup, 20 * MB, 40 * MB));
    }

    @Test public void 정상_xlsx는_통과() throws Exception {
        assertTrue(ZipGuard.checkSmallXlsx(smallXlsx()).names.contains("xl/workbook.xml"));
    }

    @Test public void DOCTYPE_파트는_BROKEN() throws Exception {
        byte[] z = ZipFixtures.zip("Contents/section0.xml",
                "<!DOCTYPE a [<!ENTITY e \"x\">]><a>&e;</a>".getBytes("UTF-8"));
        assertEquals("BROKEN", code(z, 20 * MB, 40 * MB));
    }

    @Test public void 파일_입력도_같게_거부하고_isZip은_PK만_true() throws Exception {
        Path dup = tmp.newFile("d.xlsx").toPath();
        Files.write(dup, ZipFixtures.duplicateEntry(smallXlsx(), "xl/workbook.xml"));

        assertFalse(ZipGuard.isZip(tmp.getRoot().toPath().resolve("none")));

## 5. 보안·위험·되돌리는 법
- **인증 한계**: 이번 패치는 권한 경로를 바꾸지 않습니다. uploader 업로드 화면의 접근 통제는 이번 범위 밖입니다.
- **로그에 파일 내용·본문이 들어가지 않습니다.** 거부 사유는 코드(`DUPLICATE_ENTRY` 등)만 싣습니다.
- **메모리**: `ZipGuard`는 XML처럼 보이는 파트를 entry당 최대 100MB까지 메모리에 모아 StAX로 읽습니다. 제안서 PPTX 안의 큰 슬라이드 XML이 이만큼 크지는 않지만, 동시 업로드가 여러 건이면 힙이 부담될 수 있습니다. 이 방식과 한도는 web `PrepAttachInspector`와 같아서 따랐습니다.
- **TOCTOU**: 검사 시점과 POI가 여는 시점이 달라 파일을 바꿔치기할 수 있습니다. 저장 파일은 서버 디렉터리에 있어 외부에서 쓸 수 없다고 보지만, 확인한 것은 아닙니다. 2차로 `ZipSecureFile`이 한 번 더 막습니다.
- **정상 파일 오탐 가능성**: 한도를 넘는 정상 제안서(합계 200MB 초과)와 DOCTYPE이 들어간 정상 HWPX는 거부됩니다. 운영 PPTX가 100MB·200MB를 넘는 일은 드물다고 봅니다. 실제 파일로 확인해 보세요.
- **되돌리기**: 3-1·3-2 새 파일 2개를 지우고 서비스 3곳의 가드 호출을 빼면 원상태입니다. DB는 건드리지 않습니다.
- **POI 상향(NEXT 40)과의 관계**: 이 가드는 POI 중복 entry 처리(CVE-2025-31672)의 완화일 뿐입니다. 5.x 상향은 별개로 남습니다.
