```python
def classify_court(title: str) -> str:
    """지정계획 공고=계획, 지정 결과=지정, 변경 안내·취소=기타, 법원보관금 취급은행=제외, 그 외 무관."""

    if "결과" in t or "지정" in t:
```

```python
@dataclass(frozen=True)

@dataclass(frozen=True)
class ListRow:
    seqnum: str
    title: str
    posted_at: str | None


def parse_list(text: str) -> list[ListRow]:
    """목록 HTML → 행. seqnum이 든 링크가 있는 <tr>만. 날짜는 날짜 모양의 칸."""
    rows: list[ListRow] = []
    seen: set[str] = set()
    for row in ROW_RE.findall(text):
        a = ANCHOR_RE.search(row)
        if not a or a.group(1) in seen:
            continue
        title = clean(a.group(2))
        if not title:
            continue
        date = None
        for cell in CELL_RE.findall(row):
            c = clean(cell)
            d = DATE_RE.match(c) if len(c) <= 14 else None
            if d:
                date = iso(*d.groups())
                break
        seen.add(a.group(1))
        rows.append(ListRow(a.group(1), title, date))
    return rows


def parse_downloads(text: str) -> list[tuple[str, str]]:
    return [(m.group(1), html.unescape(m.group(2)).strip()) for m in DOWNLOAD_RE.finditer(text)]


class ScourtSource:
    slug = "scourt"
    name_ko = "대법원 새소식(공탁금 보관은행)"
    base_url = LIST_URL

    def __init__(self, config: ScourtConfig | None = None, http=None, sleep=None) -> None:
        self.config = config or ScourtConfig()
        self._injected_http = http is not None
        self.http = http or HttpClient(timeout=self.config.timeout, delay=self.config.delay_sec)
        self._sleep = sleep or time.sleep
        self._report: RunReport | None = None

    def with_config(self, http=None, sleep=None, hosts_file=None, today=None, **config) -> "ScourtSource":
        # 주지 않은 연결은 기존 값을 유지한다(g2b와 같은 규칙 — 시험용 가짜 HTTP가 실제 호출로 바뀌지 않게).
        return ScourtSource(replace(self.config, **config),
                            http if http is not None else (self.http if self._injected_http else None),
                            sleep if sleep is not None else self._sleep)

    def fetch(self):
        raise SourceError("scourt는 CLI로 실행합니다: python -m collector run --source scourt")

    # ── 요청: https + 두 호스트만, 시간 초과만 백오프 재시도, 마지막 실패는 fetch_failures에 기록 ──
    @staticmethod
    def _guard(url: str) -> None:
        p = urlsplit(url)
        if p.scheme != "https" or p.hostname not in ALLOWED_HOSTS:
            raise SourceError("허용하지 않는 주소입니다")

    def _request(self, method: str, url: str, purpose: str, body: bytes | None = None):
        self._guard(url)
        backoff = tuple(self.config.retry_backoff)
        for attempt in range(len(backoff) + 1):
            try:
                if method == "POST":
                    return self.http.post(url, body, verify=True)
                return self.http.get(url, verify=True)
            except FetchTimeoutError as exc:
                if attempt >= len(backoff):
                    self._record(url, purpose, attempt + 1, exc)
                    raise
            except FetchError as exc:
                self._record(url, purpose, attempt + 1, exc)
                raise
            self._sleep(backoff[attempt])
        raise FetchError("연결 실패")                    # 도달하지 않음

    def _record(self, url, purpose, attempts, exc) -> None:
        if self._report is not None:
            self._report.record_failure(url, purpose, attempts, exc, host=urlsplit(url).hostname or HOST)

    def _list_body(self, page: int) -> bytes:
        kw = quote_plus(self.config.keyword, encoding="cp949")      # 검색어는 cp949(응답도 cp949)
        return f"searchOption=&searchWord={kw}&pageIndex={page}&pageSize={self.config.page_size}".encode("ascii")

    @staticmethod
    def _detail_url(seqnum: str) -> str:
        if not seqnum.isdigit():
            raise SourceError("공고 번호 형식이 올바르지 않습니다")
        return f"{VIEW_URL}?pageIndex=1&searchWord=&searchOption=&seqnum={seqnum}&gubun=3"

    def list_new(self, seen, since=None, report=None, hosts=None) -> Iterator[NoticeRef]:
        """hosts는 쓰지 않는다(단일 사이트). 목록은 최신순이라 since보다 오래된 쪽이 나오면 멈춘다."""
        report = report if report is not None else RunReport()
        self._report = report
        kw = self.config.keyword.replace(" ", "")
        rows: list[ListRow] = []
        failed = False
        for page in range(1, self.config.max_pages + 1):
            try:
                data, charset = self._request("POST", LIST_URL, f"목록 {page}쪽", self._list_body(page))
            except FetchError as exc:

                report.hosts_failed_kind[HOST] = error_kind(exc)
                failed = True
                break

                report.warnings.append("scourt: 목록 1쪽에서 공고를 찾지 못함(검색 결과 없음이거나 화면 형식이 바뀜 — 확인 필요)")
            rows.extend(page_rows)
            if len(page_rows) < self.config.page_size:
                break
            if since and page_rows and all(r.posted_at and r.posted_at < since for r in page_rows):
                break
        if not failed:
            report.hosts_ok.append(HOST)
        local: set[str] = set()
        for r in rows:
            if kw not in r.title.replace(" ", ""):
                continue
            if since and r.posted_at and r.posted_at < since:
                continue
            ref = NoticeRef(source=self.slug, host=HOST, number=f"seq{r.seqnum}", title=r.title,
                            detail_url=self._detail_url(r.seqnum), posted_at=r.posted_at, native_id=r.seqnum)
            if ref.key in local or seen.is_seen(ref.key):
                continue
            local.add(ref.key)
            yield ref

    def accepts(self, ref: NoticeRef) -> bool:
        return classify_court(ref.title) in ("계획", "지정", "기타")

    def fetch_raw(self, ref: NoticeRef, store) -> RawNotice:
        url = self._detail_url(ref.native_id or "")
        try:
            data, charset = self._request("GET", url, f"상세 {ref.number}")
        except FetchError as exc:
            raise SourceError(f"상세 조회 실패(scourt): {exc}") from exc
        text, encoding = decode_body(data, charset)
        masked_text, masked_n = mask_contacts(text)                 # 사용자 결정 11-12: 연락처 전부 가림
        files: list[tuple[str, bytes]] = []
        errors: list[dict] = []
        unmaskable: list[str] = []
        links = parse_downloads(text)
        if not links:
            errors.append({"name": "", "error": "첨부 없음"})        # 2017~2019 공고 — 읽기 불가 기록의 근거
        for fid, name in links[: self.config.max_attachments]:
            if not name.lower().endswith(ALLOWED_EXT):
                errors.append({"name": name[:80], "error": "허용하지 않는 확장자"})
                continue
            if not FID_RE.match(fid):
                errors.append({"name": name[:80], "error": "첨부 식별자 형식이 올바르지 않음(받지 않음)"})
                continue
            try:
                blob, _ = self._request("POST", FILE_URL, f"첨부 {ref.number} {name}"[:120],
                                        f"file={quote_plus(fid)}&path=001&downFile=x".encode("ascii"))
            except FetchError as exc:

                except ValueError as exc:

            else:

            files.append((name, blob))
        meta = {
```

```python
"""법원 공탁금 보관은행 공고(소스 scourt) 전용 추출: 원문 1건 → 법원별 후보 여러 건.

규칙(사용자 결정 11-13): last_bid=정기(계획) 공고일, term=원문에 적힌 연수만(날짜 차이로 계산하지 않음),
contract_end=원문 만료(지정 끝났으면 새 기간 끝), '6개월 전' 추정은 하지 않는다(화면 몫). 확정은 사람 몫(항상 '후보'·예상).
공고 첨부는 법원 여러 곳이 공유하므로 배치에 싣지 않는다(원문 보관소에만).
"""

from __future__ import annotations

import re
from pathlib import Path

from collector.extract.build import _html_text
from collector.extract.classify import ROLE, classify_court
from collector.extract.hwp5 import hwp5_text
from collector.extract.hwpx import ExtractError, hwpx_text
from collector.extract.terms import _D, find_period, iso
from collector.privacy import mask_contacts, vaccine_policy
from collector.sources.base import CollectedNotice

MAX_COURTS = 40
# 법원명: 지방법원(+지원·시법원·군법원). 앞이 한글이면 시작하지 않는다(문장 속 글자가 이름에 붙는 것 방지).
COURT = r"(?<![가-힣])[가-힣]{2,6}지방법원(?:\s+[가-힣]{2,5}(?:지원|시법원|군법원))*"
COURT_RE = re.compile(COURT)
MARKER_RE = re.compile(r"[【\[]\s*공개경쟁\s*대상\s*법원\s*[】\]]")
# 목록 끝: 다음 번호 제목('2. 지정기간')·'가. ' 제목·【. 단 법원명이 이어지는 번호 항목('1. 광주지방법원')은 끝이 아니다.
BOUNDARY_RE = re.compile(r"\s(?:\d{1,2}|[가-힣])\s*[.)]\s+(?![가-힣]{2,6}지방법원)|【")
_SEP = r"\s*[~∼～–\-]\s*"
ROW_RE = re.compile("(" + COURT + r")\s*" + _D + _SEP + _D)                       # 별지 표 한 줄: 법원 + 기간
TERM_PERIOD_RE = re.compile(r"(?<!\d)(\d{1,2})\s*년\s*\(\s*" + _D + _SEP + _D + r"\s*\)")   # 5년(2026. 1. 1. ~ 2030. 12. 31.)
NEWBANK_RE = re.compile(r"(?:새로운|신규|새)\s*(?:지정\s*)?은행[^0-9]{0,40}" + _D)          # 새 은행이면 … 2026. 4. 4.
NUMBER_RE = re.compile(r"제\s*(\d{4})\s*[-–]\s*(\d+)\s*호")


class CourtUnreadable(Exception):
    """대상 법원을 읽지 못함(첨부 없음·본문 깨짐·형식 다름). 실패가 아니라 '읽기 불가' 기록 — 원문은 보관, 후보는 없음."""


def _norm(t: str) -> str:
    return re.sub(r"\s+", " ", t).strip()


def _dedupe(items):
    return list(dict.fromkeys(items))


def marker_courts(text: str) -> list[str]:
    m = MARKER_RE.search(text)
    if not m:
        return []
    rest = _norm(text[m.end():])
    cut = BOUNDARY_RE.search(rest)
    block = (rest[: cut.start()] if cut else rest)[:800]
    return _dedupe(_norm(c) for c in COURT_RE.findall(block))


def appendix_rows(text: str) -> dict[str, tuple[str, str, str]]:
    """별지 표: 법원 → (시작, 끝, 근거문장). 표 칸이 문단으로 나와도 '법원 다음 기간' 순서만 본다."""
    rows: dict[str, tuple[str, str, str]] = {}
    for m in ROW_RE.finditer(_norm(text)):
        court = _norm(m.group(1))
        start, end = iso(*m.group(2, 3, 4)), iso(*m.group(5, 6, 7))
        if court in rows or not (start and end) or end < start:
            continue
        rows[court] = (start, end, m.group(0)[:120])
    return rows


def notice_period(text: str):
    """공고 전체 지정기간 (연수|None, 시작, 끝, 근거문장) 또는 None. 연수는 원문에 '5년(' 꼴로 적힌 것만."""
    nt = _norm(text)
    m = TERM_PERIOD_RE.search(nt)
    if m:
        term, start, end = int(m.group(1)), iso(*m.group(2, 3, 4)), iso(*m.group(5, 6, 7))
        if start and end and end >= start and 1 <= term <= 30:
            return term, start, end, m.group(0)[:120]
    found = find_period(nt)
    return (None, *found) if found else None


def new_bank_start(text: str):
    m = NEWBANK_RE.search(_norm(text))
    if not m:
        return None
    day = iso(*m.group(1, 2, 3))
    return (day, m.group(0)[:120]) if day else None


def parse_court_text(text: str) -> dict | None:
    courts, rows = marker_courts(text), appendix_rows(text)
    if not courts and not rows:
        return None
    return {"courts": _dedupe(courts + list(rows)), "rows": rows,
            "period": notice_period(text), "new_bank": new_bank_start(text)}


def _texts(raw_dir: Path, meta: dict) -> tuple[list[tuple[str, str]], list[str]]:
    texts: list[tuple[str, str]] = []
    notes: list[str] = []
    for a in meta.get("attachments", []):
        name, blob = a["name"], (raw_dir / a["stored"]).read_bytes()
        try:
            if name.lower().endswith(".hwp"):
                texts.append((f"hwp:{name}", hwp5_text(blob)))
            elif name.lower().endswith(".hwpx"):
                texts.append((f"hwpx:{name}", hwpx_text(blob)))
            else:
                notes.append(f"{name}: 글을 읽지 않는 형식")
        except ExtractError as exc:
            notes.append(f"{name}: {exc}")
    for err in meta.get("attachment_errors", []):
        notes.append(f"첨부 받기 실패 {err.get('name')}: {err.get('error')}".strip())
    texts.append(("상세 본문", _html_text((raw_dir / "detail.html").read_bytes(), meta.get("encoding", "utf-8"))))
    return texts, notes


def _candidate(kind: str, posted_at: str | None, origin: str, term, start, end, sentence) -> dict | None:
    fields: dict = {}
    basis: list[dict] = []
    if start and end:
        fields.update(contract_start=start, contract_end=end)
        basis += [{"field": "contract_start", "text": sentence, "from": origin},
                  {"field": "contract_end", "text": sentence, "from": origin}]
        if term:
            fields["term"] = term                                    # 원문 명시 연수 — derived 아님
            basis.append({"field": "term", "text": sentence, "from": origin})
    if posted_at:
        key = "last_bid" if kind == "계획" else "notice_at"          # 계획=정기 공고일, 지정 결과 게시일은 notice_at
        fields[key] = posted_at
        basis.append({"field": key, "text": f"공고 게시일 {posted_at}", "from": "목록 게시일"})
    if not fields:
        return None
    return {"status": "후보", "kind": kind, "role": ROLE[kind], "fields": fields, "basis": basis}


def court_notices(raw_dir: Path, rel: str, meta: dict) -> list[CollectedNotice]:
    title = meta["title"]
    kind = classify_court(title)
    if kind not in ("계획", "지정", "기타"):
        raise CourtUnreadable(f"종류 판정 '{kind}' — 후보를 만들지 않음")
    texts, notes = _texts(raw_dir, meta)
    texts = [(o, mask_contacts(t)[0]) for o, t in texts]            # 근거 문장에 연락처 금지
    parsed, origin = None, ""
    for o, t in texts:                                              # 첨부(hwp) 우선, 상세 본문은 폴백
        parsed = parse_court_text(t)
        if parsed:
            origin = o
            break
    if not parsed or len(parsed["courts"]) > MAX_COURTS:
        raise CourtUnreadable(mask_contacts("대상 법원을 읽지 못함(읽기 불가) — " + "; ".join(notes))[0][:200])
    number = meta.get("notice_no")
    if not number:
        for _, t in texts:
            m = NUMBER_RE.search(t)
            if m:
                number = f"제{m.group(1)}-{m.group(2)}호"
                break
    number = number or meta["number"]
    note = [f"공고 1건에 법원 {len(parsed['courts'])}곳 — 법원별 후보로 나눔(첨부는 원문 보관소에만)"] + notes
    if parsed["new_bank"]:
        note.append(f"은행이 바뀌면 업무 개시 {parsed['new_bank'][0]}(원문: {parsed['new_bank'][1]})")
    evidence_note = mask_contacts("; ".join(note))[0]
    out: list[CollectedNotice] = []
    for court in parsed["courts"]:
        row = parsed["rows"].get(court)
        if row:                                                     # 별지: 법원별 기간, 연수는 없음
            term, start, end, sentence = None, row[0], row[1], row[2]
        elif parsed["period"]:
            term, start, end, sentence = parsed["period"]
        else:
            term = start = end = sentence = None
        cand = _candidate(kind, meta.get("posted_at"), origin, term, start, end, sentence) if kind != "기타" else None
        out.append(CollectedNotice(
            notice_id=f"{meta['host']}#{meta['number']}#{court}",
            title=mask_contacts(title)[0],
            institution_name_ko=court,
            evidence_url=meta["detail_url"],
            posted_at=meta.get("posted_at"),
            confirmed=False,                                        # 자동 추출값은 항상 '예상'
            institution_type="법원",
            notice_info={"kind": kind, "number": number, "host": meta["host"], "institution_key": None,
                         "mgt_no": meta.get("native_id"), "department": meta.get("department"), "raw_path": rel,
                         "vaccine": meta.get("vaccine") or vaccine_policy(meta["detail_url"]),
                         "contacts_masked": meta.get("contacts_masked")},
            candidate=cand,
            evidence_note=evidence_note or None))
    return out
```

```python
            f'&amp;searchOption=&amp;seqnum={seq}&amp;gubun=3">{title}</a></td><td>{date}</td><td>12</td></tr>')


class FakeScourt:
    def __init__(self, pages=None, details=None, files=None):
        self.pages, self.details, self.files = pages or {}, details or {}, files or {}
        self.calls = []

    def get(self, url, verify=True):
        self.calls.append(("GET", url, None))
        seq = parse_qs(url.split("?", 1)[1])["seqnum"][0]
        if seq not in self.details:
            raise FetchHttpError("HTTP 404")
        return self.details[seq].encode("cp949"), None

    def post(self, url, data, verify=True):
        self.calls.append(("POST", url, data))
        if "NewsListAction" in url:
            page = int(parse_qs(data.decode("ascii"))["pageIndex"][0])
            return ("<table>" + "".join(self.pages.get(page, [])) + "</table>").encode("cp949"), None
        fid = parse_qs(data.decode("ascii"))["file"][0]
        return self.files[fid], None
```

### 5. 보안·위험·되돌리는 법
- **주소 통제**: 요청 URL은 상수와 숫자 seqnum으로만 만든다. `_guard`가 https와 `www.scourt.go.kr`·`file.scourt.go.kr` 호스트만 허용한다. 첨부 fid는 문자 검사(`^[A-Za-z0-9_.\-]{1,120}$`)를 하고, 확장자 `.hwp/.hwpx/.pdf`와 크기 20MB·파일 서명 검사는 eminwon과 같은 상수를 쓴다.
- **백신**: https이고 go.kr이므로 `vaccine_policy`가 "생략"을 낸다. 두 호스트가 모두 해당한다.
- **연락처**: 상세 HTML과 hwpx는 보관 전에 가린다. 구형 HWP는 원문 보관소(망 밖)에만 두고 글만 뽑아 쓴다. 근거 문장·`evidence_note`·읽기 불가 사유도 `mask_contacts`를 거친다. 공고 첨부는 법원별 레코드에 싣지 않는다.
- **로그**: 실패 기록(`fetch_failures`)의 URL에는 POST 본문(검색어·fid)이 들어가지 않는다. 시험으로 확인한다.
- **HttpClient.post**: 본문은 4KB 이하 바이트만 허용한다. TLS 검증·같은 출처 리다이렉트·마감 시각은 GET과 코드를 공유한다. 서버가 POST를 302로 돌리면 urllib이 GET으로 바꿔 따라가는데, 같은 출처일 때만 가능하다.
- **알려진 한계**
  - 법원명 정규식은 `…지방법원(+지원·시법원·군법원)` 형태만 안다. 가정법원·행정법원 같은 이름은 잡지 못한다.
  - 정규식은 목록 모양에 의존한다. 형식이 바뀌면 `warnings`나 `unreadable`로 드러나지만 자동 복구는 없다.
- **되돌리는 법**: 새 파일 2개(`sources/scourt.py`, `extract/court.py`)와 시험 1개를 지우고, `sources/__init__.py`의 `scourt` import를 빼면 소스가 사라진다. 기존 파일 수정은 모두 추가 위주다. `pipeline.py`와 `state.py`를 되돌릴 때는 `notices_from_raw`→`notice_from_raw`와 `unreadable`을 함께 되돌린다. `run_weekly.cmd`는 건드리지 않았다.
