/** 금고 약정 기록(039) 입력 검사. 오류 문구는 고정 문구뿐(입력값·내부 정보 없음). 서버는 링크를 열지 않는다. */
final class BankContractRules {

    static final String G_BANK = "BANK", G_RANK = "BANK_RANK", G_SRC = "BANK_CONTRACT_SRC", G_CANCEL = "BANK_CANCEL_REASON";
    static final String BANK_ETC = "ETC", CANCEL_ETC = "ETC";
    static final String RANK_FIRST = "FIRST", RANK_SECOND = "SECOND";
    static final String SRC_LINK = "LINK", SRC_DOC_NO = "DOC_NO", SRC_STAFF = "STAFF";
    static final String TYPE_LOCAL_GOV = "지자체";
    static final List<String> RANK_ORDER = Arrays.asList("FIRST", "SECOND", "NONE", "JOINT");
    private static final Pattern CODE = Pattern.compile("[A-Z0-9_]{1,30}");
    private static final Pattern TERM = Pattern.compile("\\d{4}(-\\d{2}(-\\d{2})?)?");

    /** 코드표 조회 — enabled 는 ENABLED=1 만 true(X01·X02 는 false), names 는 ENABLED=1 인 표시명. */
    interface CodeCheck {
        boolean enabled(String group, String code);
        List<String> names(String group);
    }

    static final class Parsed {
        String rank, bank, bankEtc, termFrom, termTo, srcKind, srcUrl, srcNo, checkedDate, checkedDept, note;
    }

    static final class CancelParsed {
        String reasonCd, reasonEtc;
    }

    private BankContractRules() { }

    // ---- 순위·기간 ----

    /** 1·2금고는 지자체만. TYPE 이 비어 있으면(옛 시드) 막지 않는다. */
    static boolean rankAllowed(String rank, String instType) {
        boolean ranked = RANK_FIRST.equals(rank) || RANK_SECOND.equals(rank);
        String t = instType == null ? "" : instType.trim();
        return !ranked || t.isEmpty() || TYPE_LOCAL_GOV.equals(t);
    }

    /** 겹침 검사 대상 순위 — 1금고·2금고만(순위 없음·공동 지정은 검사 안 함). */
    static boolean rankChecked(String rank) {
        return RANK_FIRST.equals(rank) || RANK_SECOND.equals(rank);
    }

    /**
     * 두 약정 기간이 겹치는가. 한쪽 끝과 다른 쪽 시작은 둘 중 더 거친 자릿수(연·연-월·일)까지 잘라 비교하고, 같으면 맞닿음(겹침 아님).
     * 시작(term_from)이 비어 있으면 그 쪽은 시작을 모르는 줄이라 '상대 끝이 먼저 오는' 경우만 겹침에서 빠진다.
     */
    static boolean overlaps(String aFrom, String aTo, String bFrom, String bTo) {
        return !(endsBeforeOrTouches(aTo, bFrom) || endsBeforeOrTouches(bTo, aFrom));
    }

    static boolean endsBeforeOrTouches(String to, String from) {
        if (to == null || from == null) return false;
        int n = Math.min(to.length(), from.length());
        return to.substring(0, n).compareTo(from.substring(0, n)) <= 0;
    }

    /** CURRENT / PAST / FUTURE — 저장값이 깨져 있으면 UNKNOWN(손으로 고친 DB 대비). */
    static String termState(String from, String to, LocalDate today) {
        try {
            if (to != null && end(to).isBefore(today)) return "PAST";
            if (from != null && start(from).isAfter(today)) return "FUTURE";
            return "CURRENT";
        } catch (RuntimeException e) {
            return "UNKNOWN";
        }
    }

    static LocalDate endOrNull(String t) {
        try { return t == null ? null : end(t); } catch (RuntimeException e) { return null; }
    }

    static String term(String raw) {
        if (raw == null) return null;
        String s = raw.trim();
        if (s.isEmpty()) return null;
        if (!TERM.matcher(s).matches()) throw bad("약정 기간은 YYYY, YYYY-MM 또는 YYYY-MM-DD 형식이어야 합니다.");
        int y = Integer.parseInt(s.substring(0, 4));
        if (y < 1990 || y > 2100) throw bad("약정 기간 범위(1990~2100)를 벗어났습니다.");
        try { start(s); end(s); } catch (DateTimeException e) { throw bad("존재하지 않는 날짜입니다."); }
        return s;
    }

    static LocalDate start(String t) {
        if (t.length() == 4) return LocalDate.of(Integer.parseInt(t), 1, 1);
        if (t.length() == 7) return YearMonth.parse(t).atDay(1);
        return LocalDate.parse(t);
    }

    static LocalDate end(String t) {
        if (t.length() == 4) return LocalDate.of(Integer.parseInt(t), 12, 31);
        if (t.length() == 7) return YearMonth.parse(t).atEndOfMonth();
        return LocalDate.parse(t);
    }

    // ---- 저장 본문 ----

    static Parsed parse(BankContractIn in, String instType, LocalDate today, CodeCheck codes) {
        Parsed p = new Parsed();
        p.rank = code(in.getRankCd(), G_RANK, codes, "허용되지 않는 금고 순위입니다.");
        if (!rankAllowed(p.rank, instType)) {
            throw bad("1금고·2금고 순위는 지자체에만 쓸 수 있습니다.");
        }
        p.bank = code(in.getBankCd(), G_BANK, codes, "허용되지 않는 은행입니다.");   // ENABLED=1 만 — X01·X02 는 여기서 400
        String etc = text(in.getBankNameEtc(), 30, "기타 은행 이름은 30자 이내여야 합니다.");
        if (BANK_ETC.equals(p.bank)) {
            if (etc == null) throw bad("기타를 고르면 은행 이름을 적어야 합니다.");
            if (looksListedBank(etc, codes.names(G_BANK))) throw bad("목록에 있는 은행은 목록에서 고르세요.");   // S3
            p.bankEtc = etc;
        } else if (etc != null) {
            throw bad("은행 이름 직접 입력은 '기타'를 고를 때만 씁니다.");
        }
        p.termFrom = term(in.getTermFrom());
        p.termTo = term(in.getTermTo());
        if (p.termTo == null) throw bad("약정 끝(YYYY, YYYY-MM 또는 YYYY-MM-DD)이 필요합니다.");
        if (p.termFrom != null && start(p.termFrom).isAfter(end(p.termTo))) throw bad("약정 시작이 끝보다 늦습니다.");

        p.srcKind = code(in.getSrcKind(), G_SRC, codes, "허용되지 않는 출처 종류입니다.");
        p.srcUrl = InstitutionEditRules.url(in.getSrcUrl());
        p.srcNo = InstitutionEditRules.notice(in.getSrcNo());
        String cd = in.getCheckedDate() == null ? null : in.getCheckedDate().trim();
        if (cd == null || cd.isEmpty()) throw bad("확인일(YYYY-MM-DD)이 필요합니다.");
        p.checkedDate = InstitutionEditRules.date(cd);
        if (LocalDate.parse(p.checkedDate).isAfter(today)) throw bad("확인일은 오늘보다 뒤일 수 없습니다.");
        p.checkedDept = text(in.getCheckedDept(), 50, "확인한 부서명은 50자 이내여야 합니다.");
        if (SRC_STAFF.equals(p.srcKind)) {
            if (p.checkedDept == null) throw bad("담당자 확인은 확인한 기관 부서명이 필요합니다.");
        } else if (p.srcUrl == null && p.srcNo == null) {
            throw bad("출처 링크 또는 문서·공고 번호가 필요합니다.");
        }
        if (SRC_LINK.equals(p.srcKind) && p.srcUrl == null) throw bad("공고·문서 링크를 고르면 링크가 필요합니다.");
        if (SRC_DOC_NO.equals(p.srcKind) && p.srcNo == null) throw bad("문서·공고 번호를 고르면 번호가 필요합니다.");
        p.note = text(in.getNote(), 100, "비고는 100자 이내여야 합니다.");

        Screen s = new Screen();
        s.free(p.bankEtc);                // 개인정보·정당·전화번호
        s.free(p.checkedDept);
        s.free(p.note);
        s.name(p.checkedDept);            // S4 사람 이름+직함. [확장 자리] note 도 보려면 s.name(p.note) 한 줄 추가(오탐이 늘어난다)
        s.ident(p.srcNo);                 // S5 번호 칸 — 개인정보·전화번호만(정당 낱말은 안 봄)
        s.ident(queryOf(p.srcUrl));       // S5 링크는 쿼리 부분만 같은 검사(경로·호스트는 안 봄)
        s.throwIfAny();
        return p;
    }

    /** 취소 사유(S10 필수) — 코드는 BANK_CANCEL_REASON, ETC 면 짧은 글(개인정보 검사). */
    static CancelParsed parseCancel(BankContractCancelIn in, CodeCheck codes) {
        CancelParsed c = new CancelParsed();
        c.reasonCd = code(in.getCancelReasonCd(), G_CANCEL, codes, "취소 사유를 목록에서 골라 주세요.");
        String etc = text(in.getCancelReasonEtc(), 50, "기타 취소 사유는 50자 이내여야 합니다.");
        if (CANCEL_ETC.equals(c.reasonCd)) {

            c.reasonEtc = etc;
        } else if (etc != null) {
            throw bad("취소 사유 직접 입력은 '기타'를 고를 때만 씁니다.");
        }
        Screen s = new Screen();
        s.free(c.reasonEtc);
        s.throwIfAny();
        return c;
    }

    static boolean same(BankContract c, Parsed p) {
        return Objects.equals(c.getRankCd(), p.rank) && Objects.equals(c.getBankCd(), p.bank)
                && Objects.equals(c.getBankNameEtc(), p.bankEtc) && Objects.equals(c.getTermFrom(), p.termFrom)
                && Objects.equals(c.getTermTo(), p.termTo) && Objects.equals(c.getSrcKind(), p.srcKind)

    }

    private static String code(String raw, String group, CodeCheck codes, String msg) {
        String v = raw == null ? null : raw.trim();
        if (v == null || !CODE.matcher(v).matches() || !codes.enabled(group, v)) throw bad(msg);
        return v;
    }

    /** 앞뒤 공백 제거, 빈 값은 null, 줄바꿈·제어 문자와 서식 문자(S7, U+202E·U+200B 등 — SafeUrl 과 같은 기준) 거부, 글자 수(코드 포인트) 상한. */
    static String text(String raw, int max, String lenMsg) {
        if (raw == null) return null;
        String v = raw.trim();
        if (v.isEmpty()) return null;
        for (int i = 0; i < v.length(); ) {
            int cp = v.codePointAt(i);
            i += Character.charCount(cp);
            if (Character.isISOControl(cp)) throw bad("줄바꿈·제어 문자는 넣을 수 없습니다.");
            if (Character.getType(cp) == Character.FORMAT) throw bad("보이지 않는 서식 문자는 넣을 수 없습니다.");
        }
        if (v.codePointCount(0, v.length()) > max) throw bad(lenMsg);
        return v;
    }

    // ---- S3: 기타 은행 이름이 목록 은행을 흉내 내는가 ----

    /** 영문 약어는 앞뒤가 영문이 아닐 때만(케이뱅크·Kbank 같은 다른 은행을 안 막으려고). 목록 은행 핵심어를 바꾸려면 이 한 줄. */
    private static final Pattern LISTED_BANK = Pattern.compile(
            "국민|농협|신한|수협|우리은행|하나은행|기업은행|부산은행|경남은행|광주은행|전북은행|제주은행|대구은행|아이엠뱅크"
                    + "|(?<![a-z])(?:kb|nh|ibk|bnk|im)(?![a-z])");

    /** NFKC·소문자·글자와 숫자만 남김(공백·기호·서식 문자 제거) 뒤 핵심어 포함 또는 코드표 이름 포함이면 true. */
    static boolean looksListedBank(String etc, List<String> codeNames) {
        String n = squash(etc);
        if (LISTED_BANK.matcher(n).find()) return true;
        for (String name : codeNames) {
            String k = squash(name);
            if (k.length() >= 3 && n.contains(k)) return true;
        }
        return false;
    }

    static String squash(String s) {
        String n = Normalizer.normalize(s, Normalizer.Form.NFKC).toLowerCase(Locale.ROOT);
        StringBuilder b = new StringBuilder(n.length());
        for (int i = 0; i < n.length(); ) {
            int cp = n.codePointAt(i);
            i += Character.charCount(cp);
            if (Character.isLetterOrDigit(cp)) b.appendCodePoint(cp);
        }
        return b.toString();
    }

    // ---- S4: 사람 이름 + 직함 ----

    private static final Pattern PERSON = Pattern.compile(
            "(?<![가-힣])([가-힣]{2,4})(\\s*)(주무관|주사보|주사|계장|팀장|과장|담당자|님|씨)(?:님)?"
                    + "(?:(?![가-힣])|(?=[은는이가의을를과와도에께]))");
    private static final String[] ORG_END = {"과", "팀", "부", "실", "국", "청", "처", "센터", "담당"};
    private static final Set<String> GENERIC = new HashSet<String>(Arrays.asList(
            "담당", "담당자", "관계자", "책임자", "직원", "고객", "해당", "실무", "업무", "부서", "기관"));
    private static final Set<String> ROLE_ONLY = new HashSet<String>(Arrays.asList("과장", "팀장", "계장"));

    /**
     * "홍길동 주무관"·"김철수 과장님"·"박민수주사" 같은 이름+직함. 부서 이름(재정과 팀장), 일반 낱말(담당자님),
     * 공백 없는 두 글자+과장·팀장·계장(재정과장 = 부서 줄임말+직함)은 통과시킨다. 사전·정규식 기반이라 새 표현은 샐 수 있다(5절).
     */
    static boolean hasPersonName(String text) {
        Matcher m = PERSON.matcher(Normalizer.normalize(text, Normalizer.Form.NFKC));
        while (m.find()) {
            String name = m.group(1), gap = m.group(2), title = m.group(3);
            if (name.length() >= 3) {
                boolean org = false;
                for (String e : ORG_END) if (name.endsWith(e)) { org = true; break; }
                if (org) continue;
            }
            if (GENERIC.contains(name)) continue;
            if (gap.isEmpty() && name.length() == 2 && ROLE_ONLY.contains(title)) continue;
            return true;
        }
        return false;
    }

    // ---- S5: 링크의 쿼리 부분 ----

    static String queryOf(String url) {
        if (url == null) return null;
        try {
            String q = new URI(url).getRawQuery();
            if (q == null) return null;
            try { return URLDecoder.decode(q, "UTF-8"); } catch (Exception e) { return q; }
        } catch (Exception e) {
            return null;
        }
    }

    // ---- 개인정보·금지어 모음 → 422 (종류만, 입력값은 싣지 않는다) ----

    private static final class Screen {
        private final Map<String, Integer> pii = new LinkedHashMap<String, Integer>();
        private final List<HeadForbidden.Kind> forb = new ArrayList<HeadForbidden.Kind>();

        /** 자유 글: 개인정보 → 정당·전화번호. */
        void free(String t) { if (t == null) return; scanPii(t); scanForbidden(t, true); }
        /** 번호·링크 쿼리: 개인정보·전화번호만(정당 낱말은 보지 않는다 — 문서 번호 오탐 방지). */
        void ident(String t) { if (t == null) return; scanPii(t); scanForbidden(t, false); }
        void name(String t) { if (t != null && hasPersonName(t)) bump("NAME"); }

        private void scanPii(String t) { for (Pii.Finding f : Pii.scanLoose(t)) bump(f.getKind()); }

        private void scanForbidden(String t, boolean party) {
            for (HeadForbidden.Kind k : HeadForbidden.scan(t)) {
                if ((k == HeadForbidden.Kind.PHONE || (party && k == HeadForbidden.Kind.PARTY)) && !forb.contains(k)) forb.add(k);
            }
        }

        private void bump(String k) { Integer n = pii.get(k); pii.put(k, n == null ? 1 : n + 1); }

        void throwIfAny() {
            if (!pii.isEmpty()) {
                List<Map<String, Object>> kinds = new ArrayList<Map<String, Object>>();
                for (Map.Entry<String, Integer> e : pii.entrySet()) {
                    Map<String, Object> k = new LinkedHashMap<String, Object>();
                    k.put("kind", e.getKey());
                    k.put("count", e.getValue());
                    kinds.add(k);
                }
                String msg = "개인정보로 보이는 값이 있어 저장하지 않았습니다. 해당 부분을 지우고 다시 입력해 주세요.";
                if (pii.containsKey("NAME")) msg += " 담당자 이름은 적지 말고 부서명만 적어 주세요.";
                throw unprocessable("PII_BLOCKED", msg, kinds);
            }
            if (!forb.isEmpty()) {
                List<Map<String, Object>> kinds = new ArrayList<Map<String, Object>>();
                for (HeadForbidden.Kind k : forb) {
                    Map<String, Object> m = new LinkedHashMap<String, Object>();
                    m.put("kind", k.name());
                    m.put("label", k.getLabel());
                    kinds.add(m);
                }
                throw unprocessable("CONTRACT_FORBIDDEN", "금고 약정 기록에 넣을 수 없는 내용(정당·전화번호)이 있습니다. 해당 부분을 지우고 다시 입력해 주세요.", kinds);
            }
        }
    }

    private static ApiException unprocessable(String code, String msg, Object kinds) {
        Map<String, Object> d = new LinkedHashMap<String, Object>();
        d.put("code", code);
        d.put("message", msg);
        d.put("kinds", kinds);
        return ApiException.withDetail(422, d);
    }

    private static ApiException bad(String m) { return ApiException.badRequest(m); }
}

    @Transactional
    public Map<String, Object> save(String instRaw, BankContractIn in, Actor actor, BoardUiIds.Ui ui) {
        String inst = requireInstId(instRaw);
        if (in == null) throw ApiException.badRequest("저장할 내용이 없습니다.");
        // ⚠️ 첫 DB 문장은 기관 행 잠금(MySQL RR 첫 일관 읽기 — NoticeRoundService.save 와 같은 이유). 위에 SELECT 를 끼우지 않는다.
        BankContractInst ref = mapper.lockInstitution(inst);
        if (ref == null) throw ApiException.notFound("institution not found");
        LocalDate today = LocalDate.parse(Times.todayKst());
        BankContractRules.Parsed p = BankContractRules.parse(in, ref.getType(), today, codeCheck());
        int contractNo;
        int rev;
        Integer excludeLine = null;   // 겹침 검사에서 뺄 자기 줄
        if (in.getContractNo() == null) {
            contractNo = mapper.selectMaxContractNo(inst) + 1;
            rev = 1;
        } else {
            BankContract cur = mapper.selectLatestLine(inst, in.getContractNo());
            if (cur == null) throw ApiException.notFound("contract not found");
            if (ACTION_CANCEL.equals(cur.getActionCd())) {
                throw conflict("ALREADY_CANCELLED", "이미 취소된 약정입니다. 새 약정으로 추가하세요.", cur);
            }
            if (!canChangeLine(cur, actor)) throw new ApiException(403, "이 약정 줄은 고칠 수 없습니다.");   // S6 자리
            if (in.getBaseRevNo() == null) throw ApiException.badRequest("고칠 때는 화면이 본 기록 번호(base_rev_no)가 필요합니다.");
            if (in.getBaseRevNo() != cur.getRevNo()) {
                throw conflict("STALE", "다른 사람이 먼저 고쳤습니다. 최신 기록을 확인하고 다시 저장하세요.", cur);
            }
            if (BankContractRules.same(cur, p)) {
                Map<String, Object> out = new LinkedHashMap<String, Object>();
                out.put("saved", false);
                out.put("code", "NO_CHANGE");
                out.put("contract_no", cur.getContractNo());
                out.put("rev_no", cur.getRevNo());
                return out;
            }
            contractNo = cur.getContractNo();
            rev = cur.getRevNo() + 1;
            excludeLine = cur.getContractNo();
        }
        // S2 — 같은 순위(1·2금고)에서 기간이 겹치는 유효 줄. 확인(confirm_overlap)하면 저장하고 줄에 겹침 표시를 남긴다.
        List<BankContract> over = findOverlaps(inst, p, excludeLine);
        boolean overlapConfirmed = false;
        if (!over.isEmpty()) {
            if (!Boolean.TRUE.equals(in.getConfirmOverlap())) throw dupRank(over);
            overlapConfirmed = true;
        }
        BankContract row = new BankContract();
        row.setInstitutionId(inst);
        row.setContractNo(contractNo);
        row.setRevNo(rev);
        row.setActionCd(ACTION_SAVE);
        row.setOriginCd(ORIGIN_MANUAL);
        row.setRankCd(p.rank);
        row.setBankCd(p.bank);
        row.setBankNameEtc(p.bankEtc);
        row.setTermFrom(p.termFrom);
        row.setTermTo(p.termTo);
        row.setSrcKind(p.srcKind);
        row.setSrcUrl(p.srcUrl);
        row.setSrcNo(p.srcNo);
        row.setCheckedDate(p.checkedDate);
        row.setCheckedDept(p.checkedDept);
        row.setNote(p.note);
        row.setOverlapConfirmed(overlapConfirmed ? 1 : 0);
        return insert(row, actor, ui, today);
    }

    @Transactional
    public Map<String, Object> cancel(String instRaw, String contractNoRaw, BankContractCancelIn in, Actor actor, BoardUiIds.Ui ui) {
        String inst = requireInstId(instRaw);
        int no = requireContractNo(contractNoRaw);
        if (in == null || in.getBaseRevNo() == null) throw ApiException.badRequest("취소할 때는 화면이 본 기록 번호(base_rev_no)가 필요합니다.");
        BankContractInst ref = mapper.lockInstitution(inst);   // 첫 DB 문장
        if (ref == null) throw ApiException.notFound("institution not found");
        BankContract cur = mapper.selectLatestLine(inst, no);
        if (cur == null) throw ApiException.notFound("contract not found");
        if (ACTION_CANCEL.equals(cur.getActionCd())) throw conflict("ALREADY_CANCELLED", "이미 취소된 약정입니다.", cur);
        if (!canChangeLine(cur, actor)) throw new ApiException(403, "이 약정 줄은 취소할 수 없습니다.");   // S6 자리
        if (in.getBaseRevNo() != cur.getRevNo()) {
            throw conflict("STALE", "다른 사람이 먼저 고쳤습니다. 최신 기록을 확인하고 다시 취소하세요.", cur);
        }
        BankContractRules.CancelParsed cp = BankContractRules.parseCancel(in, codeCheck());   // S10 사유 필수
        cur.setRevNo(cur.getRevNo() + 1);   // 값은 직전 행 그대로 복사, 행동·사유·수정자만 새로
        cur.setActionCd(ACTION_CANCEL);
        cur.setOriginCd(ORIGIN_MANUAL);
        cur.setCancelReasonCd(cp.reasonCd);
        cur.setCancelReasonEtc(cp.reasonEtc);
        return insert(cur, actor, ui, LocalDate.parse(Times.todayKst()));
    }

    private Map<String, Object> insert(BankContract row, Actor actor, BoardUiIds.Ui ui, LocalDate today) {
        SysUser me = users.selectByEmpNo(actor.getEmpNo());
        if (me == null || !HeadRules.isSales(me.getRole())) {   // 컨트롤러 canEdit 가 막는다(방어)
            throw new ApiException(403, "이 기능을 쓸 권한이 없습니다.");
        }
        row.setEditorEmpNo(actor.getEmpNo());
        row.setEditorName(me.getName());
        row.setEmpVerified(actor.isVerified() ? 1 : 0);
        row.setScreenId(ui.getScreen());
        row.setButtonId(ui.getButton());
        row.setUiValid(ui.isValid() ? 1 : 0);
        row.setSavedAt(Times.nowIso());
        try {
            mapper.insertRow(row);
        } catch (DuplicateKeyException e) {   // 잠금이 있어 정상 경로에선 오지 않는다 — PK 가 마지막 방어선
            throw conflict("STALE", "다른 사람이 먼저 고쳤습니다. 최신 기록을 확인하고 다시 저장하세요.", null);
        }
        row.setBankName(nameOf(BankContractRules.G_BANK, row.getBankCd()));
        row.setCancelReasonName(row.getCancelReasonCd() == null ? null : nameOf(BankContractRules.G_CANCEL, row.getCancelReasonCd()));
        Map<String, Object> out = new LinkedHashMap<String, Object>();
        out.put("saved", true);
        out.put("institution_id", row.getInstitutionId());
        out.put("contract_no", row.getContractNo());
        out.put("rev_no", row.getRevNo());
        out.put("action_cd", row.getActionCd());
        out.put("line", view(row, today, true));
        return out;
    }

    // ---- S2 겹침 ----

    private List<BankContract> findOverlaps(String inst, BankContractRules.Parsed p, Integer excludeLine) {
        List<BankContract> out = new ArrayList<BankContract>();
        if (!BankContractRules.rankChecked(p.rank)) return out;   // 순위 없음·공동 지정은 검사 안 함
        for (BankContract c : mapper.selectLatestValid(inst)) {
            if (excludeLine != null && c.getContractNo() == excludeLine) continue;
            if (p.rank.equals(c.getRankCd()) && BankContractRules.overlaps(c.getTermFrom(), c.getTermTo(), p.termFrom, p.termTo)) out.add(c);
        }
        Collections.sort(out, LINE_ORDER);
        return out;
    }

    private List<Integer> overlapWith(BankContract c, List<BankContract> all) {
        List<Integer> out = new ArrayList<Integer>();
        if (!BankContractRules.rankChecked(c.getRankCd())) return out;
        for (BankContract o : all) {
            if (o.getContractNo() != c.getContractNo() && c.getRankCd().equals(o.getRankCd())
                    && BankContractRules.overlaps(c.getTermFrom(), c.getTermTo(), o.getTermFrom(), o.getTermTo())) {
                out.add(o.getContractNo());
            }
        }
        return out;
    }

    private static ApiException dupRank(List<BankContract> over) {
        List<Map<String, Object>> list = new ArrayList<Map<String, Object>>();
        for (BankContract c : over) {
            Map<String, Object> m = new LinkedHashMap<String, Object>();
            m.put("contract_no", c.getContractNo());
            m.put("rev_no", c.getRevNo());
            m.put("rank_cd", c.getRankCd());
            m.put("bank_name", bankLabel(c));
            m.put("term_from", c.getTermFrom());
            m.put("term_to", c.getTermTo());
            m.put("editor_name", c.getEditorName());
            list.add(m);
        }
        Map<String, Object> d = new LinkedHashMap<String, Object>();
        d.put("code", "DUP_RANK");
        d.put("message", "같은 순위에 기간이 겹치는 약정이 있습니다. 확인한 뒤 겹침을 알고 저장하려면 확인 표시를 하고 다시 저장하세요.");
        d.put("overlaps", list);
        return ApiException.withDetail(409, d);
    }

    private static List<Map<String, Object>> codeList(List<BankContractCode> rows, String rankInstType) {
        List<Map<String, Object>> out = new ArrayList<Map<String, Object>>();
        for (BankContractCode c : rows) {
            if (rankInstType != null && !BankContractRules.rankAllowed(c.getCode(), rankInstType)) continue;

            m.put("code", c.getCode());
            m.put("name", c.getCodeName());

    /** 1금고→2금고→순위 없음→공동, 같은 순위면 끝 날짜 늦은 것, 그다음 줄 번호 큰 것. */
    private static final Comparator<BankContract> LINE_ORDER = new Comparator<BankContract>() {
        @Override
        public int compare(BankContract a, BankContract b) {
            int ra = rankIdx(a.getRankCd()), rb = rankIdx(b.getRankCd());
            if (ra != rb) return ra - rb;
            LocalDate ea = BankContractRules.endOrNull(a.getTermTo()), eb = BankContractRules.endOrNull(b.getTermTo());
            if (ea != null && eb != null && !ea.equals(eb)) return eb.compareTo(ea);
            if (ea == null ^ eb == null) return ea == null ? 1 : -1;
            return b.getContractNo() - a.getContractNo();
        }
    };

    private static int rankIdx(String r) {
        int i = BankContractRules.RANK_ORDER.indexOf(r);
        return i < 0 ? 99 : i;
    }

    private static String bankLabel(BankContract c) {
        return BankContractRules.BANK_ETC.equals(c.getBankCd()) ? c.getBankNameEtc() : c.getBankName();
    }

    /** 화면용 한 줄. 사번·화면·버튼·UI_VALID 는 내보내지 않는다(감사 전용). emp_verified=false 는 자기신고 표시(표지는 화면). */
    private static Map<String, Object> view(BankContract c, LocalDate today, boolean withAction) {
        Map<String, Object> m = new LinkedHashMap<String, Object>();
        m.put("contract_no", c.getContractNo());
        m.put("rev_no", c.getRevNo());
        if (withAction) m.put("action_cd", c.getActionCd());
        m.put("origin_cd", c.getOriginCd());
        m.put("rank_cd", c.getRankCd());
        m.put("bank_cd", c.getBankCd());
        m.put("bank_name", bankLabel(c));
        m.put("bank_name_etc", c.getBankNameEtc());
        m.put("term_from", c.getTermFrom());
        m.put("term_to", c.getTermTo());
        m.put("term_state", BankContractRules.termState(c.getTermFrom(), c.getTermTo(), today));
        m.put("src_kind", c.getSrcKind());
        m.put("src_url", c.getSrcUrl());
        m.put("src_no", c.getSrcNo());
        m.put("checked_date", c.getCheckedDate());
        m.put("checked_dept", c.getCheckedDept());
        m.put("note", c.getNote() == null ? "" : c.getNote());
        m.put("overlap_confirmed", c.getOverlapConfirmed() == 1);   // 저장할 때 겹침을 알고 확인한 줄
        if (withAction) {
            m.put("cancel_reason_cd", c.getCancelReasonCd());
            m.put("cancel_reason_name", BankContractRules.CANCEL_ETC.equals(c.getCancelReasonCd()) ? c.getCancelReasonEtc() : c.getCancelReasonName());
        }
        m.put("editor_name", c.getEditorName());
        m.put("emp_verified", c.getEmpVerified() == 1);
        m.put("saved_at", c.getSavedAt());
        return m;
    }

    private static String requireInstId(String raw) {
        if (!InstitutionHeadService.validInstId(raw)) throw ApiException.notFound("institution not found");
        return raw;
    }

    private static int requireContractNo(String raw) {
        if (raw == null || !raw.matches("[1-9][0-9]{0,8}")) throw ApiException.notFound("contract not found");
        return Integer.parseInt(raw);
    }

    private static ApiException conflict(String code, String message, BankContract cur) {
        Map<String, Object> d = new LinkedHashMap<String, Object>();
        d.put("code", code);
        d.put("message", message);
        Map<String, Object> latest = null;
        if (cur != null) {
            latest = new LinkedHashMap<String, Object>();
            latest.put("contract_no", cur.getContractNo());
            latest.put("rev_no", cur.getRevNo());
            latest.put("action_cd", cur.getActionCd());
            latest.put("editor_name", cur.getEditorName());
            latest.put("saved_at", cur.getSavedAt());
        }
        d.put("latest", latest);
        return ApiException.withDetail(409, d);
    }

## 5. 보안·위험·되돌리는 법
- **권한 경로**
  - `requireInstitutionEdit`(401/403, 접근 이력 DENIED) → `canEdit`(영업팀 계열 아니면 403, DENIED) → 서비스 안 방어 403.
  - 조회는 `require(map)`입니다.
  - 자기신고 한계: `Actor`의 사번·역할은 X-Emp-No 자기신고 위의 업무 규칙이라 사칭을 막지 못합니다. 앞단 인증 전에는 보안 경계가 아닙니다.
  - 서버는 `emp_verified`를 응답에 내보냅니다. `false`일 때 화면은 '검증됨·확인됨' 표현을 쓰지 말고 자기신고 표지를 달아야 합니다(프런트 몫).
- **접근 이력**
  - DETAIL은 `bank-contract save no=1 rev=1` / `save 409 DUP_RANK` 같은 번호·코드 문구뿐입니다.
  - 은행·비고·부서·링크·번호·취소 사유 글은 남기지 않습니다.
  - 422 응답도 종류와 건수만 싣고 입력값은 싣지 않습니다.
- **입력**
  - S3·S4·S5·S7과 취소 사유 ETC의 개인정보 검사를 반영했습니다.
  - 링크는 `SafeUrl`로 http(s)만 받고 서버는 열지 않습니다.
  - 알려진 한계
    - 이름 검사는 사전·정규식 기반이라 새 표현·주소는 못 잡습니다.
    - note에는 이름 검사를 걸지 않았습니다(오탐이 늘어서). 필요하면 `parse()`에 `s.name(p.note)` 한 줄 추가입니다.
    - HTML은 서버가 거르지 않으므로 저장형 XSS 대비는 프런트 escape가 필요합니다(security S11).
- **동시성**
  - 기관 행 FOR UPDATE + 복합 PK입니다. 겹침 검사와 줄 번호 발급이 같은 잠금 안에서 이루어집니다.
  - 같은 기관의 공고 판정 저장(037)·기관 편집과 짧게 직렬화됩니다.
- **남은 위험·정책 대기**
  - S6 남의 줄 수정·취소: 지금은 누구나 가능합니다. `BankContractService.canChangeLine`의 `return true`가 정책 자리입니다. `save()` 고치기와 `cancel()`이 둘 다 그 메서드를 부릅니다. 바꾸면 BankContractApiTest에 남의 줄 시험을 추가합니다.
  - S1 수정자 이름 사칭은 앞단 인증 전까지 남는 알려진 한계입니다. 서버는 `emp_verified`만 내보냅니다.
  - 겹침은 시작이 빈 줄을 "시작을 모름"으로 보아 상대 끝이 먼저 오는 경우를 빼면 겹침으로 봅니다. 오탐은 `confirm_overlap`으로 넘어가며, 저장된 줄에 `overlap_confirmed=true`로 남습니다.
  - 지역농협·지역수협 같은 조합이 S3 핵심어에 걸립니다. `농협`·`수협`을 핵심어로 넣었고 X01·X02는 비활성이라 "기타" 입력 경로가 막힙니다. 아래 이관 메모의 기획·데이터 확인이 필요합니다.
  - `ops/integrity_check.sql`·`db/README.md`의 039 줄은 다른 세션이 수정 중이라 이번에도 뺐습니다. 점검 쿼리는 직전 보고서 1049~1055행과 같습니다. 그 쿼리 뒤에 취소 행의 사유 코드 누락 점검을 더하는 것이 좋습니다.
    ```sql
    SELECT INSTITUTION_ID, CONTRACT_NO, REV_NO FROM INST_BANK_CONTRACT WHERE ACTION_CD='CANCEL' AND CANCEL_REASON_CD IS NULL;
    ```
- **되돌리기**
  - 코드: 새 파일(domain 3·dto 2·mapper 2·Rules·Service·Controller)을 삭제하고, `InstUiIds` 추가분을 제거하고, 시험 설정 34행과 CommonCodeSchemaTest 변경을 원복합니다.
  - DB: `DROP TABLE INST_BANK_CONTRACT;` + `DELETE FROM COMMON_CODE WHERE CODE_GROUP IN ('BANK','BANK_RANK','BANK_CONTRACT_SRC','BANK_CANCEL_REASON');`. 운영 데이터가 생긴 뒤에는 하지 않습니다.
  - 기존 `/institutions`·`contractExpiry`·`OURS_SCOPE`·`/menus`는 바꾸지 않았습니다.
