### 2. 요구/원인 요약 (근거)
- ① url 칸은 이미 `InstitutionEditRules.url` → `SafeUrl.isHttp`(InstitutionEditRules.java:274)를 부른다. BMP 서식 문자(U+200B·U+202E)는 이미 막히고, 시험도 있다(InstitutionEditRulesTest.java:49).
  - 실제 구멍은 `SafeUrl.java:19-23`이 `char` 단위로 훑는 것이다. 보조면 서식 문자(예: 태그 문자 U+E0001, `󠀁`)는 서로게이트 쌍이라 `getType(char)`가 FORMAT이 아니어서 통과한다.
  - 그래서 고칠 곳은 SafeUrl(codePoint 단위)이다. url 칸과 PlanCsv가 함께 쓰므로 둘 다 막힌다.
  - 같은 자리에서 C1 제어(U+0085)·줄/문단 구분(U+2028/9)·NBSP류 공백·짝 없는 서로게이트도 거부한다. 기준이 조금 더 엄격해지는 것이므로 PlanCsv 시험에 영향이 있는지 확인이 필요하다.
- TYPE·LAST_BID는 길이 검사가 없다(InstitutionEditRules.java:53-54). 컬럼은 TYPE 50, LAST_BID 40이다(mysql 001:29, 004:14 / oracle 001:35, 004:31, 모두 CHAR 의미). 넘기면 DB 오류로 500이 된다.
- ② 임시 파일은 이미 대부분 처리돼 있다. `LIVE_DIRS`와 exit 훅, SIGINT·SIGTERM·SIGHUP 처리가 apply-mysql.mjs:188-210에 있다. 남은 구멍은 두 가지다.
  - 강제 종료(`taskkill /F`·SIGKILL)로 끝나면 폴더가 남는다. 다음 실행 때 오래된 `apply-mysql-XXXXXX` 폴더를 치우는 방식으로 막는다.
  - Windows의 Ctrl+Break(SIGBREAK)는 처리기가 없다.
- ③ `ApiExceptionHandler.java:108`의 `log.error("처리되지 않은 예외", e)`가 스택 전체와 예외 메시지(SQL 값 등이 섞일 수 있음)를 남긴다. 응답 계약(`{"detail": GENERIC_500}`)은 기존 테스트가 고정하므로 바꾸지 않는다.
- ④ `--baseline`은 번호 이하를 일괄 기록한다(apply-mysql.mjs:389). 중간 번호만 골라 기록하는 길이 없다. 033·036·038 등을 골라, 실제 스키마를 확인한 뒤 sha256만 기록하는 `--mark-applied`를 추가한다.

export const SCHEMA_PROBES = {
  '033': "SELECT IF(COUNT(*)>=1,1,0) FROM ROLE_MENUS WHERE ROLE='전산팀장' AND MENU='admin';",   // 관리자가 끈 뒤에도 행은 남는다

  '038': "SELECT IF(COUNT(*)>=1,1,0) FROM INSTITUTIONS WHERE INSTITUTION_ID='org-9740520' AND REGION_CODE='47';",   // 데이터 정정이라 증거가 약하다(대표 1행)

/** versions → { targets(파일), skipped(이미 기록), errors }. 파일 없음·검사 없음은 errors. */
export function planMark(versions, files, applied) {
  const byVer = new Map(files.map((f) => [f.version, f]));
  const targets = [], skipped = [], errors = [];
  for (const v of versions) {
    const f = byVer.get(v);
    if (!f) errors.push(`${v}: 해당 번호의 파일이 없습니다`);
    else if (applied.has(v)) skipped.push(v);
    else if (!SCHEMA_PROBES[v]) errors.push(`${v}: 적용 여부를 확인할 검사가 없어 기록하지 않습니다(SCHEMA_PROBES 에 추가 필요)`);
    else targets.push(f);
  }
  return { targets, skipped, errors };
}

/** 오래된(기본 1시간) apply-mysql-XXXXXX 임시 폴더를 지운다 — 강제 종료로 finally·exit 훅이 못 돈 경우의 마지막 방어. 지운 폴더 이름 목록을 돌려준다. */
export function sweepStaleTempDirs(io = {}) {
  const { tmpdir = os.tmpdir(), now = Date.now(), maxAgeMs = 3600000 } = io;
  const removed = [];
  let names;
  try { names = readdirSync(tmpdir); } catch { return removed; }
  for (const n of names) {
    if (!/^apply-mysql-[A-Za-z0-9]{6}$/.test(n)) continue;
    const p = path.join(tmpdir, n);
    try {
      const st = lstatSync(p);   // 링크는 따라가지 않는다
      if (!st.isDirectory() || now - st.mtimeMs < maxAgeMs) continue;
      rmSync(p, { recursive: true, force: true });
      removed.push(n);
    } catch { /* 건드릴 수 없으면 그대로 둔다 */ }
  }
  return removed;
}

async function doMarkApplied(cfg, files, applied, opt) {
  if (applied.size === 0) {
    console.error('중단: 적용 기록이 비어 있습니다. 먼저 --baseline <번호> 로 앞 번호를 기록하세요(안 그러면 다음 실행이 001 부터 다시 돕니다).');
    return 2;
  }
  const { targets, skipped, errors } = planMark(opt.markApplied, files, applied);
  for (const v of skipped) console.log(`이미 기록됨: ${v}`);
  if (errors.length) {
    for (const e of errors) console.error(`중단: ${e}`);
    console.error('아무것도 기록하지 않았습니다.');
    return 2;
  }
  const missing = [];
  for (const f of targets) {
    const r = await query(cfg, SCHEMA_PROBES[f.version]);
    if (r.code !== 0) throw new Error(`${f.name} 검사 쿼리 실패: ${r.stderr.trim() || '오류'}`);
    if (r.stdout.trim() !== '1') missing.push(f.name);
  }
  if (missing.length) {
    console.error(`중단: 다음 파일은 DB 에 적용된 흔적이 없습니다 — ${missing.join(', ')}\n먼저 mysql 로 적용하거나 번호를 빼세요. 아무것도 기록하지 않았습니다.`);
    return 2;
  }
  if (targets.length === 0) { console.log('기록할 파일이 없습니다.'); return 0; }
  if (opt.dryRun) {
    console.log('[dry-run] 검사 통과, 기록 예정:'); for (const f of targets) console.log(`  ${f.name}`);
    return 0;
  }
  const r = await query(cfg, insertSql(targets, `${who()} (mark-applied)`));   // 현재 파일의 sha256 만 기록한다
  if (r.code !== 0) { console.error(`기록 실패: ${r.stderr.trim()}`); return 1; }
  console.log(`실행 없이 기록했습니다: ${targets.map((f) => f.name).join(', ')}`);
  return 0;
}

### 5. 보안·위험·되돌리는 법
- 권한 검사·403 경로·접근 이력은 이번 변경에서 바뀌지 않는다. 기관 PUT 권한은 InstitutionCommandService 쪽 기존 흐름 그대로다. 인증은 여전히 `X-Emp-No` 자기신고이고, 이 패치는 입력 형식·로그 내용만 다룬다.
- ③ 로그에는 클래스 이름·상관 ID만 남고 메시지·쿼리값·본문은 없다. `X-Error-Ref`는 무작위 8자리 hex이며 내부 정보가 없다. 스택은 DEBUG를 켜야 나온다. 운영에서 DEBUG를 켜면 메시지가 스택에 섞일 수 있으니 켜지 않는다.
- ④ 위험: 검사를 통과해도 "그 파일의 모든 문장이 적용됐다"는 보증은 아니다(MySQL DDL은 중간까지만 적용될 수 있다). 특히 033·038은 데이터 증거가 약하다. 다음 실행은 현재 파일의 sha256을 기록으로 보므로, 이후 파일을 고치면 '변경됨!'으로 멈춘다. 기록 값은 `applied_by`에 `(mark-applied)`로 남아 구분된다.
- ② sweep 위험: 1시간 이내 폴더는 건드리지 않는다. 다른 사용자의 같은 이름 패턴 폴더를 지울 수 있지만, `%TEMP%`는 사용자별이고 이름 패턴·디렉터리·시간을 모두 만족할 때만 지운다.
- 되돌리는 법: 각 파일을 `git checkout`으로 되돌린다. 잘못 기록한 DB는 `DELETE FROM SCHEMA_MIGRATION WHERE VERSION IN ('033', …);`로 지운다(사람이 확인한 뒤 수동).
