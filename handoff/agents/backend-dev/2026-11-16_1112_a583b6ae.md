```java
    /**
     * 열기 쪽이 읽는 중앙 목차를 java.util.zip.ZipFile 로 읽어 스트림 검사 결과와 대조한다.
     * 이름 집합·개수가 다르거나(스트림에 없는 entry, 스트림에만 있는 entry) entry 별 실제 해제 크기가 다르면 BROKEN.
     * 목차의 크기 필드는 쓰지 않고 InputStream 으로 실제 읽어 센다(한도 초과는 TOO_LARGE_UNPACKED).
     */
    private static void verifyDirectory(Path file, Result stream, long maxEntry, long maxTotal) throws Rejected {
        Set<String> seen = new java.util.HashSet<>();
        long total = 0;
        byte[] buf = new byte[8192];
        try (ZipFile zf = new ZipFile(file.toFile())) {
            Enumeration<? extends ZipEntry> en = zf.entries();
            while (en.hasMoreElements()) {
                ZipEntry e = en.nextElement();
                if (seen.size() >= MAX_ENTRIES) {
                    throw new Rejected("TOO_LARGE_UNPACKED");
                }
                String name = normalize(e.getName());
                if (!seen.add(name)) {
                    throw new Rejected("DUPLICATE_ENTRY");
                }
                Long expected = stream.sizes.get(name);
                if (expected == null) {
                    throw new Rejected("BROKEN");           // 목차에만 있는 entry = 가짜 서명 뒤 숨김
                }
                long size = 0;
                int n;
                try (InputStream is = zf.getInputStream(e)) {
                    while ((n = is.read(buf)) > 0) {
                        size += n;
                        total += n;
                        if (size > maxEntry || total > maxTotal) {
                            throw new Rejected("TOO_LARGE_UNPACKED");
                        }
                    }
                }
                if (size != expected.longValue()) {
                    throw new Rejected("BROKEN");
                }
            }
        } catch (IOException | IllegalArgumentException ex) {   // ZipException 포함
            throw new Rejected("BROKEN");
        }
        if (seen.size() != stream.sizes.size()) {
            throw new Rejected("BROKEN");                       // 스트림에만 있는 entry
        }
    }

    private static String normalize(String raw) {
        String name = raw.replace('\\', '/').toLowerCase(Locale.ROOT);
        while (name.startsWith("/")) {
            name = name.substring(1);
        }
        return name;
    }
```

```java
// new (scan 전체 교체)
    private static Result scan(InputStream src, long maxEntry, long maxTotal, int maxCells, int maxRows,
                               int maxCellChars, int maxXmlElements) throws Rejected {
        Map<String, Long> sizes = new LinkedHashMap<>();
        long[] counts = new long[3];   // [0]=칸, [1]=행, [2]=시작 요소 총수
        long[] total = new long[1];
        byte[] buf = new byte[8192];
        try (ZipInputStream in = new ZipInputStream(src)) {
            ZipEntry e;
            while ((e = in.getNextEntry()) != null) {
                String name = normalize(e.getName());
                if (sizes.containsKey(name)) {
                    throw new Rejected("DUPLICATE_ENTRY");
                }
                if (sizes.size() >= MAX_ENTRIES) {
                    throw new Rejected("TOO_LARGE_UNPACKED");
                }
                Meter m = new Meter(in, maxEntry, maxTotal, total);
                int filled = 0;
                int r;
                while (filled < SNIFF && (r = m.read(buf, filled, SNIFF - filled)) > 0) {
                    filled += r;                                  // 앞부분만 먼저 읽어 XML 인지 본다
                }
                if (looksXml(buf, filled)) {
                    InputStream xml = new SequenceInputStream(new ByteArrayInputStream(buf, 0, filled), m);
                    countXml(xml, m, counts, maxCells, maxRows, maxCellChars, maxXmlElements);
                }
                while (m.read(buf, 0, buf.length) > 0) {
                    // StAX 가 끝까지 안 읽은 나머지(뒤 공백 등)·비 XML entry 를 읽어 센다
                }
                sizes.put(name, m.size);
            }
        } catch (Over ex) {
            throw new Rejected("TOO_LARGE_UNPACKED");
        } catch (IOException | IllegalArgumentException ex) {   // 깨진 zip·잘못된 이름 인코딩
            throw new Rejected("BROKEN");
        }
        if (sizes.isEmpty()) {
            throw new Rejected("BROKEN");
        }
        return new Result(sizes, total[0]);
    }

    private static final int SNIFF = 64;

    /** 한도 초과 신호(Meter 가 던진다). */
    private static final class Over extends IOException {
        Over() {
            super("over");
        }
    }

    /** entry 하나를 읽으며 entry·전체 해제 바이트를 센다. close 는 아무것도 안 한다(ZipInputStream 을 닫지 않는다). */
    private static final class Meter extends InputStream {
        private final InputStream in;
        private final long maxEntry;
        private final long maxTotal;
        private final long[] total;
        long size;
        boolean over;

        Meter(InputStream in, long maxEntry, long maxTotal, long[] total) {
            this.in = in;
            this.maxEntry = maxEntry;
            this.maxTotal = maxTotal;
            this.total = total;
        }

        @Override
        public int read() throws IOException {
            byte[] one = new byte[1];
            int n = read(one, 0, 1);
            return n < 0 ? -1 : (one[0] & 0xFF);
        }

        @Override
        public int read(byte[] b, int off, int len) throws IOException {
            int n = in.read(b, off, len);
            if (n > 0) {
                size += n;
                total[0] += n;
                if (size > maxEntry || total[0] > maxTotal) {
                    over = true;
                    throw new Over();
                }
            }
            return n;
        }

        @Override
        public void close() {
            // 닫지 않는다
        }
    }
```

```java
// new (countXml 전체 교체; 시그니처의 첫 인자가 byte[] → InputStream + Meter, getElementText 제거)
    private static void countXml(InputStream xml, Meter meter, long[] counts, int maxCells, int maxRows,
                                 int maxCellChars, int maxXmlElements) throws Rejected {
        XMLInputFactory f = XMLInputFactory.newInstance();
        f.setProperty(XMLInputFactory.SUPPORT_DTD, false);
        f.setProperty(XMLInputFactory.IS_SUPPORTING_EXTERNAL_ENTITIES, false);
        XMLStreamReader r = null;
        try {
            r = f.createXMLStreamReader(xml);
            boolean inStr = false;
            boolean inPhonetic = false;
            long len = 0;
            while (r.hasNext()) {
                int ev = r.next();
                if (ev == XMLStreamConstants.DTD) {
                    throw new Rejected("BROKEN");
                }
                if (ev == XMLStreamConstants.START_ELEMENT) {
                    if (++counts[2] > maxXmlElements) {
                        throw new Rejected("TOO_MANY_CELLS");
                    }
                    String n = r.getLocalName();
                    if ("c".equals(n)) {
                        if (++counts[0] > maxCells) {
                            throw new Rejected("TOO_MANY_CELLS");
                        }
                    } else if ("row".equals(n)) {
                        if (++counts[1] > maxRows) {
                            throw new Rejected("TOO_MANY_CELLS");
                        }
                    } else if ("si".equals(n) || "is".equals(n)) {
                        inStr = true;
                        inPhonetic = false;
                        len = 0;
                    } else if ("rPh".equals(n)) {
                        inPhonetic = true;
                    } else if ("t".equals(n) && inStr && !inPhonetic) {
                        // getElementText() 는 문자열 전체를 만들므로 쓰지 않고, 글자 수만 청크별로 센다.
                        while (r.hasNext()) {
                            int tev = r.next();
                            if (tev == XMLStreamConstants.CHARACTERS || tev == XMLStreamConstants.CDATA
                                    || tev == XMLStreamConstants.SPACE) {
                                len += r.getTextLength();
                                if (len > maxCellChars) {
                                    throw new Rejected("CELL_TOO_LONG");
                                }
                            } else if (tev == XMLStreamConstants.END_ELEMENT) {
                                break;
                            } else if (tev == XMLStreamConstants.START_ELEMENT || tev == XMLStreamConstants.DTD) {
                                throw new Rejected("BROKEN");
                            }
                        }
                    }
                } else if (ev == XMLStreamConstants.END_ELEMENT) {
                    String n = r.getLocalName();
                    if ("si".equals(n) || "is".equals(n)) {
                        inStr = false;
                    } else if ("rPh".equals(n)) {
                        inPhonetic = false;
                    }
                }
            }
        } catch (XMLStreamException | RuntimeException ex) {
            if (meter.over) {                       // Meter 의 한도 초과가 파서 예외로 감싸졌을 수 있다
                throw new Rejected("TOO_LARGE_UNPACKED");
            }
            throw new Rejected("BROKEN");
        } finally {
            if (r != null) {
                try {
                    r.close();
                } catch (XMLStreamException ignore) {
                    // 닫기 실패는 판정에 영향 없음
                }
            }
        }
    }
```
- `Rejected`는 `Exception`의 하위 클래스라서 catch 절의 `RuntimeException`에 걸리지 않는다. 원본과 동작이 같다.
- StAX가 `Over`(IOException)를 `XMLStreamException`으로 감싸지 않고 그대로 던지면, `countXml`이 선언한 `throws Rejected`로는 컴파일되지 않을 수 있다. 그때는 `catch (XMLStreamException | RuntimeException ex)` 앞에 `catch (IOException ex)`를 추가한다. 본문은 `meter.over`이면 TOO_LARGE_UNPACKED, 아니면 BROKEN으로 같게 쓴다. 감싸는 경우는 위 코드가 처리한다.

```java
    /**
     * 두 번째 로컬 헤더 앞에 가짜 서명 4바이트(PK\1\2)를 끼운다. ZipInputStream 은 거기서 끝나 첫 entry 만 보고,
     * ZipFile 은 목차(오프셋·EOCD 시작 위치를 +4 로 고쳐 둠)로 둘 다 연다. 입력은 entry 가 2개 이상인 zip.
     */
    public static byte[] hideAfterFirst(byte[] zip) {
        int p = -1;
        for (int i = 4; i + 4 <= zip.length; i++) {
            if (zip[i] == 'P' && zip[i + 1] == 'K' && zip[i + 2] == 3 && zip[i + 3] == 4) {
                p = i;
                break;
            }
        }
        int eocd = -1;
        for (int i = zip.length - 22; i >= 0; i--) {
            if (zip[i] == 'P' && zip[i + 1] == 'K' && zip[i + 2] == 5 && zip[i + 3] == 6) {
                eocd = i;
                break;
            }
        }
        if (p < 0 || eocd < 0) {
            throw new IllegalArgumentException("zip 구조를 못 찾음");
        }
        byte[] out = new byte[zip.length + 4];
        System.arraycopy(zip, 0, out, 0, p);
        out[p] = 'P';
        out[p + 1] = 'K';
        out[p + 2] = 1;
        out[p + 3] = 2;
        System.arraycopy(zip, p, out, p + 4, zip.length - p);
        int e = eocd + 4;
        int cen = le32(out, e + 16);
        put32(out, e + 16, cen + 4);
        int count = le16(out, e + 10);
        int pos = cen + 4;
        for (int k = 0; k < count; k++) {
            int off = le32(out, pos + 42);
            if (off >= p) {
                put32(out, pos + 42, off + 4);
            }
            pos += 46 + le16(out, pos + 28) + le16(out, pos + 30) + le16(out, pos + 32);
        }
        return out;
    }

    private static int le16(byte[] b, int i) {
        return (b[i] & 0xFF) | ((b[i + 1] & 0xFF) << 8);
    }

    private static int le32(byte[] b, int i) {
        return le16(b, i) | (le16(b, i + 2) << 16);
    }

    private static void put32(byte[] b, int i, int v) {
        b[i] = (byte) v;
        b[i + 1] = (byte) (v >>> 8);
        b[i + 2] = (byte) (v >>> 16);
        b[i + 3] = (byte) (v >>> 24);
    }
```

```java
            Files.write(p, o.toByteArray());
        }
        assertFalse(ZipGuard.checkUploadFile(p).names.isEmpty());
        Path h = tmp.newFile("n.hwpx").toPath();
        Files.write(h, ZipFixtures.addEntry(ZipFixtures.zip("mimetype", "application/hwp+zip".getBytes(StandardCharsets.UTF_8)),
                "Contents/section0.xml", "<a><t>가</t></a>".getBytes(StandardCharsets.UTF_8)));
        assertTrue(ZipGuard.checkUploadFile(h).names.contains("contents/section0.xml"));
    }

    @Test
    public void 큰_t_글자수는_스트리밍으로_세어_CELL_TOO_LONG() throws Exception {
        StringBuilder sb = new StringBuilder("<sst><si><t>");
        for (int i = 0; i < 100; i++) sb.append("가나다라마바사아자차");   // 1000자
        sb.append("</t></si></sst>");
        byte[] z = ZipFixtures.zip("xl/sharedStrings.xml", sb.toString().getBytes(StandardCharsets.UTF_8));
        try {
            ZipGuard.check(z, 20 * MB, 40 * MB, Integer.MAX_VALUE, Integer.MAX_VALUE, 500);
            fail("거부되어야 한다");
        } catch (ZipGuard.Rejected e) {
            assertEquals("CELL_TOO_LONG", e.code);
        }
    }

    @Test
    public void XML_파트의_엔트리_한도_초과는_TOO_LARGE_UNPACKED() throws Exception {
        byte[] big = new byte[(int) (2 * MB)];
        java.util.Arrays.fill(big, (byte) ' ');
        byte[] xml = ("<a/>" + new String(big, StandardCharsets.UTF_8)).getBytes(StandardCharsets.UTF_8);
        assertEquals("TOO_LARGE_UNPACKED", code(ZipFixtures.zip("x.xml", xml), MB, 10 * MB));
    }
```
`code()`는 `check(byte[],maxEntry,maxTotal)`를 쓴다. 이 마지막 시험으로 스트리밍 경로에서도 한도가 지켜지는지 확인한다.

**새 파일 `uploader/src/test/java/com/kb/uploader/parse/RfpExtractorHwpxGuardTest.java`**
```java
package com.kb.uploader.parse;

import org.junit.Rule;
import org.junit.Test;
import org.junit.rules.TemporaryFolder;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;

import static org.junit.Assert.*;

/** NEXT 39 2단계: RFP 의 .hwpx 도 열기 전에 ZipGuard 를 거친다. */
public class RfpExtractorHwpxGuardTest {
    @Rule
    public TemporaryFolder tmp = new TemporaryFolder();

    private static byte[] hwpx() throws Exception {
        return ZipFixtures.addEntry(ZipFixtures.zip("mimetype", "application/hwp+zip".getBytes(StandardCharsets.UTF_8)),
                "Contents/section0.xml", "<a/>".getBytes(StandardCharsets.UTF_8));
    }

    private void assertGuardRejected(byte[] data, String code) throws Exception {
        Path p = tmp.newFile("r_" + System.nanoTime() + ".hwpx").toPath();
        Files.write(p, data);
        try {
            new RfpExtractor().extractHwpx(p);
            fail("거부되어야 한다");
        } catch (IOException e) {
            assertEquals("ZIP 거부: " + code, e.getMessage());
            assertFalse(e.getMessage().contains(p.toString()));
        }
    }

    @Test
    public void 중복_entry_hwpx는_거부() throws Exception {
        assertGuardRejected(ZipFixtures.duplicateEntry(hwpx(), "Contents/section0.xml"), "DUPLICATE_ENTRY");
    }

    @Test
    public void 목차_불일치_hwpx는_거부() throws Exception {
        assertGuardRejected(ZipFixtures.hideAfterFirst(hwpx()), "BROKEN");
    }

    @Test
    public void DOCTYPE_hwpx는_거부() throws Exception {
        assertGuardRejected(ZipFixtures.zip("Contents/section0.xml",
                "<!DOCTYPE a [<!ENTITY e \"x\">]><a>&e;</a>".getBytes(StandardCharsets.UTF_8)), "BROKEN");
    }

    @Test
    public void 가드를_통과한_파일은_ZIP_거부_메시지가_아니다() throws Exception {
        Path p = tmp.newFile("ok.hwpx").toPath();
        Files.write(p, hwpx());
        try {
            new RfpExtractor().extractHwpx(p);   // 최소 구조라 hwpxlib 에서 다른 이유로 실패하거나 빈 결과일 수 있다
        } catch (Exception e) {
            assertFalse(String.valueOf(e.getMessage()).startsWith("ZIP 거부"));
        }
    }
}
```
- `mimetype` entry가 첫 번째라 `hideAfterFirst`가 두 번째 로컬 헤더(`Contents/section0.xml`)를 숨긴다.
- `addEntry`는 `ZipOutputStream`으로 다시 쓰므로 `duplicateEntry(hwpx(), "Contents/section0.xml")`의 끝 글자 치환 방식이 그대로 동작한다.

**깨질 기존 테스트**
- 예상은 없다.
  - `FileContentServiceTest`의 hwpx 중복·DOCTYPE 시험은 스트림 단계에서 먼저 거부된다.
  - `ZipFixtures`로 만든 정상 zip은 `ZipOutputStream`이 쓴 것이라 목차와 헤더가 일치한다.
- 확인할 위험: `FileContentServiceTest`에 손으로 만든 이상한 zip이 있다면 새 목차 대조에서 BROKEN이 될 수 있다.
- `ZipGuard.Result.contentTypes` 필드를 삭제한다. 테스트와 main에서 쓰는 곳은 Grep으로 찾지 못했다.

## 5. 보안·위험·되돌리는 법
- **효과**
  - 가짜 서명 뒤 숨김 우회가 막힌다. 파일 입력에서 목차 이름 집합·개수·실제 해제 크기가 스트림 결과와 다르면 BROKEN이다.
  - 중복 entry가 목차 쪽에만 있어도 DUPLICATE_ENTRY로 거부된다.
  - hwpx RFP도 가드를 거친다.
  - XML 파트를 메모리에 모으지 않는다. 버퍼 약 230MB 문제는 8KB 청크 중심으로 바뀐다. 5번은 가능하다고 판단해 같이 제안했다.
- **비용**: 파일 입력은 압축을 두 번 푼다(스트림 한 번, `ZipFile` 한 번). 한도 안이라 시간만 늘고 메모리는 늘지 않는다. 최대 약 200MB 해제 두 번이다.
- **남는 위험**
  - sjsxp가 `<t>` 안의 텍스트를 한 청크로 줄 수 있다. 그러면 큰 단일 텍스트 노드는 그만큼 메모리를 쓴다. 아래 이관 메모 5번처럼 실측이 필요하다.
  - 가드를 통과한 뒤 POI·hwpxlib가 잡는 힙은 여전히 별도다. 칸 5M·요소 30M 한도는 web과 같은 수치이고, 힙 기준 재산정은 기획과 운영 결정이다.
  - `InstitutionService`의 `byte[]` 입력은 InputStream으로 여는 경로라 목차 대조가 없다. 이 경로는 스트림 검사로 충분하다.
  - POI의 `WorkbookFactory.create(File)`과 JDK `ZipFile`의 해석 차이는 완전히 같다고 보장할 수 없다.
  - `ZipFile`과 `ZipInputStream`의 문자셋은 둘 다 UTF-8로 같다. 같은 이름의 비UTF-8 entry는 `IllegalArgumentException`에서 BROKEN이 된다.
- **오류·로그**: 모든 거부는 코드만 쓴다. 경로·내용은 응답에 싣지 않는다. 로그 위조 여지(entry 이름을 로그에 남기는 `FileContentService` 183·186줄)는 이번 범위 밖이다.
- **되돌리는 법**: `ZipGuard.java`, `RfpExtractor.java`, `ZipFixtures.java`의 변경을 git으로 되돌리고 새 테스트 파일을 지우면 된다. DB와 `kgi-ggreport-web`은 건드리지 않는다.

## 6. 이관 메모
- `[이관: 기획]` `/api/files/search`가 GET마다 `extractText`를 실행한다. 존치·삭제·경량화·캐시 중 정한다. 운영 호출부가 있는지 확인이 먼저다. 기다리는 것은 결정이다. 근거는 `FileSearchApiController.java:38-54`, `FileContentService.java:46-48`(주석의 "삭제되었다"가 사실과 다르다).
- `[이관: 기획]` 암호화 OOXML(OLE 안의 `EncryptedPackage`) 업로드를 허용할지 정한다. `isZip=false`라 ZipGuard를 건너뛴다. 거부하면 `POIFS`에서 `EncryptedPackage`를 찾는 추가 검사가 필요하다. 근거는 `FileContentService.java:202`, `PptExtractor.java:26`.
- `[이관: 백엔드/web 쪽, 변경 안 함]` web `docfill/ZipGuard`도 같은 구조일 가능성이 있다. 스트림 검사만 하고 XML을 `ByteArrayOutputStream`에 모은다면, 목차 대조와 스트리밍을 같이 고칠지 web 세션에서 판단한다. 이번에는 읽지 않았고 바꾸지 않았다.
- `[이관: 백엔드]` 5번 residual: 큰 단일 `<t>` 노드 실측과 WebLogic 힙 기준 한도 재산정. 이 설계로는 실측하지 못했다.
- 로그에 entry 이름이 그대로 남는 것(security 6번)은 별건이다. 줄바꿈 제거 정도의 정리 후보로 남긴다.
