**`src/main/resources/db/oracle/036_admin_audit.sql` (새 파일, 머리 주석은 MySQL과 같은 뜻)**
```sql
-- (머리 주석 동일. 11g 호환: IDENTITY 없음, ADD … DEFAULT … NOT NULL 과 MODIFY NULL 만 쓴다. 적용은 DBA.)
-- 되돌리기: DELETE FROM PERM_CHANGE_LOG WHERE TARGET_EMP_NO IS NULL; DROP INDEX IDX_PERM_CHG_ROLE;
--   ALTER TABLE PERM_CHANGE_LOG DROP (SCREEN_ID, BUTTON_ID, UI_VALID, EMP_VERIFIED, TARGET_ROLE);
--   ALTER TABLE PERM_CHANGE_LOG MODIFY (TARGET_EMP_NO NOT NULL);
ALTER TABLE PERM_CHANGE_LOG ADD (
    SCREEN_ID    VARCHAR2(30 CHAR) DEFAULT 'UNKNOWN' NOT NULL,
    BUTTON_ID    VARCHAR2(30 CHAR) DEFAULT 'UNKNOWN' NOT NULL,
    UI_VALID     NUMBER(1)         DEFAULT 0 NOT NULL,
    EMP_VERIFIED NUMBER(1)         DEFAULT 0 NOT NULL,
    TARGET_ROLE  VARCHAR2(50 CHAR)
);
ALTER TABLE PERM_CHANGE_LOG MODIFY (TARGET_EMP_NO NULL);
CREATE INDEX IDX_PERM_CHG_ROLE ON PERM_CHANGE_LOG (TARGET_ROLE, CHANGED_AT);
```
- H2(MODE=Oracle)가 `MODIFY (TARGET_EMP_NO NULL)`에서 실패하면 그 줄을 `ALTER TABLE PERM_CHANGE_LOG MODIFY (TARGET_EMP_NO VARCHAR2(7 CHAR) NULL);`로 바꾼다. Oracle 11g에서도 유효한 문장이다.
- **이미 적용한 로컬 MySQL은** 034(와 035가 있으면 그것까지) 다음에 036을 한 번만 돌린다. 036은 PERM_CHANGE_LOG만 건드리므로 035보다 먼저 돌려도 안전하다.
- 참고(범위 밖): oracle 008·012·034 자체가 IDENTITY(12c)를 쓴다. 그래서 운영 DB가 정말 11g라면 036과 상관없이 008부터 안 맞는다 → 열린 질문 1.

```java
package com.kbstar.kgi.ggreport.web.support;

import javax.servlet.http.HttpServletRequest;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashSet;
import java.util.Set;

/**
 * 시스템 관리 쓰기(/sysadmin/**·PUT /menus)가 싣는 "어느 화면·어느 버튼"({@code X-Screen-Id}·{@code X-Button-Id}).
 * {@link BoardUiIds} 와 같은 규칙 — 허용 목록 밖·없음은 원문을 저장하지 않고 UNKNOWN(400 아님).
 * ⚠️ 클라이언트 자기신고 값이다. 서버가 확인하는 것은 "허용된 이름인가"뿐이다. 화면을 만들면 여기와 admin.js 상수를 같이 더한다.
 */
public final class AdminUiIds {

    public static final Set<String> SCREENS = Collections.unmodifiableSet(new HashSet<String>(Arrays.asList(
            "admin_users", "admin_perms", "admin_menus")));
    public static final Set<String> BUTTONS = Collections.unmodifiableSet(new HashSet<String>(Arrays.asList(
            "user_create", "user_save", "user_lock", "user_unlock", "user_retire", "user_restore",
            "approver_save", "perm_save", "menu_save")));

    private AdminUiIds() {
    }

    public static BoardUiIds.Ui of(HttpServletRequest req) {
        String s = clean(req.getHeader(BoardUiIds.H_SCREEN));
        String b = clean(req.getHeader(BoardUiIds.H_BUTTON));
        boolean sv = s != null && SCREENS.contains(s);
        boolean bv = b != null && BUTTONS.contains(b);
        return new BoardUiIds.Ui(sv ? s : BoardUiIds.UNKNOWN, bv ? b : BoardUiIds.UNKNOWN, sv && bv);
    }

    private static String clean(String v) {
        return v == null ? null : v.trim();
    }
}
```

**`service/PermChangeRows.java` (새 파일, 패키지 전용)** — 서비스 네 곳에 중복된 `log()` 몸통을 하나로 모은다.

```java
package com.kbstar.kgi.ggreport.web.service;

import com.kbstar.kgi.ggreport.web.domain.PermChange;
import com.kbstar.kgi.ggreport.web.support.BoardUiIds;
import com.kbstar.kgi.ggreport.web.web.Actor;

/** PERM_CHANGE_LOG 1행 — 누가(사번·검증 여부)·어느 화면·어느 버튼. 사번이 자기신고면 EMP_VERIFIED=0. */
final class PermChangeRows {

    private PermChangeRows() {
    }

    static PermChange of(Actor actor, BoardUiIds.Ui ui, String targetEmpNo, String item,
                         String before, String after, String now) {
        BoardUiIds.Ui u = ui == null ? BoardUiIds.unknown() : ui;
        PermChange c = new PermChange();
        c.setChangedAt(now);
        c.setActorEmpNo(actor == null ? null : actor.getEmpNo());
        c.setEmpVerified(actor != null && actor.isVerified());
        c.setTargetEmpNo(targetEmpNo);
        c.setItem(item);
        c.setBeforeValue(before);
        c.setAfterValue(after);
        c.setScreenId(u.getScreen());
        c.setButtonId(u.getButton());
        c.setUiValid(u.isValid());
        return c;
    }
}
```

```java
import com.kbstar.kgi.ggreport.web.web.Actor;

/**
 * 관리자는 자기 자신의 정보·상태·결재자·권한을 바꾸지 않는다 — 다른 관리자가 바꾼다(2026-11-16 사용자 결정).

final class SelfChange {

    private SelfChange() {
    }

    static void refuse(Actor actor, String targetEmpNo, String message) {
        if (actor != null && actor.isKnown() && actor.getEmpNo().equals(targetEmpNo)) {
            throw new ApiException(400, message);
        }
    }
}
```

## 5. 보안·위험·되돌리는 법
- **자기신고의 한계**
  - 본인 변경 차단(두 사람 규칙)은 X-Emp-No가 자기신고라서 다른 관리자 사번을 사칭하면 통과한다. 실수와 '혼자 처리'를 막는 업무 규칙이지 사칭 방지가 아니다. SSO가 붙으면(EMP_VERIFIED=1) 의미가 생긴다.
  - 화면·버튼 값도 자기신고다. 서버는 "허용된 이름인가"만 확인한다.
  - 화면에서도 EMP_VERIFIED=0 행을 '검증됨'으로 보이면 안 된다.
- **403·400 경로**
  - 관리 API의 401/403은 기존 AdminGuard가 DENIED로 ACCESS_LOG에 남긴다(S7도 같다).
  - 본인 변경 400은 업무 거부라 DENIED로 남기지 않는다. ACCESS_LOG에는 API 400으로 남는다.
  - 오류 문구는 고정 문구이고 내부 정보가 없다.
- **이력 내용**: PERM_CHANGE_LOG에는 사번, 항목, 이전·이후 값(역할·직위·등급·이름, 1/0), 화면, 버튼만 남는다. 요청 본문 원문, 쿼리값, 비밀번호는 넣지 않는다. 이름은 기존 user:NAME 변경 이력과 같은 수준이다.
- **업무 결과 잠금("관리자가 결과를 못 바꾸게")**
  - 참여 결정과 최종 확정은 USERS.ROLE이 정확히 일치해야 한다(`MenuGuard.requireRole:115-123`). 관리자 역할로는 할 수 없다.
  - 남는 우회로는 "관리자가 자기 역할을 영업부장으로 바꾸기"였는데 S1이 막는다.
  - "관리자가 공범의 역할을 바꾸기"는 막지 못한다. 대신 이제 화면·버튼까지 이력에 남는다.
- **S6 위험**
  - 증적은 서비스 커밋 뒤 별도 트랜잭션에 기록한다(`create:107`의 선례와 같다). 기록이 실패하면 응답은 500인데 결정·확정은 이미 반영된 상태다. 서버 로그에는 ERROR 한 줄이 남는다.
  - 같은 트랜잭션에 묶으려면 PrepAudit을 BidCaseCommandService에 주입해야 한다. 이번에는 선례를 따랐다.
- **S7 위험**: 지금 화면에 호출자가 없어 영향이 없다. 영업 계열이 운영 절차로 이 API를 써 왔다면 403이 된다.
- **되돌리는 법**
  - Java만 되돌려도 된다. 036 칼럼이 남아 있어도 기존 INSERT는 기본값(UNKNOWN/0)으로 채워지고, TARGET_EMP_NO NULL 허용도 해가 없다.
  - DB까지 되돌리려면 036 머리 주석의 문장을 쓴다. 역할 행을 지우면 이력이 사라진다(운영 금지).

## 7. 열린 질문
1. 운영 DB가 실제로 Oracle 11g인가? 그렇다면 기존 008·012·024·034의 IDENTITY부터 맞지 않는다. 036은 11g에서도 된다.
2. 관리자가 한 명뿐일 때의 권장안은 **예외 없이 막기**다.
   - 이유: 마스터 예외를 두면 자기신고 사번 하나로 다시 혼자 바꿀 수 있게 된다.
   - 혼자 남은 관리자는 먼저 다른 사용자에게 권한관리를 주고(대상이 본인이 아니라 허용된다), 그 사람이 바꾼다. 비상시에는 DBA가 SQL로 처리한다.
   - 운영 시작 전에 활성 관리자가 2명 이상인지 확인해야 한다(033 이후 전산팀장 + 전산팀 역할 기본).
   - 이 안으로 가도 되는지 사용자 확인이 필요하다.
3. `/institutions/{id}/corpus/validate`(읽기 전용)도 관리자 전용으로 바꿀까? 바꾸면 CorpusApiTest의 검사 호출 6건에 머리글을 더해야 한다.
