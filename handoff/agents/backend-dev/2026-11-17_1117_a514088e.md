    /**
     * scanExternal + 한 글자씩 띄운 구간("0 1 0 1 2 3 …", 8쌍 이상)의 공백을 지워 한 번 더 본다(security 하-2).
     * 전체 공백 제거는 오탐이 커서 쓰지 않는다. PDF·HWPX 본문처럼 글자 사이가 벌어진 추출물용이다.
     */
    public static List<Finding> scanExternalDeep(String text, long deadlineNanos) {
        List<Finding> out = scanExternal(text, deadlineNanos);
        String spaced = spacedRuns(text, deadlineNanos);
        if (spaced != null) {
            out.addAll(scanExternal(spaced, deadlineNanos));
        }
        return out;
    }

    private static String spacedRuns(String text, long deadlineNanos) {
        if (text == null) {
            return null;
        }
        StringBuilder out = null;
        int n = text.length();
        int i = 0;
        while (i < n) {
            if ((i & 0x3FF) == 0) {
                checkDeadline(deadlineNanos);
            }
            if (!isSpaceLike(text.charAt(i)) && i + 1 < n && text.charAt(i + 1) == ' ') {
                StringBuilder run = new StringBuilder();
                int j = i;
                int pairs = 0;
                while (j + 1 < n && !isSpaceLike(text.charAt(j)) && text.charAt(j + 1) == ' ') {
                    run.append(text.charAt(j));
                    j += 2;
                    pairs++;
                }
                if (j < n && !isSpaceLike(text.charAt(j))) {
                    run.append(text.charAt(j));
                    j++;
                }
                if (pairs >= 8) {
                    if (out == null) {
                        out = new StringBuilder();
                    }
                    out.append(run).append('\n');
                }
                i = Math.max(j, i + 1);
            } else {
                i++;
            }
        }
        return out == null ? null : out.toString();
    }

    private static boolean isSpaceLike(char c) {
        return Character.isWhitespace(c) || c == '\u00A0' || c == '\u3000';
    }

            return pdf ? inspectPdf(file, deadline) : inspectHwpx(file, deadline);
        } catch (Reject r) {
            return single(r.reason, r.locator, r.count);
        } catch (Pii.ScanTimeoutException | TimeoutSignal t) {
            return single("TIMEOUT", "file", 1);
        } catch (IOException | RuntimeException e) {
            return single("PARSE_FAILED", "file", 1);
        }
    }

    private static String sniff(Path file) throws IOException {
        byte[] head = new byte[8];
        int n = 0;
        try (InputStream in = Files.newInputStream(file)) {
            int r;
            while (n < head.length && (r = in.read(head, n, head.length - n)) > 0) {
                n += r;
            }
        }
        if (n >= 5 && head[0] == 0x25 && head[1] == 0x50 && head[2] == 0x44 && head[3] == 0x46 && head[4] == 0x2D) {
            return "pdf";
        }
        if (n >= 4 && head[0] == 0x50 && head[1] == 0x4B && head[2] == 0x03 && head[3] == 0x04) {
            return "zip";
        }
        if (n >= 4 && (head[0] & 0xFF) == 0xD0 && (head[1] & 0xFF) == 0xCF && head[2] == 0x11 && (head[3] & 0xFF) == 0xE0) {
            return "ole";
        }
        return "other";
    }

    // ── PDF ───────────────────────────────────────────────────────────

    private static List<Hit> inspectPdf(Path file, long deadline) throws IOException {
        byte[] raw = Files.readAllBytes(file);
        if (count(raw, "startxref") >= 2) {
            throw new Reject("INCREMENTAL", "xref", 1);                 // 증분 저장은 이전 판의 객체가 남는다
        }
        PDDocument doc;
        try {
            doc = PDDocument.load(file.toFile(), MemoryUsageSetting.setupMixed(8 * MB));
        } catch (InvalidPasswordException e) {
            throw new Reject("ENCRYPTED", "file", 1);
        }
        raw = null;
        try {
            if (doc.isEncrypted()) {
                throw new Reject("ENCRYPTED", "file", 1);
            }
            int pages = doc.getNumberOfPages();
            if (pages > MAX_PDF_PAGES) {
                throw new Reject("PAGES_LIMIT", "file", pages);
            }
            if (doc.getDocument().getTrailer().containsKey(COSName.PREV)) {
                throw new Reject("INCREMENTAL", "xref", 1);
            }
            Reject active = findActive(doc);
            if (active != null) {
                throw active;
            }
            Hits hits = new Hits();
            long chars = 0;
            long nonSpace = 0;
            PDFTextStripper st = new PDFTextStripper();
            for (int p = 1; p <= pages; p++) {
                checkTime(deadline);
                st.setStartPage(p);
                st.setEndPage(p);
                String t = st.getText(doc);
                chars += t.length();
                if (chars > MAX_TEXT_CHARS) {
                    throw new Reject("TEXT_LIMIT", "body", 1);
                }
                nonSpace += nonSpace(t);
                scanText(hits, t, "body:p" + p, deadline);
            }
            // 문서 정보 전 항목(이름·값)
            PDDocumentInformation info = doc.getDocumentInformation();
            if (info != null) {
                StringBuilder sb = new StringBuilder();
                COSDictionary d = info.getCOSObject();
                for (COSName k : new ArrayList<COSName>(d.keySet())) {
                    sb.append(k.getName()).append('\n');
                    String v = str(d.getDictionaryObject(k));
                    if (v != null) {
                        sb.append(v).append('\n');
                    }
                }
                scanText(hits, sb.toString(), "meta", deadline);
            }
            // XMP
            PDMetadata md = doc.getDocumentCatalog().getMetadata();
            if (md != null) {
                try (InputStream in = md.exportXMPMetadata()) {
                    byte[] xmp = readLimited(in, (int) MB);
                    if (xmp == null) {
                        throw new Reject("TEXT_LIMIT", "xmp", 1);
                    }
                    scanText(hits, new String(xmp, StandardCharsets.UTF_8), "xmp", deadline);
                }
            }
            // 주석 · 양식 · 책갈피 · 이미지 수
            int images = 0;
            for (int p = 0; p < pages; p++) {
                checkTime(deadline);
                PDPage page = doc.getPage(p);
                for (PDAnnotation a : page.getAnnotations()) {
                    COSDictionary ad = a.getCOSObject();
                    scanText(hits, join(ad, "Contents", "T", "Subj", "RC"), "annot", deadline);
                }
                if (page.getResources() != null) {
                    COSBase xo = page.getResources().getCOSObject().getDictionaryObject(COSName.XOBJECT);
                    if (xo instanceof COSDictionary) {
                        COSDictionary xd = (COSDictionary) xo;
                        for (COSName k : new ArrayList<COSName>(xd.keySet())) {
                            COSBase v = xd.getDictionaryObject(k);
                            if (v instanceof COSStream && COSName.IMAGE.equals(((COSStream) v).getCOSName(COSName.SUBTYPE))) {
                                images++;
                            }
                        }
                    }
                }
            }
            PDAcroForm form = doc.getDocumentCatalog().getAcroForm();
            if (form != null) {
                for (PDField f : form.getFieldTree()) {
                    checkTime(deadline);
                    scanText(hits, join(f.getCOSObject(), "V", "DV", "TU", "T"), "form", deadline);
                }
            }
            PDDocumentOutline outline = doc.getDocumentCatalog().getDocumentOutline();
            if (outline != null) {
                StringBuilder sb = new StringBuilder();
                int[] budget = {MAX_OUTLINE_ITEMS};
                outlineText(outline.children(), sb, budget, 0);
                scanText(hits, sb.toString(), "bookmark", deadline);
            }
            if (images > 0 && pages > 0 && nonSpace < (long) PDF_MIN_CHARS_PER_PAGE * pages) {
                hits.add("IMAGE_ONLY", "body", images);
            }
            return hits.list();
        } finally {
            doc.close();
        }
    }

    private static void outlineText(Iterable<PDOutlineItem> items, StringBuilder sb, int[] budget, int depth) {
        if (depth > 20) {
            return;
        }
        for (PDOutlineItem it : items) {
            if (budget[0]-- <= 0) {
                return;
            }
            String t = it.getTitle();
            if (t != null) {
                sb.append(t).append('\n');
            }
            if (it.hasChildren()) {
                outlineText(it.children(), sb, budget, depth + 1);
            }
        }
    }

    /**
     * 카탈로그·트레일러에서 닿는 모든 객체를 훑어 능동 내용을 찾는다(압축된 객체 스트림 안도 COSObject 가 풀어 준다).
     * 거부: JS·JavaScript·XFA, EmbeddedFiles·EF·RichMedia·3D·FileAttachment, Launch·JavaScript 동작.
     * /OpenAction·/AA 아래에서는 URI·SubmitForm·ImportData·GoToR·GoToE 도 거부한다(링크 주석의 URI 는 허용 — 사용자가 눌러야 열린다).
     */
    private static Reject findActive(PDDocument doc) {
        Set<COSBase> plain = Collections.newSetFromMap(new IdentityHashMap<COSBase, Boolean>());
        Set<COSBase> auto = Collections.newSetFromMap(new IdentityHashMap<COSBase, Boolean>());
        Deque<Object[]> stack = new ArrayDeque<Object[]>();
        stack.push(new Object[]{doc.getDocument().getTrailer(), Boolean.FALSE});
        int nodes = 0;
        while (!stack.isEmpty()) {
            if (++nodes > MAX_GRAPH_NODES) {
                return new Reject("PARSE_FAILED", "graph", 1);
            }
            Object[] e = stack.pop();
            COSBase b = (COSBase) e[0];
            boolean isAuto = ((Boolean) e[1]).booleanValue();
            if (b instanceof COSObject) {
                b = ((COSObject) b).getObject();
            }
            if (b == null || !(isAuto ? auto : plain).add(b)) {
                continue;
            }
            if (b instanceof COSArray) {
                COSArray arr = (COSArray) b;
                for (int i = 0; i < arr.size(); i++) {
                    stack.push(new Object[]{arr.get(i), Boolean.valueOf(isAuto)});
                }
            } else if (b instanceof COSDictionary) {
                COSDictionary d = (COSDictionary) b;
                Reject r = checkDict(d, isAuto);
                if (r != null) {
                    return r;
                }
                for (COSName k : new ArrayList<COSName>(d.keySet())) {
                    String key = k.getName();
                    if ("Parent".equals(key)) {
                        continue;
                    }
                    boolean childAuto = isAuto || "OpenAction".equals(key) || "AA".equals(key);
                    stack.push(new Object[]{d.getItem(k), Boolean.valueOf(childAuto)});
                }
            }
        }
        return null;
    }

    private static Reject checkDict(COSDictionary d, boolean auto) {
        if (d.containsKey("JS") || d.containsKey("JavaScript") || d.containsKey("XFA")) {
            return new Reject("SCRIPT", "active", 1);
        }
        if (d.containsKey("EmbeddedFiles") || d.containsKey("EF") || d.containsKey("RichMediaContent")) {
            return new Reject("EMBEDDED", "active", 1);
        }
        if ("EmbeddedFile".equals(d.getNameAsString("Type"))) {
            return new Reject("EMBEDDED", "active", 1);
        }
        String sub = d.getNameAsString("Subtype");
        if ("FileAttachment".equals(sub) || "RichMedia".equals(sub) || "3D".equals(sub)) {
            return new Reject("EMBEDDED", "active", 1);
        }
        String s = d.getNameAsString("S");
        if (s != null && (ACTION_ALWAYS.contains(s) || (auto && ACTION_AUTO.contains(s)))) {
            return new Reject("SCRIPT", "active", 1);
        }
        return null;
    }

    private static List<Hit> inspectHwpx(Path file, long deadline) throws IOException {
        long ratioCap = Math.max(MB, ZIP_RATIO * Files.size(file));
        Hits hits = new Hits();
        try (ZipFile zf = new ZipFile(file.toFile())) {
            if (zf.size() > ZIP_MAX_ENTRIES) {
                throw new Reject("ZIP_LIMIT", "entries", zf.size());
            }
            long[] total = {0};
            long[] chars = {0};
            long[] body = {0};
            boolean mimeOk = false;
            int images = 0;
            Enumeration<? extends ZipEntry> en = zf.entries();
            while (en.hasMoreElements()) {
                checkTime(deadline);
                ZipEntry e = en.nextElement();
                String name = e.getName();
                if (!safeEntryName(name)) {
                    throw new Reject("FORMAT_MISMATCH", "entry", 1);
                }
                if (e.isDirectory()) {
                    continue;
                }
                String low = name.toLowerCase(Locale.ROOT);
                if (low.startsWith("scripts/")) {
                    throw new Reject("SCRIPT", "entry", 1);
                }
                String ext = extOf(low);
                boolean isText = XML_EXT.contains(ext) || "txt".equals(ext);
                boolean isMime = "mimetype".equals(low);
                boolean isManifest = "meta-inf/manifest.xml".equals(low);
                Drained d;
                try (InputStream in = zf.getInputStream(e)) {
                    d = drain(in, isText || isMime, total, ratioCap, deadline);
                } catch (ZipException ze) {
                    String m = ze.getMessage();
                    throw new Reject(m != null && m.toLowerCase(Locale.ROOT).contains("encrypt") ? "ENCRYPTED" : "PARSE_FAILED", "entry", 1);
                }
                if (d.nested) {
                    throw new Reject("EMBEDDED", "entry", 1);          // 중첩 zip·OLE·실행 파일
                }
                if (low.startsWith("bindata/")) {
                    if (!IMAGE_EXT.contains(ext)) {
                        throw new Reject("EMBEDDED", "entry", 1);
                    }
                    images++;
                    continue;
                }
                if (isMime) {
                    if (!"application/hwp+zip".equals(new String(d.data, StandardCharsets.UTF_8).trim())) {
                        throw new Reject("FORMAT_MISMATCH", "mimetype", 1);
                    }
                    mimeOk = true;
                    continue;
                }
                if (isManifest && containsAscii(d.data, "encryption-data")) {
                    throw new Reject("ENCRYPTED", "entry", 1);
                }
                if (isText) {
                    scanEntry(low, ext, d.data, hits, chars, body, deadline);
                }
            }
            if (!mimeOk) {
                throw new Reject("FORMAT_MISMATCH", "mimetype", 1);
            }
            if (images > 0 && body[0] < HWPX_MIN_BODY_CHARS) {
                hits.add("IMAGE_ONLY", "body", images);
            }
            return hits.list();
        }
    }

    private static void scanEntry(String low, String ext, byte[] data, Hits hits, long[] chars, long[] body, long deadline) {
        String loc = low.startsWith("contents/section") ? "body" : ("txt".equals(ext) ? "preview" : "meta");
        if ("txt".equals(ext)) {
            String s = decodeText(data);
            addChars(chars, s.length());
            scanText(hits, s, loc, deadline);
            return;
        }
        if (containsAscii(data, "<!doctype") || containsAscii(data, "<!entity")) {
            throw new Reject("XXE_REJECTED", "entry", 1);
        }
        StringBuilder text = new StringBuilder();
        StringBuilder attrs = new StringBuilder();
        XMLStreamReader r = null;
        try {
            r = factory().createXMLStreamReader(new ByteArrayInputStream(data));
            while (r.hasNext()) {
                int ev = r.next();
                if (ev == XMLStreamConstants.CHARACTERS || ev == XMLStreamConstants.CDATA) {
                    text.append(r.getText());
                } else if (ev == XMLStreamConstants.COMMENT) {
                    attrs.append(r.getText()).append('\n');
                } else if (ev == XMLStreamConstants.PROCESSING_INSTRUCTION) {
                    attrs.append(r.getPIData()).append('\n');
                } else if (ev == XMLStreamConstants.START_ELEMENT) {
                    for (int i = 0; i < r.getAttributeCount(); i++) {
                        attrs.append(r.getAttributeValue(i)).append('\n');
                    }
                } else if (ev == XMLStreamConstants.END_ELEMENT) {
                    if (BREAKS.contains(r.getLocalName())) {
                        text.append('\n');
                    }
                } else if (ev == XMLStreamConstants.DTD || ev == XMLStreamConstants.ENTITY_REFERENCE) {
                    throw new Reject("XXE_REJECTED", "entry", 1);
                }
                if (text.length() + attrs.length() > MAX_TEXT_CHARS) {
                    throw new Reject("TEXT_LIMIT", loc, 1);
                }
            }
        } catch (XMLStreamException x) {
            throw new Reject("PARSE_FAILED", "entry", 1);
        } finally {
            if (r != null) {
                try {
                    r.close();
                } catch (XMLStreamException ignore) {
                    // 닫기 실패는 판정에 영향 없음
                }
            }
        }
        addChars(chars, text.length() + attrs.length());
        if ("body".equals(loc)) {
            body[0] += nonSpace(text.toString());
        }
        scanText(hits, text.toString(), loc, deadline);
        scanText(hits, attrs.toString(), loc, deadline);
    }

    private static XMLInputFactory factory() {
        XMLInputFactory f = XMLInputFactory.newInstance();
        f.setProperty(XMLInputFactory.SUPPORT_DTD, false);
        f.setProperty(XMLInputFactory.IS_SUPPORTING_EXTERNAL_ENTITIES, false);
        try {
            f.setProperty(XMLConstants.ACCESS_EXTERNAL_DTD, "");
        } catch (IllegalArgumentException ignore) {
            // 구현체가 모르는 속성이면 DOCTYPE 사전 차단이 막는다
        }
        return f;
    }

    private static final class Drained {
        byte[] data;
        boolean nested;
    }

    private static Drained drain(InputStream in, boolean keep, long[] total, long ratioCap, long deadline) throws IOException {
        ByteArrayOutputStream bo = keep ? new ByteArrayOutputStream() : null;
        byte[] buf = new byte[16384];
        byte[] head = new byte[4];
        int hl = 0;
        long n = 0;
        int r;
        while ((r = in.read(buf)) != -1) {
            n += r;
            total[0] += r;
            if (n > ZIP_ENTRY_MAX || total[0] > ZIP_TOTAL_MAX || total[0] > ratioCap) {
                throw new Reject("ZIP_LIMIT", "entry", 1);
            }
            if (hl < 4) {
                int c = Math.min(4 - hl, r);
                System.arraycopy(buf, 0, head, hl, c);
                hl += c;
            }
            if (bo != null) {
                bo.write(buf, 0, r);
            }
            checkTime(deadline);
        }
        Drained d = new Drained();
        d.data = bo == null ? null : bo.toByteArray();
        d.nested = (hl >= 4 && head[0] == 0x50 && head[1] == 0x4B && head[2] == 0x03 && head[3] == 0x04)
                || (hl >= 4 && (head[0] & 0xFF) == 0xD0 && (head[1] & 0xFF) == 0xCF && head[2] == 0x11 && (head[3] & 0xFF) == 0xE0)
                || (hl >= 2 && head[0] == 0x4D && head[1] == 0x5A);
        return d;
    }

    private static boolean safeEntryName(String n) {
        return !n.isEmpty() && n.length() <= 260 && !n.contains("..") && !n.startsWith("/")
                && n.indexOf('\\') < 0 && n.indexOf(':') < 0;
    }

    private static String decodeText(byte[] d) {
        Charset cs = StandardCharsets.UTF_8;
        int off = 0;
        if (d.length >= 2 && (d[0] & 0xFF) == 0xFF && (d[1] & 0xFF) == 0xFE) {
            cs = StandardCharsets.UTF_16LE;
            off = 2;
        } else if (d.length >= 2 && (d[0] & 0xFF) == 0xFE && (d[1] & 0xFF) == 0xFF) {
            cs = StandardCharsets.UTF_16BE;
            off = 2;
        } else if (d.length >= 3 && (d[0] & 0xFF) == 0xEF && (d[1] & 0xFF) == 0xBB && (d[2] & 0xFF) == 0xBF) {
            off = 3;
        }
        return new String(d, off, d.length - off, cs).replace("\u0000", "");
    }

    private static void scanText(Hits hits, String text, String locator, long deadline) {
        if (text == null || text.isEmpty()) {
            return;

        for (Pii.Finding f : Pii.scanExternalDeep(text, deadline)) {
            hits.add(reasonOf(f.getKind()), locator, 1);

    private static void checkTime(long deadline) {
        if (Thread.currentThread().isInterrupted() || System.nanoTime() - deadline > 0) {

        }
    }

    private static void addChars(long[] chars, int n) {
        chars[0] += n;

        }
    }

    private static long nonSpace(String s) {
        long c = 0;
        for (int i = 0; i < s.length(); i++) {
            if (!Character.isWhitespace(s.charAt(i))) {
                c++;
            }
        }
        return c;
    }

    private static String join(COSDictionary d, String... keys) {
        StringBuilder sb = new StringBuilder();
        for (String k : keys) {
            String v = str(d.getDictionaryObject(k));
            if (v != null) {
                sb.append(v).append('\n');
            }
        }

    }

    private static String str(COSBase b) {
        if (b instanceof COSObject) {
            b = ((COSObject) b).getObject();
        }
        if (b instanceof COSString) {
            return ((COSString) b).getString();
        }
        if (b instanceof COSName) {
            return ((COSName) b).getName();
        }
        if (b instanceof COSStream) {
            try {
                return ((COSStream) b).toTextString();
            } catch (RuntimeException e) {
                return null;
            }
        }
        if (b instanceof COSArray) {
            StringBuilder sb = new StringBuilder();
            COSArray a = (COSArray) b;
            for (int i = 0; i < a.size(); i++) {
                String s = str(a.get(i));
                if (s != null) {
                    sb.append(s).append('\n');
                }
            }
            return sb.toString();
        }
        return null;
    }

    private static byte[] readLimited(InputStream in, int max) throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        byte[] buf = new byte[8192];
        int r;
        long n = 0;
        while ((r = in.read(buf)) != -1) {
            n += r;
            if (n > max) {
                return null;
            }
            out.write(buf, 0, r);
        }
        return out.toByteArray();
    }

    private static int count(byte[] data, String ascii) {
        byte[] p = ascii.getBytes(StandardCharsets.US_ASCII);
        int c = 0;
        outer:
        for (int i = 0; i + p.length <= data.length; i++) {
            for (int j = 0; j < p.length; j++) {
                if (data[i + j] != p[j]) {
                    continue outer;
                }
            }
            c++;
            i += p.length - 1;
        }
        return c;
    }

    /** 대소문자를 무시하고 NUL 바이트는 건너뛴다(UTF-16 로 쓴 DOCTYPE 도 잡는다). pat 은 소문자 ASCII. */
    private static boolean containsAscii(byte[] d, String pat) {
        int m = 0;
        for (byte b : d) {
            if (b == 0) {
                continue;
            }
            char c = (char) (b & 0xFF);
            if (c >= 'A' && c <= 'Z') {
                c += 32;
            }
            if (c == pat.charAt(m)) {
                m++;
                if (m == pat.length()) {
                    return true;
                }
            } else {
                m = c == pat.charAt(0) ? 1 : 0;
            }
        }
        return false;
    }

    private static Outcome one(String ext, String reason, String locator) {
        return new Outcome(ext, single(reason, locator, 1));
    }

    private static List<Hit> single(String reason, String locator, int count) {
        List<Hit> l = new ArrayList<Hit>();
        l.add(new Hit(reason, locator, count));
        return l;
    }

    private static Set<String> set(String... v) {
        return Collections.unmodifiableSet(new HashSet<String>(Arrays.asList(v)));
    }

 * {@code corpus/ext_check/<CHECK_KEY>/<check_id>_<순번>.<확장자>} 로 보관 ⑤ 처리 뒤(422 포함) 원본 삭제·보관 manifest 재작성은 호출자(InboxService)가 한다.
 *
 * <p>응답·오류에는 파일명·경로·값을 싣지 않는다 — 사유 코드·위치 코드·건수·check_id(형식 검사 통과값)·서버가 만든 키/이름뿐이다.
 */
@Service
public class ExtCheckImporter {

    /** true = 걸린 check 만 제외(기본), false = 하나라도 걸리면 묶음 전체 422(팀이 바꿀 수 있는 스위치). */
    static final boolean EXCLUDE_PER_CHECK = true;

    private static final long MAX_FILE_BYTES = 20L * 1024 * 1024;
    private static final int MAX_ROWS_PER_CHECK = 20;
    private static final ObjectMapper JSON = new ObjectMapper();
    private static final SecureRandom RANDOM = new SecureRandom();

    private final AppProperties properties;
    private final InstitutionMapper institutions;

    public ExtCheckImporter(AppProperties properties, InstitutionMapper institutions) {
        this.properties = properties;
        this.institutions = institutions;
    }

    /** 처리 결과. failure 가 있으면 호출자는 정리(원본 삭제·manifest 재작성·보관) 뒤 그것을 던진다. */
    static final class Result {
        final Map<String, Object> response = new LinkedHashMap<String, Object>();
        byte[] archiveManifest;
        ApiException failure;
    }

    private static final class Row {
        final String reason;
        final String locator;
        final int seq;          // 0 = check 자체, 1.. = 첨부 순번
        int count;

        Row(String reason, String locator, int seq, int count) {
            this.reason = reason;
            this.locator = locator;
            this.seq = seq;
            this.count = count;
        }
    }

    private static final class Staged {
        final Path file;
        final String ext;
        final int seq;

        Staged(Path file, String ext, int seq) {
            this.file = file;
            this.ext = ext;
            this.seq = seq;
        }
    }

    private static final class BatchReject extends RuntimeException {
        private static final long serialVersionUID = 1L;
        final String code;

        BatchReject(String code, String message, String where) {
            super(code + ": " + message + " (" + where + ")", null, false, false);
            this.code = code;
        }

        ApiException toApi() {
            return ApiException.withDetail(422, Collections.singletonMap("errors", Collections.singletonList(getMessage())));
        }
    }

    // ── 처리 ──────────────────────────────────────────────────────────

    /**
     * @throws ApiException 422 LINK_NOT_ALLOWED — 구조가 안전하지 않으면 <b>아무것도 지우지 않고</b> 던진다
     *         (junction 을 따라가 밖의 파일을 지우지 않기 위해서다).
     */
    Result process(Path batchDir, String batchId, JsonNode manifest) {
        Path filesReal = verifyLayout(batchDir, batchId);
        JsonNode checks = manifest.path("checks");
        List<Map<String, Object>> passed = new ArrayList<Map<String, Object>>();
        List<Map<String, Object>> excluded = new ArrayList<Map<String, Object>>();
        List<Path> storedDirs = new ArrayList<Path>();
        ArrayNode archived = JSON.createArrayNode();
        Path tmp = null;
        ApiException failure = null;
        String failureCode = null;
        try {
            tmp = Files.createTempDirectory("extchk-");
            for (int i = 0; i < checks.size(); i++) {
                evalCheck(checks.get(i), i, batchDir, filesReal, tmp, passed, excluded, storedDirs, archived);
            }
        } catch (BatchReject r) {
            failure = r.toApi();
            failureCode = r.code;
        } catch (IOException e) {
            failure = new ApiException(500, "외부 확인 첨부를 처리하지 못했습니다.");
            failureCode = "PROCESSING_FAILED";
        } catch (RuntimeException e) {
            failure = new ApiException(500, "외부 확인 첨부를 처리하지 못했습니다.");
            failureCode = "PROCESSING_FAILED";
        } finally {
            if (tmp != null) {
                deleteTree(tmp);
            }
        }
        if (failure == null && !EXCLUDE_PER_CHECK && !excluded.isEmpty()) {
            Map<String, Object> d = new LinkedHashMap<String, Object>();
            d.put("code", "EXTERNAL_CHECK_REJECTED");
            d.put("excluded", excluded);
            failure = ApiException.withDetail(422, d);
            failureCode = "EXTERNAL_CHECK_REJECTED";
        }
        if (failure != null) {                       // 묶음 거부 — 이미 보관한 것도 되돌린다
            for (Path d : storedDirs) {
                deleteTree(d);
            }
            passed.clear();
            archived.removeAll();
        }
        Result res = new Result();
        res.failure = failure;
        res.archiveManifest = archiveManifest(manifest, archived, failureCode);
        res.response.put("batch_id", batchId);
        res.response.put("kind", "external_checks");
        res.response.put("checks_total", checks.size());
        res.response.put("passed_count", passed.size());
        res.response.put("excluded_count", countDistinctChecks(excluded));
        res.response.put("passed", passed);
        res.response.put("excluded", excluded);
        return res;
    }

    private static int countDistinctChecks(List<Map<String, Object>> excluded) {
        java.util.Set<Object> ids = new java.util.LinkedHashSet<Object>();
        for (Map<String, Object> m : excluded) {
            ids.add(m.get("check_id"));
        }
        return ids.size();
    }

    private void evalCheck(JsonNode check, int index, Path batchDir, Path filesReal, Path tmp,
                           List<Map<String, Object>> passed, List<Map<String, Object>> excluded,
                           List<Path> storedDirs, ArrayNode archived) throws IOException {
        String checkId = check.path("check_id").asText();     // 형식 검사 통과값 [A-Za-z0-9_-]{1,40}
        List<Row> rows = new ArrayList<Row>();

        scanField(rows, check.path("institution").path("name_ko"), "institution");
        scanField(rows, check.path("institution").path("type"), "institution");
        scanField(rows, check.path("notice").path("title"), "notice");
        scanField(rows, check.path("notice").path("number"), "notice");
        scanUrl(rows, check.path("notice").path("url"), "notice");
        scanField(rows, check.path("press").path("title"), "press");
        scanField(rows, check.path("press").path("publisher"), "press");
        scanUrl(rows, check.path("press").path("url"), "press");

        String nameKo = check.path("institution").path("name_ko").asText(null);
        String institutionId = nameKo == null ? null : institutions.selectIdByName(nameKo);
        if (institutionId == null) {
            addRow(rows, "INSTITUTION_NOT_FOUND", "institution", 0, 1);
        }

        List<Staged> staged = new ArrayList<Staged>();
        JsonNode atts = check.path("attachments");
        if (rows.isEmpty() && atts.isArray()) {              // 이미 제외된 check 의 첨부는 열지 않는다
            for (int j = 0; j < atts.size(); j++) {
                JsonNode att = atts.get(j);
                int seq = j + 1;
                String where = "checks[" + index + "].attachments[" + j + "]";
                Path copy = copyToTemp(att, filesReal, tmp, index + "_" + seq, where);
                ExtCheckInspector.Outcome o = ExtCheckInspector.inspect(copy, ExtCheckInspector.extOf(att.path("path").asText()));
                if (o.hits.isEmpty()) {
                    staged.add(new Staged(copy, o.ext, seq));
                } else {
                    for (ExtCheckInspector.Hit h : o.hits) {
                        addRow(rows, h.reason, h.locator, seq, h.count);
                    }
                }
            }
        }

        if (!rows.isEmpty()) {
            int shown = 0;
            ObjectNode ex = archived.addObject();
            ex.put("check_id", checkId);
            ex.put("excluded", true);
            ArrayNode reasons = ex.putArray("reasons");
            for (Row r : rows) {
                reasons.add(r.reason + "@" + r.locator);
                if (shown++ < MAX_ROWS_PER_CHECK) {
                    Map<String, Object> m = new LinkedHashMap<String, Object>();
                    m.put("check_id", checkId);
                    m.put("reason", r.reason);
                    m.put("locator", r.locator);
                    m.put("count", r.count);
                    m.put("file_seq", r.seq);
                    excluded.add(m);
                }
            }
            return;
        }

        String key = randomKey();
        List<Map<String, Object>> files = new ArrayList<Map<String, Object>>();
        if (!staged.isEmpty()) {
            Path dir = Paths.get(properties.getExtCheckRoot()).resolve(key);
            Files.createDirectories(dir);
            storedDirs.add(dir);
            for (Staged s : staged) {
                String stored = checkId + "_" + s.seq + "." + s.ext;     // 서버가 만든 이름 — 원본 파일명은 쓰지 않는다
                Files.copy(s.file, dir.resolve(stored));
                Map<String, Object> f = new LinkedHashMap<String, Object>();
                f.put("seq", s.seq);
                f.put("stored_name", key + "/" + stored);
                f.put("ext", s.ext);
                files.add(f);
            }
        }
        Map<String, Object> p = new LinkedHashMap<String, Object>();
        p.put("check_id", checkId);
        p.put("institution_id", institutionId);
        p.put("check_key", key);
        p.put("files", files);
        passed.add(p);
        archived.add(sanitized(check, key, files));
    }

    // ── 사본 만들기 ───────────────────────────────────────────────────

    private Path copyToTemp(JsonNode att, Path filesReal, Path tmp, String tag, String where) throws IOException {
        String rel = att.path("path").asText();                        // files/<ASCII 이름> — 형식 검사 통과값
        String name = rel.substring(rel.lastIndexOf('/') + 1);
        if (filesReal == null) {
            throw new BatchReject("LINK_NOT_ALLOWED", "첨부 위치가 안전하지 않습니다", where);
        }
        Path src = filesReal.resolve(name);
        Path real;
        BasicFileAttributes a;
        try {
            real = src.toRealPath();
            a = Files.readAttributes(src, BasicFileAttributes.class, LinkOption.NOFOLLOW_LINKS);
        } catch (IOException e) {
            throw new BatchReject("FILE_UNREADABLE", "첨부를 읽을 수 없습니다", where);
        }
        if (!isChild(filesReal, real, name) || !a.isRegularFile()) {
            throw new BatchReject("LINK_NOT_ALLOWED", "첨부 위치가 안전하지 않습니다", where);
        }
        Path dst = tmp.resolve(tag + ".bin");
        MessageDigest md = sha256();
        long n = 0;
        try (InputStream in = Files.newInputStream(src, LinkOption.NOFOLLOW_LINKS);      // 기본 데이터 스트림만 — ADS 는 떨어진다
             OutputStream out = Files.newOutputStream(dst)) {
            byte[] buf = new byte[16384];
            int r;
            while ((r = in.read(buf)) != -1) {
                n += r;
                if (n > MAX_FILE_BYTES) {
                    throw new BatchReject("FILE_TOO_LARGE", "첨부가 상한(20 MiB)을 넘습니다", where);
                }
                md.update(buf, 0, r);
                out.write(buf, 0, r);
            }
        }
        if (!hex(md.digest()).equals(att.path("sha256").asText())) {
            throw new BatchReject("SHA256_MISMATCH", "선언한 sha256 과 파일이 다릅니다", where);
        }
        if (n != att.path("size").asLong(-1L)) {
            throw new BatchReject("SIZE_MISMATCH", "선언한 크기와 파일이 다릅니다", where);
        }
        return dst;
    }

    private Path verifyLayout(Path batchDir, String batchId) {
        try {
            Path inboxReal = Paths.get(properties.getInboxRoot()).toRealPath();
            Path batchReal = batchDir.toRealPath();
            if (!isChild(inboxReal, batchReal, batchId)) {
                throw layout();
            }
            Path files = batchDir.resolve("files");
            if (!Files.exists(files, LinkOption.NOFOLLOW_LINKS)) {
                return null;
            }
            Path filesReal = files.toRealPath();               // junction·심볼릭 링크면 부모가 batchReal 이 아니다
            if (!isChild(batchReal, filesReal, "files") || !Files.isDirectory(filesReal, LinkOption.NOFOLLOW_LINKS)) {
                throw layout();
            }
            return filesReal;
        } catch (IOException e) {
            throw layout();
        }
    }

    private static ApiException layout() {
        return ApiException.withDetail(422, Collections.singletonMap("errors",
                Collections.singletonList("LINK_NOT_ALLOWED: 배치 폴더 구조가 안전하지 않습니다")));
    }

    /** child 의 실제 경로가 parent/name 인가(Windows 라 이름은 대소문자 무시). */
    private static boolean isChild(Path parent, Path child, String name) {
        return child.getParent() != null && parent.equals(child.getParent())
                && child.getFileName() != null && child.getFileName().toString().equalsIgnoreCase(name);
    }

    // ── 글자 검사(manifest 글자) ──────────────────────────────────────

    private static void scanField(List<Row> rows, JsonNode n, String locator) {
        if (n == null || !n.isTextual()) {
            return;
        }
        scanString(rows, n.asText(), locator);
    }

    /** url 은 ?·# 아래를 서버가 자르므로 자른 값만 검사·보관한다. */
    private static void scanUrl(List<Row> rows, JsonNode n, String locator) {
        if (n == null || !n.isTextual()) {
            return;
        }
        scanString(rows, cutUrl(n.asText()), locator);
    }

    private static void scanString(List<Row> rows, String text, String locator) {
        try {
            for (Pii.Finding f : Pii.scanExternal(text)) {
                addRow(rows, ExtCheckInspector.reasonOf(f.getKind()), locator, 0, 1);
            }
        } catch (Pii.ScanTimeoutException e) {
            addRow(rows, "TIMEOUT", locator, 0, 1);
        }
    }

    private static String cutUrl(String u) {
        int q = u.indexOf('?');
        int h = u.indexOf('#');
        int cut = q < 0 ? h : (h < 0 ? q : Math.min(q, h));
        return cut < 0 ? u : u.substring(0, cut);
    }

    private static void addRow(List<Row> rows, String reason, String locator, int seq, int count) {
        for (Row r : rows) {
            if (r.reason.equals(reason) && r.locator.equals(locator) && r.seq == seq) {
                r.count += count;
                return;
            }
        }
        rows.add(new Row(reason, locator, seq, count));
    }

    // ── 보관 manifest(화이트리스트 복사 — 모르는 필드는 버린다) ─────────

    private static ObjectNode sanitized(JsonNode check, String key, List<Map<String, Object>> files) {
        ObjectNode o = JSON.createObjectNode();
        o.put("check_id", check.path("check_id").asText());
        o.put("check_key", key);
        ObjectNode inst = o.putObject("institution");
        for (String k : new String[]{"name_ko", "type", "region_code", "sub_region_code"}) {
            copyText(inst, k, check.path("institution").path(k));
        }
        for (String k : new String[]{"status", "checked_at", "source_kind", "source_slug", "error_code"}) {
            copyText(o, k, check.path(k));
        }
        JsonNode match = check.path("match");
        if (match.isObject()) {
            ObjectNode m = o.putObject("match");
            copyText(m, "confidence", match.path("confidence"));
            ArrayNode basis = m.putArray("basis");
            for (JsonNode b : match.path("basis")) {
                if (b.isTextual()) {
                    basis.add(b.asText());
                }
            }
        }
        JsonNode notice = check.path("notice");
        if (notice.isObject()) {
            ObjectNode n = o.putObject("notice");
            for (String k : new String[]{"title", "number", "posted_at", "deadline_at"}) {
                copyText(n, k, notice.path(k));
            }
            if (notice.path("url").isTextual()) {
                n.put("url", cutUrl(notice.path("url").asText()));
            }
        }
        JsonNode press = check.path("press");
        if (press.isObject()) {
            ObjectNode p = o.putObject("press");
            for (String k : new String[]{"title", "publisher", "published_at"}) {
                copyText(p, k, press.path(k));
            }
            if (press.path("url").isTextual()) {
                p.put("url", cutUrl(press.path("url").asText()));
            }
        }
        ArrayNode st = o.putArray("stored");
        for (Map<String, Object> f : files) {
            ObjectNode s = st.addObject();
            s.put("seq", ((Integer) f.get("seq")).intValue());
            s.put("name", String.valueOf(f.get("stored_name")));
        }
        return o;                                  // attachments·reported_*·sha256·size 는 담지 않는다(자기신고, 원본은 삭제)
    }

    private static void copyText(ObjectNode dst, String key, JsonNode src) {
        if (src != null && src.isTextual()) {
            dst.put(key, src.asText());
        }
    }

    private static byte[] archiveManifest(JsonNode manifest, ArrayNode archived, String failureCode) {
        ObjectNode root = JSON.createObjectNode();
        root.put("schema_version", 1);
        root.put("schema_minor", manifest.path("schema_minor").asInt(1));
        copyText(root, "batch_id", manifest.path("batch_id"));
        copyText(root, "collected_at", manifest.path("collected_at"));
        copyText(root.putObject("source"), "slug", manifest.path("source").path("slug"));
        root.putArray("records");
        root.set("checks", archived);
        if (failureCode != null) {
            root.put("rejected", failureCode);
        }
        return root.toString().getBytes(StandardCharsets.UTF_8);
    }

    // ── 원본 삭제 ─────────────────────────────────────────────────────

    /**
     * 배치 폴더 안에서 manifest.json 만 남기고 모두 지운다. 디렉터리는 실제 경로가 제자리일 때만 안으로 들어가고,
     * junction·링크는 <b>링크 자체만</b> 지운다(대상을 따라가지 않는다). 하나라도 못 지우면 false.
     */
    static boolean scrub(Path batchDir) {
        boolean ok = true;
        try (DirectoryStream<Path> ds = Files.newDirectoryStream(batchDir)) {
            for (Path p : ds) {
                if ("manifest.json".equals(p.getFileName().toString())) {
                    continue;
                }
                ok &= deleteTree(p);
            }
        } catch (IOException | RuntimeException e) {
            return false;
        }
        return ok;
    }

    static boolean deleteTree(Path p) {
        boolean ok = true;
        try {
            BasicFileAttributes a = Files.readAttributes(p, BasicFileAttributes.class, LinkOption.NOFOLLOW_LINKS);
            if (a.isDirectory() && !a.isSymbolicLink() && isRealDir(p)) {
                try (DirectoryStream<Path> ds = Files.newDirectoryStream(p)) {
                    for (Path c : ds) {
                        ok &= deleteTree(c);
                    }
                }
            }
            Files.delete(p);
        } catch (IOException | RuntimeException e) {
            return false;
        }
        return ok;
    }

    private static boolean isRealDir(Path p) {
        try {
            Path real = p.toRealPath();
            Path parentReal = p.toAbsolutePath().getParent().toRealPath();
            return parentReal.equals(real.getParent()) && real.getFileName().toString().equalsIgnoreCase(p.getFileName().toString());
        } catch (IOException | RuntimeException e) {
            return false;
        }
    }

    // ── 작은 도우미 ───────────────────────────────────────────────────

    private static String randomKey() {
        byte[] b = new byte[16];
        RANDOM.nextBytes(b);
        return hex(b);
    }

    private static MessageDigest sha256() {
        try {
            return MessageDigest.getInstance("SHA-256");
        } catch (NoSuchAlgorithmException e) {
            throw new IllegalStateException("SHA-256 을 쓸 수 없습니다");
        }
    }

    private static String hex(byte[] b) {
        StringBuilder sb = new StringBuilder(b.length * 2);
        for (byte x : b) {
            sb.append(Character.forDigit((x >> 4) & 0xF, 16)).append(Character.forDigit(x & 0xF, 16));
        }
        return sb.toString();
    }
}

### 4-6. 먼저 확인해 주십시오(손 추적, 실행 못 함)
1. **컴파일**: pdfbox 메서드 이름이 어긋날 수 있는 줄입니다.
   - `COSDictionary.getNameAsString(String)`, `containsKey(String)`
   - `getCOSName(COSName)`, `COSStream.toTextString()`
   - `PDDocumentInformation.getCOSObject()`, `PDField.getCOSObject()`
   - `PDOutlineNode.children()`, `PDFieldTree` 순회
   - `COSName.PREV`, `COSName.IMAGE`, `COSName.XOBJECT`
2. **junction 시험**: `isRealDir`와 `toRealPath` 부모 비교가 기대대로 동작하는지 확인하는 시험입니다. 이 시험이 Windows에서 건너뛰지 않고 통과하면 가정이 확인된 것입니다.
3. **암호 PDF 시험**: AES-128이 JDK 8 설정에서 되는지 확인합니다. 안 되면 `sp.setPreferAES(false)`를 추가합니다.
4. **이미지만 PDF**: 빈 쪽에 `AppendMode.APPEND`로 그림을 그리는 동작.
5. **정규식**: `SEP_RUN_EXT`로 `PiiExternalTest` 기존 31행이 그대로인지, 특히 `010-1234-5678 / 02-1234-5678`입니다. 이 입력은 접힘 뒤 `010-1234-5678-02-1234-5678`이 되는데 손으로는 휴대폰 1건·유선 1건이 맞습니다.

## 5. 보안·위험·되돌리는 법

**권한(403)·접근 이력·자기신고**
- 권한 경로는 바꾸지 않았습니다. `InboxController.requireSalesTeam`이 그대로 맨 앞에서 `batch_id` 형식·실재 검사보다 먼저 돕니다. 그 외 역할은 403이고 접근 이력 `DENIED`입니다.
- 새 시험이 디자이너 403과 "배치 그대로"를 확인합니다.
- 접근 이력에는 본문·쿼리값·개인정보를 싣지 않는 기존 규칙 그대로입니다. 이번 패치는 이력 코드를 건드리지 않습니다.
- 인증은 여전히 `X-Emp-No` 자기신고(`Actor`)입니다. `reported_*`도 자기신고라 서버는 쓰지 않고 보관본에도 담지 않습니다. "검증됨" 표현은 쓰지 않았습니다.

**이번 패치가 닫는 것**
- 경로 이탈, symlink·junction, 사본 처리(TOCTOU), 응답·오류의 값·파일명 에코, ADS(기본 스트림만 복사).

- 정규식 백트래킹과 마감(`scanLoose` 구간 포함).

**남은 위험·한계**
- **한계 1: 원본 삭제 못 한 경우.** 응답의 `originals_deleted:false`와 서버 로그뿐입니다. 사용자에게 표시할 문구는 프런트·디자인 몫입니다. Windows에서 파일 잠금이 있으면 지우지 못할 수 있습니다.
- **한계 2: 멈춘 스레드.** pdfbox가 멈추면 데몬 스레드 1개가 남습니다. 설계의 "멈춘 스레드가 있으면 새 반입을 503"은 이번에 넣지 않았습니다(HWP 묶음과 함께 넣는 편이 낫습니다).
- **한계 3: 선형 PDF.** 선형화(웹 최적화)된 PDF는 `startxref`가 2개라 `INCREMENTAL`로 제외됩니다. 설계 문안대로이고 과잉 제외입니다. 전처리에서 "선형화 해제"를 요구하는지는 데이터 몫입니다.
- **한계 4: 과잉 제외.** 구분자 접힘이 넓어져서 "1234 / 5678"처럼 서로 다른 숫자가 이어진 글에서 과잉 제외가 날 수 있습니다.
- **한계 5: 글자 인코딩.** PDF 글자가 이미지 위에 겹친 경우나 폰트 매핑이 깨진 PDF는 추출이 틀려 `IMAGE_ONLY` 또는 오탐이 날 수 있습니다.
- **한계 6: 묶음 전체 거부 시 보관본.** 보관본에는 `rejected: <코드>`와 빈 checks만 남습니다. 고친 뒤 다시 낼 때는 **새 batch_id**여야 합니다(SCHEMA ④ 불변 원칙).
- **한계 7: 기존 허용 범위.** `records` 배치의 `InboxService.java:71` 400 문구는 입력 `batch_id`를 되돌려 줍니다(security 11-11 P3, 하). 이번엔 바꾸지 않았습니다. 필요하면 `"batch_id 형식이 잘못됐습니다"`로 줄이는 한 줄입니다.

- **코드**
  - 새 파일 `ExtCheckInspector.java`·`ExtCheckImporter.java`·시험 3개를 지웁니다.
  - `InboxService`의 S-1·S-2·S-4를 되돌리면 1단계 거부(`CHECKS_NOT_YET_SUPPORTED`)가 돌아옵니다.
