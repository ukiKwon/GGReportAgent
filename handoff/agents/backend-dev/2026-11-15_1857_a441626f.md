```js
  const w = (m) => warnings.push(`${r.id} ${m}`);
  for (const k of ['orgCode', 'g2bCode']) {
    if (r[k] !== null && !CODE20_RE.test(r[k])) { w(`${k} 형식 불량 '${r[k]}' → NULL`); r[k] = null; }
  }
  if (r.orgCode !== null && !/^\d{7}$/.test(r.orgCode)) w(`orgCode 가 7자리 숫자가 아님 '${r.orgCode}'`);
  if (r.opStatus === null) r.opCode = 'OPERATING';
  else if (OP_CODE[r.opStatus]) r.opCode = OP_CODE[r.opStatus];
  else throw new Error(`opStatus 알 수 없는 값: ${r.id} '${r.opStatus}'`);
  if (r.openDate !== null && !validOpenDate(r.openDate)) { w(`openDate 형식 불량 '${r.openDate}' → NULL`); r.openDate = null; }
  if (r.opCode === 'PLANNED' && r.openDate === null) w('개원 예정인데 openDate 가 없음');

  r.ceUrl = null; r.ceNo = null;
  if (r.confirmed === true) {
    if (r.contractEnd === null) w('confirmed 인데 contractEnd 가 없음 → 추정으로 둠');

    }
  }
```

**`service/InstitutionRules.java`** (신규, 순수 함수 — 시험하기 쉽게)
```java
package com.kbstar.kgi.ggreport.web.service;

import com.kbstar.kgi.ggreport.web.domain.Institution;
import java.time.LocalDate;
import java.time.YearMonth;
import java.time.ZoneId;

/** 기관 계산 규칙. 서울 날짜 기준. */
public final class InstitutionRules {
    public static final ZoneId SEOUL = ZoneId.of("Asia/Seoul");
    private InstitutionRules() { }

    /** 개원 예정(PLANNED)인데 개원일이 지났다 — 상태는 바꾸지 않고 '확인 필요'만 낸다.
     *  연·연월 정밀도는 그 기간의 마지막 날을 기준으로 한다(아직 개원 못 한 달을 오탐하지 않으려고). */
    public static boolean openCheckNeeded(String opStatus, String openDate, LocalDate today) {
        if (!"PLANNED".equals(opStatus) || openDate == null) return false;
        try {
            LocalDate last;
            switch (openDate.length()) {
                case 4:  last = LocalDate.of(Integer.parseInt(openDate), 12, 31); break;
                case 7:  last = YearMonth.parse(openDate).atEndOfMonth(); break;
                case 10: last = LocalDate.parse(openDate); break;
                default: return false;
            }
            return today.isAfter(last);
        } catch (RuntimeException e) {
            return false;
        }
    }

    public static Institution decorate(Institution i, LocalDate today) {
        if (i != null) i.setOpenCheckNeeded(openCheckNeeded(i.getOpStatus(), i.getOpenDate(), today));
        return i;
    }
}
```

```java
package com.kbstar.kgi.ggreport.web.service;

import com.kbstar.kgi.ggreport.web.domain.Institution;
import com.kbstar.kgi.ggreport.web.domain.InstitutionUpdateIn;
import com.kbstar.kgi.ggreport.web.web.ApiException;

import java.net.URI;
import java.net.URISyntaxException;
import java.time.LocalDate;
import java.time.YearMonth;
import java.util.ArrayList;
import java.util.List;
import java.util.Objects;
import java.util.regex.Pattern;

/** PUT /institutions/{id} 의 입력 검사와 '확정/추정·운영 상태' 규칙. 오류 문구는 고정 문구뿐(내부 정보 없음). */
final class InstitutionEditRules {

    static final String OP_GROUP = "INST_OP_STATUS";
    static final String OP_PLANNED = "PLANNED";
    private static final Pattern DATE = Pattern.compile("\\d{4}-\\d{2}-\\d{2}");
    private static final Pattern OPEN = Pattern.compile("\\d{4}(-\\d{2}(-\\d{2})?)?");
    private static final Pattern NO = Pattern.compile("[\\p{L}\\p{N}][\\p{L}\\p{N} _./()\\-]{0,49}");
    private static final String URL_MSG = "출처 링크는 http 또는 https 주소(500자 이내)여야 합니다.";

    interface CodeCheck { boolean enabled(String group, String code); }

    static final class Change {
        final String field, oldVal, newVal, srcUrl, srcNo;
        Change(String field, String oldVal, String newVal, String srcUrl, String srcNo) {
            this.field = field; this.oldVal = oldVal; this.newVal = newVal; this.srcUrl = srcUrl; this.srcNo = srcNo;
        }
    }

    private InstitutionEditRules() { }

    /** 검증하고, 결정된 값을 upd 에 써 넣어(묶음 칸은 한 번에 저장) 바뀐 칸 목록을 돌려준다. 위반이면 400. */
    static List<Change> resolve(Institution cur, InstitutionUpdateIn upd, CodeCheck codes) {
        List<Change> out = new ArrayList<>();
        plain(out, "REGION_CODE", cur.getRegionCode(), upd.present("regionCode"), upd.getRegionCode());
        plain(out, "TYPE", cur.getType(), upd.present("type"), upd.getType());
        plain(out, "LAST_BID", cur.getLastBid(), upd.present("lastBid"), upd.getLastBid());
        plain(out, "TERM", str(cur.getTerm()), upd.present("term"), str(upd.getTerm()));
        contractEnd(out, cur, upd);
        opStatus(out, cur, upd, codes);
        return out;
    }

    private static void plain(List<Change> out, String f, String oldV, boolean sent, String newV) {
        if (sent && !Objects.equals(oldV, newV)) out.add(new Change(f, cut(oldV), cut(newV), null, null));
    }

    private static void contractEnd(List<Change> out, Institution cur, InstitutionUpdateIn upd) {
        boolean dateSent = upd.present("contractEnd"), confSent = upd.present("contractEndConfirmed");
        boolean urlSent = upd.present("contractEndSrcUrl"), noSent = upd.present("contractEndSrcNo");
        if (!dateSent && !confSent && !urlSent && !noSent) return;

        String oldDate = cur.getContractEnd();
        String newDate = oldDate;
        if (dateSent) newDate = upd.getContractEnd() == null ? null : date(upd.getContractEnd());
        boolean dateChanged = !Objects.equals(oldDate, newDate);

        Boolean confIn = upd.getContractEndConfirmed();
        if (confSent && confIn == null) throw bad("확정 여부는 true 또는 false 여야 합니다.");
        boolean oldConf = cur.isContractEndConfirmed();
        boolean newConf;
        if (newDate == null) {
            if (confSent && confIn) throw bad("입찰 예정일이 없으면 확정할 수 없습니다.");
            newConf = false;
        } else if (dateChanged) {
            if (!confSent) throw bad("입찰 예정일을 바꿀 때는 확정/추정(contract_end_confirmed)을 함께 보내야 합니다.");
            newConf = confIn;
        } else {
            newConf = confSent ? confIn : oldConf;
        }
        boolean confChanged = newConf != oldConf;
        if (!dateChanged && !confChanged && !urlSent && !noSent) return;   // 바뀐 것 없음 — 출처 없는 옛 확정 행을 건드리지 않는다

        String url = null, no = null;
        if (newConf) {
            boolean keep = !dateChanged && oldConf;   // 같은 날짜의 확정이 이어질 때만 옛 출처를 이어 쓴다
            url = urlSent ? url(upd.getContractEndSrcUrl()) : (keep ? cur.getContractEndSrcUrl() : null);
            no = noSent ? notice(upd.getContractEndSrcNo()) : (keep ? cur.getContractEndSrcNo() : null);
            if (url == null && no == null) throw bad("확정하려면 출처(공고 링크 또는 번호)가 필요합니다.");
        }
        boolean srcChanged = newConf && oldConf
                && (!Objects.equals(url, cur.getContractEndSrcUrl()) || !Objects.equals(no, cur.getContractEndSrcNo()));
        upd.setContractEnd(newDate);
        upd.setContractEndConfirmed(newConf);
        upd.setContractEndSrcUrl(url);
        upd.setContractEndSrcNo(no);
        if (dateChanged) out.add(new Change("CONTRACT_END", cut(oldDate), cut(newDate), null, null));
        if (confChanged || srcChanged) out.add(new Change("CONTRACT_END_CONFIRMED", yn(oldConf), yn(newConf), url, no));

    private static void opStatus(List<Change> out, Institution cur, InstitutionUpdateIn upd, CodeCheck codes) {
        boolean stSent = upd.present("opStatus"), odSent = upd.present("openDate");
        boolean urlSent = upd.present("opSrcUrl"), noSent = upd.present("opSrcNo");
        if (!stSent && !odSent && !urlSent && !noSent) return;

        String oldSt = cur.getOpStatus() == null ? "OPERATING" : cur.getOpStatus();
        String newSt = oldSt;
        if (stSent) {
            newSt = upd.getOpStatus();
            if (newSt == null || !codes.enabled(OP_GROUP, newSt)) throw bad("허용되지 않는 운영 상태입니다.");
        }
        String oldOd = cur.getOpenDate();
        String newOd = oldOd;
        if (odSent) newOd = upd.getOpenDate() == null ? null : openDate(upd.getOpenDate());
        if (OP_PLANNED.equals(newSt) && newOd == null) {
            throw bad("개원 예정에는 개원 예정일(연, 연-월 또는 연-월-일)이 필요합니다.");
        }
        boolean stChanged = !oldSt.equals(newSt), odChanged = !Objects.equals(oldOd, newOd);

        String url = cur.getOpSrcUrl(), no = cur.getOpSrcNo();
        if (stChanged) {
            url = urlSent ? url(upd.getOpSrcUrl()) : null;
            no = noSent ? notice(upd.getOpSrcNo()) : null;
            if (url == null && no == null) throw bad("운영 상태를 바꾸려면 출처(링크 또는 번호)가 필요합니다.");
        } else {
            if (urlSent) url = url(upd.getOpSrcUrl());
            if (noSent) no = notice(upd.getOpSrcNo());
        }
        upd.setOpStatus(newSt);
        upd.setOpenDate(newOd);
        upd.setOpSrcUrl(url);
        upd.setOpSrcNo(no);
        if (stChanged) out.add(new Change("OP_STATUS", oldSt, newSt, url, no));
        if (odChanged) out.add(new Change("OPEN_DATE", cut(oldOd), cut(newOd), null, null));
    }

    static String date(String v) {
        String s = v.trim();
        if (!DATE.matcher(s).matches()) throw bad("날짜는 YYYY-MM-DD 형식이어야 합니다.");
        try {
            LocalDate d = LocalDate.parse(s);
            if (d.getYear() < 2000 || d.getYear() > 2100) throw bad("날짜 범위(2000~2100)를 벗어났습니다.");
        } catch (java.time.DateTimeException e) {
            throw bad("존재하지 않는 날짜입니다.");
        }
        return s;
    }

    static String openDate(String v) {
        String s = v.trim();
        if (!OPEN.matcher(s).matches()) throw bad("개원 예정일은 YYYY, YYYY-MM 또는 YYYY-MM-DD 형식이어야 합니다.");
        try {
            int y = Integer.parseInt(s.substring(0, 4));
            if (y < 2000 || y > 2100) throw bad("개원 예정일 범위(2000~2100)를 벗어났습니다.");
            if (s.length() == 7) YearMonth.parse(s);
            if (s.length() == 10) LocalDate.parse(s);
        } catch (java.time.DateTimeException e) {
            throw bad("존재하지 않는 날짜입니다.");
        }
        return s;
    }

    static String url(String raw) {
        if (raw == null) return null;
        String v = raw.trim();
        if (v.isEmpty()) return null;
        if (v.length() > 500) throw bad(URL_MSG);
        for (int i = 0; i < v.length(); i++) {
            char c = v.charAt(i);
            if (c <= ' ' || c == 0x7f) throw bad(URL_MSG);
        }
        URI u;
        try { u = new URI(v); } catch (URISyntaxException e) { throw bad(URL_MSG); }
        String sc = u.getScheme();
        if (sc == null || !(sc.equalsIgnoreCase("http") || sc.equalsIgnoreCase("https"))
                || u.getHost() == null || u.getUserInfo() != null) {
            throw bad(URL_MSG);
        }
        return v;
    }

    static String notice(String raw) {
        if (raw == null) return null;
        String v = raw.trim();
        if (v.isEmpty()) return null;
        if (!NO.matcher(v).matches()) throw bad("공고 번호 형식이 올바르지 않습니다(50자 이내, 문자·숫자·공백·-_./()).");
        return v;
    }

    private static ApiException bad(String m) { return ApiException.badRequest(m); }
    private static String yn(boolean b) { return b ? "Y" : "N"; }
    private static String str(Integer i) { return i == null ? null : String.valueOf(i); }
    private static String cut(String s) { return s == null ? null : (s.length() > 60 ? s.substring(0, 60) : s); }
```

**권한 검사·403 경로**
- PUT은 기존 `requireInstitutionEdit`를 그대로 씁니다. 401(사번 없음)·403(권한 없음)은 이미 `DENIED`로 접근 이력에 남고, 본문·쿼리값은 안 남습니다.
- 400(검증 실패)은 접근 이력의 상태코드로만 남습니다. 새 이력 표에는 성공한 변경만 들어갑니다.
- 자기신고 한계: 사번·화면·버튼은 `X-Emp-No`·`X-Screen-Id`·`X-Button-Id`입니다. 위조할 수 있고, `EMP_VERIFIED=0`이면 "검증됨"으로 표시하지 않습니다. 앞단 인증이 붙기 전에는 이 이력이 위조 방지가 아니라 사후 추적용입니다.

**변경 이력**
- 어느 칸을 누가·어느 화면·어느 버튼으로·언제 바꿨는지, 옛 값·새 값과 출처를 `INST_EDIT_LOG`에 남깁니다.
- 값은 날짜·코드·http(s) 링크·공고 번호뿐입니다. 개인정보와 비밀값은 없습니다.
- 칸 갱신과 이력은 같은 트랜잭션입니다. 이력 없이 값만 바뀌는 길이 없습니다.
- 접근 이력과 달리 기관 편집 이력은 본문 값 일부(날짜·코드)를 담습니다. 기관 기본 정보라 개인정보가 아닙니다. security가 이 구분을 확인해야 합니다.

**입력·출처**
- 링크는 http·https만 받습니다. 사용자정보(`user:pw@`)는 거부하고, 500자와 제어문자·공백을 막습니다. 서버는 링크를 열지 않습니다.
- 링크는 응답에 그대로 나갑니다. 화면이 `href`로 쓸 때 스킴을 다시 확인하고 `rel="noopener noreferrer"`를 달아야 합니다(이관 메모).
- 오류 문구는 고정 문구입니다. 스택·SQL·표 이름은 싣지 않습니다.

**위험**
- **골든 00·01 변경**: Python 스냅샷과 갈라집니다. Python 서버(`server/`)는 안 건드립니다. 골든을 파이썬 쪽 시험이 읽는지는 확인하지 못했습니다.
- **MySQL SET 평가 순서**: `updateFromImport`의 확정·출처 CASE는 반드시 `CONTRACT_END` 대입보다 앞이어야 합니다. 031의 `CONTRACT_END_CONFIRMED`는 마지막이어야 합니다. 시험 ⑤가 031 쪽을 고정합니다. 반입 쪽은 API 시험(CSV)이 잡습니다.
- **031 재실행**: 사람이 내린 '추정'을 되살릴 수 있어 한 번만 돕니다.
- **경합**: PUT은 마지막 쓰기가 이깁니다(기존과 같음). 같은 기관 동시 저장은 이력이 둘 다 남지만 값은 나중 것입니다. 낙관적 잠금은 이번 범위 밖입니다.
- `contract_end` 형식(YYYY-MM-DD)은 값이 바뀔 때만 검사합니다. 기존 값은 다시 검증하지 않습니다.
- **출처 없는 옛 확정 행**: 031로 들어온 확정 행 중 출처가 없는 것은 날짜·확정 표시를 바꾸지 않는 한 그대로 둡니다.

**되돌리기**
- 코드는 되돌릴 수 있습니다. 칸은 NULL·기본값이라 코드를 되돌려도 기존 동작에는 영향이 없습니다.
- DB는 030 헤더에 되돌리기 SQL이 있습니다(데이터 소멸, 운영 금지). 이미 적용한 파일은 고치지 않고 새 번호로 갑니다.
- 거부 규칙을 푸는 것은 `contract_end` 분기 한 곳을 고치면 됩니다.

### 6. 이관 메모

- **[이관: 프런트]**
  - 합치기 키는 `id > org_code·g2b_code > 지역+정규화 이름` 순으로 하고, 번들에 생성기 `makeId`와 같은 규칙으로 id를 넣습니다. 기다리는 것은 030 응답 칸입니다. 근거는 planner 1844 §5·§8, qa P2입니다.
  - 번들 `confirmed`·`opStatus`·`openDate` 얹기를 끊습니다. 서버의 `contract_end_confirmed`·`op_status`·`open_date`가 이깁니다. `op_status`는 코드(`OPERATING`·`PLANNED`)로 오므로 화면에 `운영`·`개원 예정` 두 줄짜리 표가 필요합니다. 개원 예정 제외는 서버 행 기준으로 합니다.
  - 지도 색·임박 목록에서는 `op_status=PLANNED`를 빼고, 그 밖의 목록에는 '개원 예정' 표시를 답니다. `open_check_needed`가 참이면 '개원 확인 필요' 표시를 합니다.
  - `PUT /institutions/{id}` 규칙:
    - 입찰 예정일을 바꿀 때는 `contract_end_confirmed`를 반드시 같이 보냅니다.
    - 확정은 `contract_end_src_url` 또는 `contract_end_src_no` 중 하나 이상이 필수입니다.
    - `op_status`를 바꾸려면 `op_src_url` 또는 `op_src_no`가 필수입니다.
    - `PLANNED`에는 `open_date`(연·연월·연월일)가 필수입니다.
    - 위반하면 400이고 `detail`에 한글 고정 문구가 옵니다.
    - 출처 없이 확정이던 옛 행을 그대로 보내면(날짜·확정 변화 없음) 거부하지 않습니다.
  - 요청 헤더에 `X-Screen-Id`(`institution_edit`·`map_popover`·`bid_board`)와 `X-Button-Id`(`save_basic`·`save_schedule`·`save_open_status`·`open_edit`)를 보냅니다. 이름이 다르면 백엔드 허용 목록(`InstUiIds`)도 같이 바꿉니다.
  - 링크를 `href`에 넣기 전에 http(s)인지 다시 확인하고 `rel="noopener noreferrer"`를 답니다.
- **[이관: 디자인]**
  - '개원 확인 필요' 표시, 확정/추정 배지와 출처 입력칸, 운영 상태를 바꾸는 화면. 기다리는 것은 위 규칙 확정입니다. 근거는 질문 1·2의 사용자 답입니다.
