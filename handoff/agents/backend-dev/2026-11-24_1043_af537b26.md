```java
// 상수 추가
/** 상자에 꽉 채울 때 잘라내도 되는 최대 비율(한 축). 넘으면 비율대로 상자 안에 넣는다(contain). */
static final double MAX_CROP = 0.35;
/** 세로 사진은 위 여유(머리 위)를 덜 자른다: 잘라낼 양의 30% 를 위, 70% 를 아래에서. */
static final double TOP_SHARE = 0.30;

/** 사진(w×h px)을 상자(boxW×boxH mm)에 꽉 채울 때 자를 비율 {l,t,r,b}(0~1). 자를 게 없거나 MAX_CROP 초과면 null. */
static double[] cropFor(double boxW, double boxH, int w, int h) {
    double boxR = boxW / boxH;
    double r = (double) w / h;
    if (Math.abs(r - boxR) < 1e-6) {
        return null;
    }
    if (r < boxR) {                       // 세로로 긴 사진: 폭을 맞추고 위아래를 자른다
        double f = 1 - r / boxR;
        return f > MAX_CROP ? null : new double[]{0, f * TOP_SHARE, 0, f * (1 - TOP_SHARE)};
    }
    double f = 1 - boxR / r;              // 가로로 긴 사진: 높이를 맞추고 좌우를 가운데 기준으로 자른다
    return f > MAX_CROP ? null : new double[]{f / 2, 0, f / 2, 0};
}
```

## 5. 보안·위험·되돌리기
- 권한·입력 경로 무변경. 사진은 이미 서버가 재인코딩·크기 검증한 JPEG; srcRect 값은 0~1 검증, 오류 문구는 기존 "docx build failed" 유지.
- 위험: ①얼굴 잘림 — 비율 3:4 이상 증명사진은 약 21%만 잘림, 위 30% 치우침으로 완화. 35% 초과는 contain. ②POI srcRect API 차이(위 대안). ③Word 외 뷰어(구버전 뷰어)에서 srcRect 무시 시 사진이 찌그러져 보일 수 있음 — 안전을 우선하면 대안 ImageIO 픽셀 자르기(`planPhoto`에서 디코딩 후 자른 JPEG로 교체, 화질 재압축·메모리는 업로드 한도 이내라 허용 범위, 단 원본 되돌리기 불가). Word 실물 확인 뒤 결정 권장.
- 되돌리기: `cropFor`가 항상 null을 돌려주게 하면(MAX_CROP=0) 현행 contain과 동일.
