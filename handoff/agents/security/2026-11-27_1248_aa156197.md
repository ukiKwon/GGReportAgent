   ```sql
   -- 활성 담당인데 결재자가 자동(AUTO)이거나 비어 있는 영업팀원
   SELECT a.BID_CASE_ID, a.EMP_NO FROM BID_ASSIGNMENT a JOIN USERS u ON u.EMP_NO=a.EMP_NO
    WHERE a.ACTIVE_YN=1 AND u.ROLE='영업팀' AND (u.APPROVER_MANUAL<>1 OR u.APPROVER_EMP_NO IS NULL);
   -- 담당자들의 결재자가 둘 이상으로 갈리는 혼합 건(팀장 담당은 본인 사번으로 셈)
   SELECT a.BID_CASE_ID, COUNT(DISTINCT CASE WHEN u.ROLE='영업팀장' THEN u.EMP_NO ELSE u.APPROVER_EMP_NO END) t
     FROM BID_ASSIGNMENT a JOIN USERS u ON u.EMP_NO=a.EMP_NO WHERE a.ACTIVE_YN=1 GROUP BY a.BID_CASE_ID HAVING t>1;
   ```
