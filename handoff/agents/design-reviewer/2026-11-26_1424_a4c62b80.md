## 4. 실측 스크립트 (메인이 browser_evaluate로 그대로 실행)
**준비**
1. 영업 계정으로 로그인한다. 랜딩이 뜨면 [로그인]을 한 번 더 누른다.
2. 입찰 현황판 목표 표를 연다.
3. 먼저 `innerWidth`를 확인한다. 배율 90%이므로 창 1152×810 → innerWidth 1280, 창 1296×810 → 1440이어야 한다.
4. 아래 스크립트는 읽기만 한다.
```js
() => {
  const px = el => el ? Math.round(el.getBoundingClientRect().width) : null;
  const cut = el => el ? el.scrollWidth > el.clientWidth : null;
  const tt = document.querySelector('.hb-tt'), ts = document.querySelector('.hb-ts');
  const heads = [...(tt?.querySelectorAll('thead th') || [])];
  const row = tt?.querySelector('tbody tr:not(.grp)');
  const grp = tt?.querySelector('tr.grp th');
  const stg = [...(tt?.querySelectorAll('.hb-stg.two') || [])];
  const sb = [...document.querySelectorAll('.hb-sb')];
  const ft = document.querySelector('.hb-stgs .ft'), fin = document.querySelector('.hb-fin');
  const note = ft?.firstElementChild;
  let grpOff = null;
  if (grp) { const r = grp.getBoundingClientRect(), rg = document.createRange(); rg.selectNodeContents(grp); const t = rg.getBoundingClientRect();
    grpOff = { h: Math.round(r.height), top: Math.round(t.top - r.top), bottom: Math.round(r.bottom - t.bottom) }; }
  const lh = note ? parseFloat(getComputedStyle(note).lineHeight) || 18 : 18;
  return {
    innerWidth, dpr: devicePixelRatio,
    cols: heads.map(h => ({ t: h.textContent.trim(), w: px(h) })),
    sortBtns: heads.map(h => h.querySelector('.hb-sort')).filter(Boolean).map(b => ({ t: b.textContent.trim(), w: px(b), h: Math.round(b.getBoundingClientRect().height), thInner: b.parentElement.clientWidth - parseFloat(getComputedStyle(b.parentElement).paddingLeft) - parseFloat(getComputedStyle(b.parentElement).paddingRight) })),
    hOverflow: ts ? { sw: ts.scrollWidth, cw: ts.clientWidth, over: ts.scrollWidth > ts.clientWidth } : null,
    rowH: row ? Math.round(row.getBoundingClientRect().height) : null,
    nameAlign: row ? getComputedStyle(row.children[1]).textAlign : null,
    stageCells: { n: stg.length, w: px(stg[0]?.closest('td')), lbCut: stg.filter(s => cut(s.querySelector('.lb'))).length, ln2Cut: stg.filter(s => cut(s.querySelector('.ln2'))).length },
    band: sb.map(b => ({ t: b.textContent.replace(/\s+/g,' ').trim(), w: px(b), cut: [...b.querySelectorAll('*')].some(cut) || cut(b), unk: b.classList.contains('unk'), color: getComputedStyle(b).color, border: getComputedStyle(b).borderStyle })),
    note: note ? { lines: Math.round(note.getBoundingClientRect().height / lh), w: px(note) } : null,
    fin: fin && ft ? { right: Math.round(ft.getBoundingClientRect().right - fin.getBoundingClientRect().right), overlap: note ? note.getBoundingClientRect().right > fin.getBoundingClientRect().left : null, w: px(fin) } : null,
    grp: grpOff
  };
}
```
