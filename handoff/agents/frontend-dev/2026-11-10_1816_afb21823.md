## 2. 요구 요약과 근거
- 내 담당 조회는 새 엔드포인트가 필요 없다. `team.js`가 이미 쓰는 `GET /me/assignments`가 있고, 응답은 `[{bid_case_id, assigned_at}]`이며 `institution_id`는 없다.
- 그래서 `선택 기관 -> store 레코드(currentRecord).bidCaseId -> 내 목록의 bid_case_id` 순서로 맞춘다.
- 서버 규칙(`InstitutionController.run`)은 role이 영업팀장이거나 기관 입찰 건의 활성 담당자일 때만 허용한다. 화면의 영업팀장 판정은 `gateAccess`와 같은 `profile.team`을 쓰고, `team.js`의 `tm.LEAD`('영업팀장')와 같은 값이다.
- 처리 방식은 버튼을 숨기고(`display:none`) 이유 한 줄을 새 `#wf-run-hint`에 보여주는 것이다. 새 id를 하나 더하는 것이라 기존 테스트에는 영향이 없다.
- 403은 기존 `act()`가 `failText`로 서버 한글 메시지를 그대로 `textContent`에 넣는다. 그래서 이스케이프는 이미 된다.

### 3-1. `frontend/js/workflow.js` (`runnable` 정의 바로 뒤에 추가)
old:

new:

```
    return !status.running && !status.pending_gate && Number(status.stage) < 9;

  // 실행 권한 — 영업팀장 또는 그 기관 입찰 건의 활성 담당자(서버 /run이 아니면 403, NEXT 70).
  // mineIds: GET /me/assignments의 bid_case_id 배열. null이면 아직 못 받았거나 조회 실패.
  workflow.RUN_LEAD = '영업팀장';
  workflow.mineBidCaseIds = function (list) {
    if (!Array.isArray(list)) return null;
    return list.map(function (x) { return x && x.bid_case_id; }).filter(Boolean);
  };
  workflow.runAccess = function (profile, session, record, mineIds, mineFailed) {
    const p = profile || {}, s = session || {};
    if (s.mode !== 'account' || !/^[0-9]{7}$/.test(s.emp || '')) {
      return { allowed: false, reason: '분석 실행은 로그인한 영업팀장 또는 이 기관 입찰 건의 담당자만 할 수 있습니다.' };
    }
    if ((p.team || '').trim() === workflow.RUN_LEAD) return { allowed: true, reason: '' };
    if (mineFailed) return { allowed: false, reason: '담당 정보를 확인하지 못해 실행 버튼을 숨겼습니다. 화면을 새로 고쳐 주세요.' };
    if (!Array.isArray(mineIds)) return { allowed: false, reason: '' };   // 조회 중 — 깜빡임 방지로 말은 하지 않는다
    if (record && record.bidCaseId && mineIds.indexOf(record.bidCaseId) >= 0) return { allowed: true, reason: '' };
    return { allowed: false, reason: '분석 실행은 영업팀장 또는 이 기관 입찰 건의 담당자만 할 수 있습니다.' };
  };
```

## 4. 테스트
적용 순서는 테스트 추가 후 구현이다. 깨지는 기존 테스트는 없다. 순수 함수만 추가했고 `syncButtons`는 DOM이 필요해서 단위 시험이 없다. 아래를 `frontend/test/workflow.test.js` 맨 끝에 추가한다.

```js
const LEAD_P = { team: '영업팀장' }, MEM_P = { team: '영업팀' };
const REC = { institutionId: 'i1', bidCaseId: 'BC-1' };
test('runAccess: 영업팀장은 담당 조회 없이도 보인다', function () {
  assert.strictEqual(wf.runAccess(LEAD_P, OK_S, REC, null, true).allowed, true);
  assert.strictEqual(wf.runAccess(LEAD_P, OK_S, null, [], false).allowed, true);
});
test('runAccess: 그 기관 입찰 건의 담당자는 보인다', function () {
  assert.strictEqual(wf.runAccess(MEM_P, OK_S, REC, ['BC-9', 'BC-1'], false).allowed, true);
});
test('runAccess: 담당이 아닌 팀원은 숨기고 합니다체 이유를 준다', function () {
  const r = wf.runAccess(MEM_P, OK_S, REC, ['BC-9'], false);
  assert.strictEqual(r.allowed, false);
  assert.strictEqual(r.reason, '분석 실행은 영업팀장 또는 이 기관 입찰 건의 담당자만 할 수 있습니다.');
  assert.strictEqual(wf.runAccess({ team: '영업부장' }, OK_S, REC, ['BC-1'], false).allowed, true);   // 부장도 담당이면 서버가 허용
  assert.strictEqual(wf.runAccess(MEM_P, OK_S, null, ['BC-1'], false).allowed, false);                  // 기관 미선택
});
test('runAccess: 조회 실패·조회 전·둘러보기는 숨긴다', function () {
  const f = wf.runAccess(MEM_P, OK_S, REC, null, true);
  assert.strictEqual(f.allowed, false);
  assert.ok(f.reason.indexOf('확인하지 못') >= 0);
  assert.deepStrictEqual(wf.runAccess(MEM_P, OK_S, REC, null, false), { allowed: false, reason: '' });
  assert.strictEqual(wf.runAccess(LEAD_P, { mode: 'guest', emp: '' }, REC, [], false).allowed, false);
});
test('mineBidCaseIds: 배열이 아니면 null(실패 취급), 항목에서 bid_case_id만 뽑는다', function () {
  assert.deepStrictEqual(wf.mineBidCaseIds([{ bid_case_id: 'A', assigned_at: 1 }, null, {}]), ['A']);
  assert.strictEqual(wf.mineBidCaseIds({ detail: 'x' }), null);
  assert.strictEqual(wf.mineBidCaseIds(undefined), null);
});
test('run 403: 서버 한글 메시지를 그대로 쓴다(화면은 textContent라 이스케이프됨)', function () {
  const m = '분석 실행은 영업팀장 또는 이 기관 입찰 건의 담당자만 할 수 있습니다.';
  assert.strictEqual(wf.failText(403, { detail: m }), m);
  assert.strictEqual(wf.failText(403, { message: m }), m);
});
```

## 5. 위험·되돌리는 법
- 위험: `/me/assignments`에 institution_id가 없어서 store 레코드의 `bidCaseId`에 의존한다. 입찰 건이 없는 기관의 담당자는 버튼이 숨고, 서버 규칙(`isActiveAssigneeOfInstitution`)과 기관 단위로 조금 다를 수 있다. 서버가 더 넓게 허용하면 화면이 더 좁은 쪽이다.
- 서버가 이 화면에서 `/me/assignments`로 401을 주면 실패 처리(버튼 숨김)가 된다.
- 되돌리기: 3-1~3-6을 되돌린다. 모두 workflow.js 안의 추가분이다.

## 6. 가정
- 영업팀장 판정은 `profile.team === '영업팀장'`이다(gateAccess·team.js와 같은 기준).
- 숨김은 `style.display`로 한다. `#wf-run`에 CSS가 `display`를 지정했을 가능성 때문에 `hidden` 속성은 쓰지 않았다.
- 403 본문이 `detail`/`message`/`error` 중 하나를 쓴다고 가정한다. 이는 `failText`가 이미 가정하는 키다.
- `nav.js`·사이드바·`index.html`·board 계열 파일은 건드리지 않았다.
