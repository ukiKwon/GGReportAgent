### 2. app.js를 노드에서 불러올 수 있는가
**가능합니다. 다만 module.exports가 없어서 loadFresh의 반환값은 못 쓰고, 가짜 self에 붙은 selfObj.app을 읽어야 합니다.**
- app.js는 `(function(root){…})(typeof self !== 'undefined' ? self : this)`로 감싸져 있고 `root.app = app`으로 노출합니다(1061줄). module.exports는 없습니다.
- 불러올 때 최상위에서 도는 외부 의존은 `document.addEventListener('DOMContentLoaded', app.init)` 한 줄입니다(1059줄). 그래서 `global.document = { addEventListener(){} }` 스텁이 필요합니다. 157줄의 keydown 등록은 함수 안쪽이라 불러올 때는 실행되지 않습니다.
- installIdentityHeader(931~946줄)는 root.fetch, root.store.loadSession(), root.login.withIdentity를 씁니다. 가짜 self에 fetch·store·login을 넣어야 합니다.
- login.js는 노드에서 require할 수 있습니다(127줄 module.exports). roles.js는 스스로 require합니다(10줄).
- **함정이 하나 있습니다.** prep.js의 req는 전역 fetch를 직접 부르고 root.fetch를 쓰지 않습니다(842줄). 브라우저에서는 self가 window라 같은 객체이지만, 노드의 가짜 self에서는 다릅니다. 그래서 설치 뒤 `global.fetch = fakeSelf.fetch`로 감싼 함수를 연결해야 prep.postConfirm이 래퍼를 지나갑니다.
- login.withIdentity 순수 함수는 이미 노출돼 있습니다. 하지만 그 함수만 보면 래퍼가 실제 요청에 적용되는지는 보장되지 않아서, installIdentityHeader를 통째로 설치하는 방식을 택했습니다.

### 3. 구현안 — 시험 코드
**붙일 위치:** frontend/test/map-detail.test.js 맨 끝(291줄 `});` 다음). loadFresh(207줄)를 재사용하므로 그 뒤여야 합니다.

```js
// ── security 11-15 하: 사번(X-Emp-No)이 화면 머리글과 함께 실제 요청에 실린다 ──
// app.js는 module.exports가 없어 loadFresh 반환값이 아니라 가짜 self의 app을 쓴다. 최상위 외부 의존은 document.addEventListener 한 줄뿐.
test('installIdentityHeader 설치 후 prep.postConfirm 요청: X-Emp-No가 X-Screen-Id·X-Button-Id와 함께 실린다', async function () {
  const sent = [];
  const rawFetch = function (url, init) {
    sent.push({ url: url, init: init });
    return Promise.resolve({ ok: true, status: 201, json: function () { return Promise.resolve({ bid_case_id: 'B1' }); } });
  };
  let session = { mode: 'account', id: 'x', emp: '3918264' };
  const fakeSelf = {
    fetch: rawFetch,
    store: { loadSession: function () { return session; } },
    login: require('../js/login.js')
  };
  const prevDoc = global.document, prevFetch = global.fetch;
  global.document = { addEventListener: function () {} };
  try {
    loadFresh('../js/app.js', fakeSelf);
    assert.strictEqual(typeof fakeSelf.app.installIdentityHeader, 'function');
    fakeSelf.app.installIdentityHeader();
    assert.strictEqual(fakeSelf.fetch.__identity, true, '전역 fetch가 신원 래퍼로 바뀌었다');
    global.fetch = fakeSelf.fetch;            // 브라우저에서는 self === window라 같은 객체 — prep.js는 전역 fetch를 직접 부른다

    const prep = require('../js/prep.js');
    await prep.postConfirm('I1', 'bid_board');
    const h = sent[0].init.headers;
    assert.strictEqual(sent[0].url, '/preps');
    assert.strictEqual(h['X-Emp-No'], '3918264');
    assert.strictEqual(h['X-Screen-Id'], 'bid_board');
    assert.strictEqual(h['X-Button-Id'], 'confirm');

    await prep.postConfirm('I2', 'map_popover');
    assert.strictEqual(sent[1].init.headers['X-Emp-No'], '3918264');
    assert.strictEqual(sent[1].init.headers['X-Screen-Id'], 'map_popover');

    session = null;                            // 로그아웃 상태면 사번을 붙이지 않되 화면 머리글은 그대로
    await prep.postConfirm('I3', 'bid_board');
    assert.strictEqual(sent[2].init.headers['X-Emp-No'], undefined);
    assert.strictEqual(sent[2].init.headers['X-Screen-Id'], 'bid_board');
  } finally {
    if (prevDoc === undefined) delete global.document; else global.document = prevDoc;
    global.fetch = prevFetch;
  }
});
```

### 4. 테스트
- 깨질 기존 테스트는 없습니다. 추가만 하고, 전역 document·fetch는 finally에서 복원합니다. 시험 수는 +1건입니다.
- 실행 명령은 `node --test "frontend/test/map-detail.test.js"`이고, 전체는 `node --test "frontend/test/*.test.js"`입니다.
- 이 시험이 사번을 실제로 고정하는지 보려면 변이 확인을 권합니다. app.js:940의 withIdentity 줄을 잠시 주석 처리하면 시험이 실패해야 합니다. app.js는 다른 세션이 작업 중이므로 메인이 판단해 주세요.

### 5. 위험·되돌리는 법
- 위험은 가정 3·4가 틀릴 때입니다. 그러면 loadFresh 단계에서 ReferenceError가 나고, 에러 메시지에 누락된 전역 이름이 나옵니다. 그 이름을 fakeSelf 또는 global에 추가하면 됩니다.
- 되돌리는 법은 시험 블록 하나를 삭제하는 것입니다. 소스는 건드리지 않습니다.

1. map-detail.test.js 상단에 assert가 선언돼 있고 loadFresh는 207줄에 있습니다.
2. login.isEmpNo('3918264')가 true입니다. 기존 시험(250줄)에서도 쓰는 7자리 값입니다. isEmpNo의 정확한 형식은 직접 읽지 않았습니다.
