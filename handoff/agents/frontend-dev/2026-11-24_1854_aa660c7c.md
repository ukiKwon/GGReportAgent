## 2. 요구 요약과 설계 결정
- **분류 계산은 `logic.js` 순수 함수 8개로 뺐습니다(DOM 없음).**

| 함수 | 하는 일 |
|---|---|
| `logic.tipBuckets(info)` | 겹치지 않는 6구간을 돌려줍니다. 0건은 뺍니다. |
| `logic.recordBucket(rec, today)` | 기관 하나의 구간을 정합니다. 기준은 `regionMix([r])`이라 면 색·카드와 같습니다. |
| `logic.groupSummary(info)` | 묶음 머리 요약을 만듭니다. |
| `logic.faceGroups(faces)` | 지자체 레코드를 공유하는 면을 union-find로 한 묶음에 모읍니다. |
| `logic.rankGroups(ranked, groups, today)` | 임박순·관심도순 목록을 묶음으로 나눕니다. |
| `logic.restLabel(n, open)` | 펼침 단추 글을 만듭니다. |
| `logic.groupScrollDelta(top, viewH)` | 묶음 맞춤 스크롤 양을 계산합니다. |
| `logic.groupStatusText(g)` | sr-only 알림 글을 만듭니다. |

- **묶음 순서와 소속:**
  - 묶음 순서는 현재 정렬로 만든 한 줄 목록에서 각 묶음이 처음 나오는 순서입니다.
  - 소속은 `recordsForFeature`와 같은 규칙(`render.rankGroupDefs`)이라 머리 수와 카드 수의 근거가 같습니다.
  - 어느 면에도 없는 기관은 "시·군·구 미확인" 묶음에 넣고, 이 묶음 머리는 단추가 아닙니다.
- **선택 상태:**
  - `render._selectedName`(기관)과 `render._selectedGroup`(묶음 코드) 중 하나만 설정됩니다.
  - 해제는 `clearSelection` 한 곳에서 합니다.
  - `selectGroup(code, {noScroll, quiet})`이 면 `.selected`, 링, 카드 선택, 목록 `.on`, 펼침, 스크롤, 알림을 처리합니다.
- **`onSubregionClick`:** `selectGroup`으로 위임합니다. 묶음을 못 찾으면 예전 경로(`selectInstitution` + `focusCard`)로 떨어지므로 기존 `render.test.js`의 onSubregionClick 테스트 2건은 수정 없이 통과합니다. 카드 클릭도 `onSubregionClick(d, ev)`를 그대로 쓰므로 `map-subcard.test.js:184`도 통과합니다.
- **펼침 상태:**
  - `render.state.rankOpen`(Set)에 둡니다.
  - 전국에서 새로 들어올 때(`state.currentRegion !== code`)만 비웁니다. `drawNational`은 건드리지 않습니다.
  - 필터·창 폭 때문에 같은 지역을 다시 그릴 때와 ♥ 때는 유지합니다.
- **숨긴 행 처리:** `highlightCard(name, true)`가 접힌 행이면 그 묶음을 먼저 펼칩니다. 그래서 `selectInstitution`, 티커, 검색 모두 보장됩니다.

```js
      { key: 'to12', label: '1년 미만', n: info.to12 || 0 },
      { key: 'to24', label: '2년 미만', n: info.to24 || 0 },
      { key: 'over24', label: '2년 이상', n: info.over24 || 0 },
      { key: 'unknown', label: '알 수 없음', n: info.unknown || 0 },
    ];
    return rows.filter(function (r) { return r.n > 0; });
  };
  // 기관 하나가 어느 구간인가 — regionMix([r])와 같은 판정이라 면 색·카드·툴팁과 어긋나지 않는다.
  // 'notice'(공고 중) | 'overdue'(경과) | within1 | within6 | to12 | to24 | over24 | unknown
  logic.recordBucket = function (r, today) {
    const i = logic.regionMix([r], today);
    if (i.notice) return 'notice';
    if (i.overdue) return 'overdue';
    if (i.unknown) return 'unknown';
    if (i.within1) return 'within1';
    if (i.within6) return 'within6';
    if (i.to12) return 'to12';
    if (i.to24) return 'to24';
    return i.over24 ? 'over24' : 'unknown';
  };
  // 목록에서 기본으로 접는 구간 = 1년 이상 남은 기관(팀 결정 11-24).
  logic.RANK_REST = { to24: true, over24: true };
  // 묶음 머리 요약(0건 줄 숨김). 2년 미만·2년 이상은 접힌 행이라 요약에 쓰지 않는다.
  logic.groupSummary = function (info) {
    if (!info) return [];
    const rows = [
      { key: 'notice', label: '공고 중', n: info.notice || 0 },
      { key: 'overdue', label: '경과', n: info.overdue || 0 },
      { key: 'within1', label: '1개월 미만', n: info.within1 || 0 },
      { key: 'within6', label: '6개월 미만', n: Math.max(0, (info.within6 || 0) - (info.within1 || 0)) },
      { key: 'to12', label: '1년 미만', n: info.to12 || 0 },
      { key: 'unknown', label: '알 수 없음', n: info.unknown || 0 },
    ];
    return rows.filter(function (r) { return r.n > 0; });
  };
  // 면 → 묶음. faces = [{code, name, muni:[{key(instKey), name}]}]. 같은 지자체 레코드를 공유하는 면(수원 4구 등)은 한 묶음으로 합친다
  // (union-find, 대표 = 앞쪽 면). 합쳐진 묶음 이름 = 공유한 지자체 이름에서 '청'을 뗀 것('수원시청' → '수원시'), 면 하나면 그 면 이름.
  // 반환: [{code(대표 면 코드), name, faceCodes:[...]}] — 면 순서대로.
  logic.faceGroups = function (faces) {
    const list = faces || [], parent = list.map(function (f, i) { return i; });
    function find(i) { while (parent[i] !== i) { parent[i] = parent[parent[i]]; i = parent[i]; } return i; }
    const owner = {}, shared = [];
    list.forEach(function (f, i) {
      (f.muni || []).forEach(function (m) {
        if (owner[m.key] === undefined) { owner[m.key] = i; return; }
        if (owner[m.key] === i) return;
        const a = find(owner[m.key]), b = find(i);
        if (a !== b) parent[Math.max(a, b)] = Math.min(a, b);
        shared.push(m);
      });
    });
    const rootName = {};
    shared.forEach(function (m) {
      const root = find(owner[m.key]);
      if (!rootName[root]) rootName[root] = logic.normalizeMuniName(m.name);
    });
    const out = [], at = {};
    list.forEach(function (f, i) {
      const root = find(i);
      if (!at[root]) { at[root] = { code: list[root].code, name: rootName[root] || list[root].name, faceCodes: [] }; out.push(at[root]); }
      at[root].faceCodes.push(f.code);
    });
    return out;
  };
  // 오른쪽 목록을 시군구 묶음으로 나눈다(DOM 없음). ranked = 정렬(임박순·관심도순)·기관 단위 중복 제거를 마친 목록,
  // groups = faceGroups 결과에 keys(소속 instKey 배열)를 붙인 것. 묶음은 목록에서 처음 나온 순서, 묶음 안 순서는 목록 순서.
  // 어느 묶음에도 없으면 {code:'', name:'시·군·구 미확인'}. 행이 0인 묶음은 만들지 않는다. 한 기관이 여러 묶음에 걸리면 앞 묶음이 갖는다.
  // 반환 항목: {code, name, faceCodes, recs, total, info, summary, shown(기본 표시), rest(1년 이상 — 접힘)}.
  logic.rankGroups = function (ranked, groups, today) {
    const where = {};
    (groups || []).forEach(function (g, gi) {
      (g.keys || []).forEach(function (k) { if (where[k] === undefined) where[k] = gi; });
    });
    const out = [], made = {};
    (ranked || []).forEach(function (r) {
      const gi = where[logic.instKey(r)];
      const id = gi === undefined ? -1 : gi;
      let g = made[id];
      if (!g) {
        const src = id < 0 ? { code: '', name: '시·군·구 미확인', faceCodes: [] } : groups[id];
        g = made[id] = { code: src.code, name: src.name, faceCodes: src.faceCodes || [], recs: [] };
        out.push(g);
      }
      g.recs.push(r);
    });
    out.forEach(function (g) {
      g.total = g.recs.length;
      g.info = logic.regionMix(g.recs, today);
      g.summary = logic.groupSummary(g.info);
      g.shown = []; g.rest = [];
      g.recs.forEach(function (r) { (logic.RANK_REST[logic.recordBucket(r, today)] ? g.rest : g.shown).push(r); });
    });
    return out;
  };
  logic.restLabel = function (n, open) { return '1년 이상 남은 ' + n + '건 ' + (open ? '접기' : '보기'); };
  // 묶음 머리를 맞출 스크롤 양. top = 묶음 위쪽 − 고정 머리 아래 기준선(px), viewH = 보이는 높이.
  // 이미 보이는 영역의 위쪽 1/3 안이면 0(보던 목록을 튀기지 않는다), 아니면 머리를 맨 위에 맞추는 양.
  logic.groupScrollDelta = function (top, viewH) {
    return (top >= 0 && top <= viewH / 3) ? 0 : top;
  };
  // 묶음 선택 뒤 스크린리더(role=status) 알림.
  logic.groupStatusText = function (g) {
    const near = (g.shown || []).length, far = (g.rest || []).length, parts = [];
    if (near) parts.push('입찰 가까운 ' + near + '건');
    if (far) parts.push('1년 이상 ' + far + '건');
    return g.name + ' 묶음을 목록에서 표시했습니다.' + (parts.length ? ' ' + parts.join(', ') + '.' : '');
  };

  // 마커로 그릴 기관: 지자체(면)는 뺀다. 나머지는 종류(kindOf) 6종 중 켠 것만, 종류를 모르는 유형(가로 막대)은 항상 표시.
```

## 5. 위험과 되돌리는 법
- **전국 툴팁 문안이 곳에서 건으로 바뀝니다.** 규격 8절이 이미 되돌리기 쉽다고 명시한 변경입니다. 사용자가 "곳"을 원하면 카드 aria·툴팁·묶음 머리를 함께 바꿔야 합니다.
- **실제 화면 미검증:**
  - sticky 머리 두 겹의 높이 계산(`--rank-head-h`, `rank-head` 줄바꿈 시 어긋날 수 있음)
  - 툴팁 줄바꿈
  - 수원 같은 일반구 시의 합침
  
  머지 전에 qa-verifier의 실사용 확인(Gyeonggi 수원·성남, 서울 강남)이 필요합니다.
- **성능:** `rankGroupDefs`가 면마다 `recordsForFeature`를 한 번 더 부릅니다. `drawRegion`이 이미 면마다 부르는 함수라 호출 횟수가 2배가 됩니다. 경기(약 45면)에서 체감이 느리면 `drawRegion`의 `subMix` 계산 때 쓴 레코드를 재사용하도록 합칩니다. `recordBucket`은 기관마다 `regionMix([r])`를 불러 수백 건이면 충분히 가볍습니다.
- **일반구 합침 판정:** "같은 지자체 레코드를 공유"가 기준입니다. 일반구 시의 지자체 레코드가 면마다 다른 키(예: 구청별 레코드)로 들어 있으면 합쳐지지 않고 면별 묶음이 됩니다. 이름은 `normalizeMuniName`으로만 정규화합니다.
- **`noticeWhen` 단순화(R4):** 툴팁 전용이라 영향은 한 곳입니다. 마감일 미확인의 공고일 날짜와 "결과 공고를 확인하세요" 꼬리가 툴팁에서 빠집니다. 규격 예시("공고 후 41일")와 맞춘 것이고, 되돌리려면 R4 한 블록만 되돌리면 됩니다.
- **전국 aria의 "6개월 안":** 전국 aria는 `▲n`과 같은 말이라 "6개월 안"을 유지했습니다. 전국 범례 문구(`notice.test.js:255`)도 그대로 고정돼 있습니다. P2-1은 단위(건)만 통일한 것입니다.
- **범례 'p' 항목:** `'섞임 (급한 곳·여유 있는 곳 반반)'`의 "곳"은 규격이 손대라고 하지 않아 그대로 뒀습니다.
- **되돌리기:** 위 패치는 `render.js`, `logic.js`, `index.html`, `app.js` 4개 파일과 테스트 파일에만 있습니다. `git checkout -- <파일>`로 파일별로 되돌립니다. 새 테스트 파일은 삭제하면 됩니다. 되돌리면 map-subcard 등 수정한 기존 테스트도 함께 되돌려야 합니다.

- **[이관: 디자인]** 툴팁 꼬리 조사를 받침에 맞췄습니다("경과는", "결과 대기는", "알 수 없음은"). 규격 예시는 "는" 하나였습니다. 확인이 필요합니다.
- **[이관: 디자인]** 규격 CSS의 `.rk-gbtn:hover { background:#eef1f5 }`는 토큰만 쓰는 규칙에 맞게 `color-mix(in srgb, var(--fg) 6%, var(--panel2))`로 바꿨습니다. 선택된 묶음 위 hover는 틴트를 덮지 않도록 `.rk-grp.on .rk-gbtn:hover { background:transparent }`를 더했습니다. 확인이 필요합니다.
- **[이관: 기획]** 기본 숨김 경계를 "1년 이상 남은 기관"(to24·over24)으로 한 것은 규격 7절의 확인 요청이 그대로 남아 있습니다. 기다리는 것: 사용자가 말한 "입찰이 다가오면"의 기준이 1년인지. 근거: 카드 분류의 최댓값이 "1년 미만"입니다.
