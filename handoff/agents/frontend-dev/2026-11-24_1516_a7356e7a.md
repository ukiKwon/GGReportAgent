```js
  // ── 이벤트(두 host 공용) ──
  function onClick(e) {

    if (!t) return;

    else if (act === 'refresh') { if (!P.refreshing) { setErr(''); load('refresh'); } }

    else if (act === 'scope-team') setScope('team');

    else if (act === 'prep') openPrep();
    else if (act === 'board') openNav(root.nav && root.nav.BOARD_ID);
    else if (act === 'workflow') openNav('workflow');
  }

    if (!b) return;

  }
  function onInput(e) {

  }
  function wireHost(h) {
    if (h.getAttribute && h.getAttribute('data-ts-wired') === '1') return;
    h.addEventListener('click', onClick);
    h.addEventListener('keydown', onKey);
    h.addEventListener('input', onInput);
    h.setAttribute('data-ts-wired', '1');
  }

  // 경로 2의 자리(#ts-root). index.html에 마크업이 없으면 #tab-soon 안에 만든다.
  pt.host = function () {
    let h = el('ts-root');
    if (!h) {
      h = document.createElement('div');
      h.id = 'ts-root';
      h.style.display = 'none';
      (el('tab-soon') || document.body).appendChild(h);

  pt.mount = function (host, cd, label) {

    const pp = el('pp-root'); if (pp && pp.classList) pp.classList.remove('is-steps');

    const h = el('ts-list-title'); if (h && h.focus) h.focus();
    if (!P.step) { P.state = 'err'; paintList(); return; }
    load('open');
  };
  // 경로 1: prepboard.js가 기관 이름 클릭·Enter에서 부른다. ctx: { row, data, hooks, back(bid) }. 표는 CSS(#pp-root.is-steps)로 가린다 — prep.js showView가 hidden을 다시 쓰기 때문.
  pt.openFromBoard = function (ctx) {
    if (typeof document === 'undefined' || !ctx || !ctx.row) return false;
    const pp = el('pp-root');
    if (!pp) return false;
    let dt = el('ts-detail-pp');
    if (!dt) {
      dt = document.createElement('div');
      dt.id = 'ts-detail-pp';
      dt.className = 'ts-dhost';
      pp.insertBefore(dt, el('pp-modal') || null);
    }
    const th = el('ts-root'); if (th) th.innerHTML = '';   // 같은 id가 둘이 되지 않게 경로 2의 틀을 비운다
    P = fresh();
    P.from = 'board'; P.host = dt; P.row = ctx.row; P.bid = str(ctx.row.bid_case_id); P.marks = ctx.data && Array.isArray(ctx.data.marks) ? ctx.data.marks : [];
    P.ctx = ctx; P.mode = 'steps'; P.idx = 0; P.rov = 0;
    wireHost(dt);
    pp.classList.add('is-steps');
    renderDetail(dt, pb.T.back);
    return true;
  };
  // 계정이 바뀌면 앞 계정의 목록·상세를 버린다(app.onProfileChanged). 경로 2 화면이 열려 있었으면 새로 받는다.
  pt.onProfileChange = function () {
    if (typeof document === 'undefined') return;
    const wasTask = P.from === 'task' && P.host && P.host.style && P.host.style.display !== 'none' ? { host: P.host, cd: P.cd, label: P.label } : null;
    P = fresh();
    const d = el('ts-detail-pp'); if (d) d.innerHTML = '';
    const pp = el('pp-root'); if (pp && pp.classList) pp.classList.remove('is-steps');
    const h = el('ts-root'); if (h) h.innerHTML = '';
    if (wasTask) pt.mount(wasTask.host, wasTask.cd, wasTask.label);
  };

  if (typeof module !== 'undefined' && module.exports) module.exports = pt;
  else root.preptask = pt;
})(typeof self !== 'undefined' ? self : this);
```

설계 메모:
- 같은 id(`ts-title`·`ts-steps`·`ts-live`)가 동시에 둘이 되지 않게 한 번에 한 host만 내용을 갖습니다. `mount`는 경로 1 틀을 비우고, `openFromBoard`는 경로 2 틀을 비웁니다.
- 경로 1의 돌아가기는 `ctx.back(bid)` → prepboard `pb.show(hooks,{reload:true, focusBid})`입니다. prepboard가 hooks를 보관하므로 prep.js의 `boardHooks`를 내보낼 필요가 없습니다.
- 경로 2에서 목록으로 돌아올 때는 다시 불러오지 않고 받아 둔 목록을 그대로 씁니다. 읽기 전용이라 값이 바뀔 일이 없습니다.
- `[사전준비 상세 열기]`는 경로 1에서 `hooks.openCase`(prep.js enterCase), 경로 2에서 `prep.openCase`를 부릅니다. prep.js는 건드리지 않습니다.
- `P.row`는 열 때 잡은 행 객체입니다. 상세를 보는 동안 목록은 새로 받지 않습니다.

## 5. 위험·되돌리는 법

- **표 가림 방식**: `#pp-root.is-steps #pp-board{display:none}`(CSS)과 prep.js의 `hidden`이 따로 놉니다. 상세에서 `[사전준비 상세 열기]`를 누르면 `openCase`를 먼저 부르고 그다음 `is-steps`를 지웁니다. 순서를 바꾸면 `leave()`가 스크롤을 0으로 덮습니다. 이 때문에 `openPrep`의 순서와 `leave()`의 가드를 같이 넣었습니다.
- **알려진 한계(규격 4절 P1과 같음)**: 경로 1에서 `[사전준비 상세 열기]`로 간 뒤 `[← 제안 준비 현황으로]`를 누르면 단계 상세가 아니라 표로 돌아갑니다. 경로 2에서 열면 `prep.openCase`가 사전준비 화면(사이드바 '제안 준비 현황')으로 이동합니다.
- **감사 기록 부담**: `/prep-board` GET에 `X-Button-Id: open`을 붙이면 서버가 열람 감사 행을 남깁니다(PrepBoardApiTest.java:180). task 메뉴를 열 때마다 한 행씩 늘어납니다. 서버 변경은 없지만 감사 표를 보는 쪽에는 알려야 합니다(이관 메모).
- **되돌리기**: 7개 파일의 패치를 거꾸로 적용하면 됩니다. preptask.js·preptask.test.js는 삭제합니다. index.html의 `<script>` 한 줄만 빼도 앱은 기존 동작으로 돌아갑니다. `root.preptask`가 없으면 `openRow`는 기존 `hooks.openCase`로 가고 `isTask`는 false가 됩니다. 이때 nav에는 task 항목이 남아 '준비중' 화면이 열리고, nav·app 패치는 되돌려야 합니다.
- **기록 필요**: 규격 6-6의 전/후 장(97장 후보)은 적용 뒤 메인이 `docs/design_history/index.html`에 추가해야 합니다. 이것은 패치에 포함하지 않았습니다. 이번 변경은 화면(마크업·CSS·문구·메뉴)이 바뀌므로 PUSH-GATE에서 design-reviewer가 규격 대조를 해야 합니다.
