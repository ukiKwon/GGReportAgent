## 2. 요구 요약과 코드 근거
- **화면 판정**:
  - 서버 `canViewCase`는 `isHead || (영업 계열 && 활성 담당)`이고, 상태는 `PrepService.TASK_STATES`(409)다.
  - 화면의 같은 판정은 이미 있는 `view.viewer.is_assignee`·`view.viewer.is_head`(PrepService:1362–1363)와 `view.state_cd ∈ prep.TASK_STATES`(prep.js:28)로 한다. 새 서버 필드는 필요 없다.
  - 그래도 판정이 어긋나면 서버가 403·409로 막고, 화면은 5-3 문구로 알린다.
  - 상태 조회(`GET /llm/script-draft/status`)는 위 판정이 참일 때만, 창을 열기 전에 한다. `enabled !== true`이거나 실패하면 단추를 그리지 않는다.
- **요청 헤더**: `ScriptDraftUiIds`는 `X-Screen-Id: prep_detail`, `X-Button-Id: script_ai_draft`일 때만 값을 인정한다. 다르면 UNKNOWN으로 기록할 뿐 거절하지는 않는다. 기존 대본 Word 요청과 같은 방식(`'X-Screen-Id': 'prep_detail', 'X-Button-Id': '…'`)으로 맞췄다. 본문은 없고 `Content-Type`도 보내지 않는다. 사번 헤더는 `app.installIdentityHeader`가 같은 출처 `/…` 요청에 자동으로 붙인다(app.js:1103).
- **응답 해석**: BUSY·COOLDOWN은 HTTP 200에 `state:"EMPTY"`, `reason:"BUSY|COOLDOWN"`으로 온다.
  - 요청 전체 실패: `reason`이 BUSY·COOLDOWN·TIMEOUT·UNREACHABLE·MODEL_MISSING·LLM_ERROR·BAD_OUTPUT이거나 응답 모양이 이상하다.
  - 전부 빔: `reason`이 ALL_BLANKED 또는 NO_BASIS이고 칸 글이 없다.
  - 일부 또는 전부 도착: 칸 글이 하나라도 있다.
  - 이 구분을 `pt.aiOutcome`(순수 함수)이 한다.

```js
    node.className = l && l.err ? 'dlg-err' : 'dlg-hint';   // 둘은 패딩·선이 같아 높이가 흔들리지 않는다
    if (hadFocus) { const nb = node.querySelector('button') || el('bs-ai-go'); if (nb) nb.focus(); }
  }
  function aiRenderHead(sc) {
    const pt = root.preptask, ai = sc.ai;
    if (!ai || !ai.on) return;
    const go = el('bs-ai-go'), help = el('bs-ai-h'), bs = el('bs-ai-basis');
    if (go) {
      go.textContent = pt.aiGoText(ai.busy, ai.hadResult);
      if (ai.busy) go.setAttribute('aria-disabled', 'true'); else go.removeAttribute('aria-disabled');   // Tab은 머문다. 중복 요청은 aiGo가 막는다
    }
    if (help) help.textContent = ai.hadResult ? pt.T.aiHelpAgain : pt.T.aiHelp;
    if (bs) { bs.innerHTML = pt.aiBasisHtml(ai.basis); bs.hidden = !ai.basis.length; }
    aiRenderLine(sc);
  }
  function aiRenderSlot(sc, key) {
    const pt = root.preptask, dl = root.dlg, f = aiFieldOf(key), node = f && el(f.id + '-ai-s'), ta = f && el(f.id);
    if (!node || !ta) return;
    const mark = sc.aiMarks[key] || null;
    node.innerHTML = pt.aiSlotHtml(f, { box: (sc.ai && sc.ai.boxes[key]) || null, mark: mark, append: ta.value.trim() !== '' });
    const aim = f.id + '-aim', cur = ta.getAttribute('aria-describedby');   // 칸에 들어가면 'AI 초안을 넣은 칸' 문장도 읽힌다
    const next = mark ? dl.mergeDescribedBy(cur, aim) : dl.removeDescribedBy(cur, aim);
    if (next) ta.setAttribute('aria-describedby', next); else ta.removeAttribute('aria-describedby');
  }
  function aiRenderSlots(sc) { root.preptask.AI_FIELDS.forEach(function (f) { aiRenderSlot(sc, f.key); }); }
  function aiEnd(ai) {   // 타이머·마감을 풀고 이 실행의 번호를 무효로 한다(늦게 온 응답·옛 타이머가 새 실행을 건드리지 않게)
    if (ai.timer) clearInterval(ai.timer);
    if (ai.cut) clearTimeout(ai.cut);
    ai.timer = 0; ai.cut = 0; ai.busy = false; ai.ctl = null; ai.run += 1;
  }
  function aiClose(sc) {   // 창을 닫을 때(확인 뒤 '버리고 닫기' 포함): 요청을 끊고 상태를 버린다. 칸의 글·'넣은 칸' 표시는 남기되 되돌리기는 거둔다
    const ai = sc.ai;
    if (!ai) return;
    const ctl = ai.ctl;
    ai.dead = true; aiEnd(ai);
    if (ctl) ctl.abort();
    Object.keys(sc.aiMarks).forEach(function (k) { sc.aiMarks[k].undo = null; sc.aiMarks[k].prevMark = null; });
    sc.ai = null;
  }
  function aiGo(sc) {
    const pt = root.preptask, ai = sc.ai;
    if (!ai || !ai.on || ai.dead || ai.busy || S.script !== sc) return;   // aria-disabled 단추는 눌러도 아무것도 하지 않는다
    const run = ai.run + 1;
    ai.run = run; ai.busy = true; ai.timedOut = false; ai.t0 = Date.now(); ai.srMin = 0;
    ai.line = pt.aiLine({ kind: 'busy', sec: 0 });
    aiRenderHead(sc);   // 포커스는 단추에 그대로 둔다
    aiSay(sc, pt.T.aiSrStart);
    ai.timer = setInterval(function () { aiTick(sc, ai, run); }, 1000);
    const ctl = typeof AbortController === 'function' ? new AbortController() : null;
    ai.ctl = ctl;
    ai.cut = setTimeout(function () {   // 9분 — 서버 마감(480초)보다 길게
      if (ai.run !== run) return;
      ai.timedOut = true;
      if (ctl) ctl.abort(); else aiDone(sc, ai, run, { ok: false, status: 0 });
    }, pt.AI_FETCH_MS);
    const init = { method: 'POST', headers: { 'X-Screen-Id': 'prep_detail', 'X-Button-Id': 'script_ai_draft' } };   // ScriptDraftUiIds가 기대하는 값(자기신고)
    if (ctl) init.signal = ctl.signal;
    fetch(base(sc.bid) + '/board-script-draft', init)
      .then(function (r) {
        return r.json().catch(function () { return {}; }).then(function (j) { return r.ok ? { ok: true, status: r.status, res: j } : { ok: false, status: r.status }; });
      })
      .catch(function () { return { ok: false, status: 0 }; })
      .then(function (out) { aiDone(sc, ai, run, out); });
  }
  function aiTick(sc, ai, run) {
    const pt = root.preptask;
    if (ai.dead || ai.run !== run || !ai.busy) return;
    const sec = Math.round((Date.now() - ai.t0) / 1000);
    const next = pt.aiLine({ kind: sec >= pt.AI_LONG_SEC ? 'long' : 'busy', sec: sec });
    const same = !!ai.line && ai.line.a === next.a && ai.line.b === next.b;
    ai.line = next;
    if (!(same && !next.t)) aiRenderLine(sc, same);   // 8분 뒤 줄은 글이 안 바뀌므로 다시 그리지 않는다
    const min = Math.floor(sec / 60);
    if (min >= 2 && min % 2 === 0 && min > ai.srMin) { ai.srMin = min; aiSay(sc, pt.T.aiSrTick(min)); }
  }
  function aiDone(sc, ai, run, out) {
    const pt = root.preptask;
    if (ai.dead || ai.run !== run) return;   // 닫았거나 그만뒀거나 새로 시작했다 — 늦게 온 응답은 버린다
    const sec = Math.round((Date.now() - ai.t0) / 1000), timedOut = ai.timedOut;
    aiEnd(ai);
    const oc = out && out.ok ? pt.aiOutcome(out.res) : 'fail';
    if (oc === 'fail') {   // 옛 초안·옛 근거는 그대로 둔다(실패해도 잃는 것이 없게)
      const status = out && out.ok ? 200 : (out && out.status) || 0;
      const reason = out && out.ok ? str(out.res && out.res.reason) : (status ? '' : (timedOut ? 'TIMEOUT' : 'UNREACHABLE'));
      ai.line = pt.aiLine({ kind: 'fail', msg: pt.aiFailText(status, reason) });
    } else {
      ai.hadResult = true;
      ai.boxes = pt.aiBoxes(out.res);   // 새 결과가 와야 옛 초안을 바꾼다
      ai.basis = pt.aiBasisLines(out.res);
      const blanks = pt.AI_FIELDS.filter(function (f) { return !ai.boxes[f.key].text; }).map(function (f) { return f.label; });
      ai.line = oc === 'blank' ? pt.aiLine({ kind: 'allblank' }) : pt.aiLine({ kind: 'arrived', sec: sec, blanks: blanks });
      aiRenderSlots(sc);
    }
    aiRenderHead(sc);
    aiSay(sc, pt.aiLinePlain(ai.line));
  }
  function aiStop(sc) {
    const pt = root.preptask, ai = sc.ai;
    if (!ai || !ai.busy) return;
    const ctl = ai.ctl;
    aiEnd(ai);
    ai.line = pt.aiLine({ kind: 'stopped' });
    if (ctl) ctl.abort();
    aiRenderHead(sc);
    aiSay(sc, pt.aiLinePlain(ai.line));
    const go = el('bs-ai-go');
    if (go) go.focus();   // [그만두기] → 묶음 단추
  }
  function aiShow(sc) {   // [초안 보기] → 처음 온 초안 상자의 제목으로(스스로 도착해서는 포커스를 옮기지 않는다)
    const pt = root.preptask, ai = sc.ai;
    const list = pt.AI_FIELDS.filter(function (f) { return ai.boxes[f.key] && ai.boxes[f.key].text; });
    const f = list[0] || pt.AI_FIELDS.filter(function (x) { return ai.boxes[x.key]; })[0];
    const h = f ? el(f.id + '-ai-h') : null;
    if (!h) return;
    h.focus();
    if (h.scrollIntoView) h.scrollIntoView({ block: 'nearest' });
  }
  function aiToEnd(ta) { ta.focus(); if (ta.setSelectionRange) ta.setSelectionRange(ta.value.length, ta.value.length); }
  function aiPut(sc, key) {
    const pt = root.preptask, dl = root.dlg, f = aiFieldOf(key), box = f && sc.ai.boxes[key], ta = f && el(f.id);
    if (!f || !box || !box.text || !ta) return;
    const m = pt.aiMerge(ta.value, box.text);
    if (m.err) { const e = el(f.id + '-ai-e'); if (e) e.textContent = m.err; return; }   // 넣지 않는다. 단추는 막지 않는다
    sc.aiMarks[key] = { after: m.value, undo: ta.value, prevMark: sc.aiMarks[key] || null };
    ta.value = m.value;
    delete sc.ai.boxes[key];
    const n = el(f.id + '-n');
    if (n) n.textContent = pt.scriptCountText(ta.value.length);
    dl.clearInvalid(ta);
    aiRenderSlot(sc, key);
    aiToEnd(ta);
  }
  function aiDrop(sc, key) {   // [버리기]·[알림 닫기]
    const f = aiFieldOf(key), ta = f && el(f.id);
    if (!f || !ta) return;
    delete sc.ai.boxes[key];
    aiRenderSlot(sc, key);
    ta.focus();
  }
  function aiUndo(sc, key) {
    const pt = root.preptask, f = aiFieldOf(key), ta = f && el(f.id), m = sc.aiMarks[key];
    if (!f || !ta || !m || m.undo == null) return;
    ta.value = m.undo;
    if (m.prevMark) sc.aiMarks[key] = m.prevMark; else delete sc.aiMarks[key];   // 되돌리면 AI 칸 표시도 걷는다(이전 넣기가 있었으면 그 표시로)
    const n = el(f.id + '-n');
    if (n) n.textContent = pt.scriptCountText(ta.value.length);
    aiRenderSlot(sc, key);
    aiToEnd(ta);
  }
  function aiAct(sc, act, key) {
    const ai = sc.ai;
    if (!ai || ai.dead || S.script !== sc) return;
    if (act === 'stop') aiStop(sc);
    else if (act === 'show') aiShow(sc);
    else if (act === 'put') aiPut(sc, key);
    else if (act === 'undo') aiUndo(sc, key);
    else if (act === 'drop' || act === 'dismiss') aiDrop(sc, key);
  }
  function aiInput(sc, ta) {   // 칸을 고치는 중 — 되돌리기 거두기·AI 표시 걷기·[칸에 넣기]/[뒤에 붙이기] 이름 바꾸기
    const pt = root.preptask;
    if (!sc.ai) return;
    const f = pt.AI_FIELDS.filter(function (x) { return x.id === ta.id; })[0];
    if (!f) return;
    const m = sc.aiMarks[f.key];
    if (m && !ta.value.trim()) { delete sc.aiMarks[f.key]; aiRenderSlot(sc, f.key); }   // 칸을 비우면 'AI 칸'에서 뺀다
    else if (m && m.undo != null && ta.value !== m.after) { m.undo = null; m.prevMark = null; aiRenderSlot(sc, f.key); }   // 한 글자라도 고치면 [되돌리기]만 사라지고 표시 줄은 남는다
    const e = el(f.id + '-ai-e');
    if (e) e.textContent = '';
    const put = document.querySelector('#' + f.id + '-ai [data-ai-act="put"]');
    if (put) { const l = pt.aiPutLabel(f, ta.value.trim() !== ''); put.textContent = l.text; put.setAttribute('aria-label', l.aria); }
  }
```

### 3-D. [서버 연결 뒤 적용] `ai_fields`를 Word 요청에 싣는 패치 (지금은 적용하지 않는다)
- **지금 실어도 되는가**:
  - `PrepScriptIn`에 `@JsonIgnoreProperties(ignoreUnknown = true)`가 붙어 있다(`dto/PrepScriptIn.java:10`).
  - `JacksonConfig`는 SNAKE_CASE와 `serializationInclusion`만 바꾸고 `FAIL_ON_UNKNOWN_PROPERTIES`는 건드리지 않는다. Spring Boot 기본값이 꺼짐이다.
  - 그래서 모르는 `ai_fields` 키는 400이 아니라 **조용히 무시**된다. 서버 연결 전에 실어도 해롭지는 않다.
  - 다만 기록이 되지 않는데 되는 것처럼 보이게 하지 않으려고, 서버의 `PrepScriptIn`·`PrepController`가 받을 준비가 된 뒤에 적용한다.
```diff
-      headers: { 'Content-Type': 'application/json', 'X-Screen-Id': 'prep_detail', 'X-Button-Id': 'script_docx' }, body: JSON.stringify(pt.scriptBody(vals)) })
+      headers: { 'Content-Type': 'application/json', 'X-Screen-Id': 'prep_detail', 'X-Button-Id': 'script_docx' }, body: JSON.stringify(scriptReqBody(sc, vals)) })
```
`onScriptGo` 앞(C-3 묶음 안 어디든)에 추가:
```js
  // 요청 본문 = 기존 입력값 + 'AI 초안에서 시작한 칸' 키들(시스템 기록용 — Word에는 표시하지 않는다). 칸을 비웠으면 aiFieldsOf가 뺀다.
  function scriptReqBody(sc, vals) {
    const pt = root.preptask, body = pt.scriptBody(vals);   // 시험이 'pt.scriptBody(vals)' 글자를 이 구간에서 찾는다
    const af = pt.aiFieldsOf(sc.aiMarks, vals);
    if (af.length) body.ai_fields = af;
    return body;
  }
```
이 패치를 적용하면 `preptask.test.js:832`는 `pt.scriptBody(vals)`가 `scriptReqBody` 안에 있어 통과한다. 새 시험 파일 마지막의 `ai_fields는 아직 없다` 단정(아래 4-A 마지막 줄)은 반대로 고치거나 지운다.

## 5. 규격과 다르게 한 곳과 이유
1. **Word 성공 후 창을 유지할 때의 문구 자리.**
   - 규격 5-2는 "고정 줄"이라고만 썼다. 그대로 쓰면 진행 중인 AI 줄(`#bs-ai-line`)의 경과 시간을 덮어쓴다.
   - 그래서 이미 있는 안내 줄 `#bs-busy`(role=status)에 쓴다. 다음 Word 시도 때 `scriptBusy`가 그 줄을 비운다.
2. **`.bs-ai`의 `grid-column:1 / -1`을 뺐다.** 단추 블록이 `.bs-grid` 밖(legend와 grid 사이)에 있어서 의미가 없다.
3. **넣은 칸을 사람이 모두 지우면 'AI 칸' 표시도 걷는다.** 규격 4-2는 "한 글자라도 고치면 되돌리기만 사라진다"까지만 말한다. 빈 칸에 'AI 초안을 넣은 칸입니다'가 남는 것을 막고, `ai_fields`에서도 빠지게 한 것이다(디자인 이관 메모 "넣은 뒤 칸을 비우면 빼고"와 맞춘다). [되돌리기]를 눌러도 표시를 걷는다.
4. **사유·한도 줄 `…-ai-e`에 `role="status"`를 달았다.** 단추를 눌러 문장이 생길 때 스크린리더가 읽게 하려는 것이다. 포커스는 옮기지 않는다.
5. **응답 사유 이름은 5-3의 네 갈래로 묶었다.**
   - MODEL_MISSING은 '연결 안 됨' 문장으로 보낸다.
   - BAD_OUTPUT·LLM_ERROR·그 밖은 500 문장('서버에서 오류가 났습니다')으로 보낸다. 규격에 이 둘의 문장이 없어서 새 문구를 만들지 않았다.
6. **근거 줄의 '기관 정보'는 서버가 알려 주지 않는다.** `intro` 글이 왔으면 기관 정보를 썼다고 화면이 추정한다. 서버 `askable` 규칙상 intro는 기관명이 있어야 물으므로 사실과 어긋나지 않는다. 확실히 하려면 아래 서버 요청 3번.
7. **상태 조회에 5초 제한을 뒀다.** 규격에 없다. 서버가 멈췄을 때 창이 안 열리는 것을 막으려는 것이다. 시간이 넘으면 '단추 없음'으로 연다.
8. **넣은 칸 표시(`sc.aiMarks`)는 창을 닫았다 열어도 남는다.**
   - 칸의 글을 간직하는 것과 같은 수명이다. 단 닫는 순간 [되돌리기]는 거둔다(되돌릴 이전 값을 들고 있지 않는다).
   - 기능이 그 사이 꺼져도 표시 줄은 남는다.
   - 규격은 닫은 뒤의 표시를 말하지 않아 팀이 정했다.
9. **1초마다는 경과 시간 글자만 바꾼다.** 줄 전체를 다시 그리면 [그만두기]에 있던 키보드 포커스가 매초 풀린다. 상태가 바뀔 때(8분 줄·도착 등)만 다시 그리고, 그때 포커스가 줄 안에 있었으면 새 단추로 옮겨 준다.
10. **막힌 묶음 단추는 CSS 한 줄을 추가했다**(`#bs-ai-go[aria-disabled="true"]`). 규격의 "DESIGN 비활성 규칙"을 기존 `#hb-prep[aria-disabled]`와 같은 모양(muted 글자·기본 커서)으로 옮긴 것이다.

## 6. 서버에 확인·요청할 것 (이관 메모에도 반복)
- **[그만두기]·창 닫기로 fetch를 끊으면 서버 자리는 언제 돌아오는가 — 코드로 확인한 결과:**
  - `PrepScriptDraftController.draft`는 요청 스레드 하나에서 `drafts.reserve` → `try { … drafts.call(plan) } finally { drafts.release() }`를 **동기**로 돈다.
  - 브라우저가 연결을 끊어도 Tomcat은 요청 스레드를 멈추지 않는다. 코드에 연결 끊김을 보는 곳도 없다.
  - `OllamaLlmClient`의 watchdog(72줄, 120줄 근처)은 connect+read 전체 마감만 넘기면 호출을 끊는다. 마감은 설계서 기준 480초다.
  - 그래서 **[그만두기]나 창 닫기 뒤에도 서버는 마감(최대 480초)까지 동시 1건 자리(`Semaphore(1)`)를 쥔다.**
  - 그동안 본인이 다시 누르면 200 + `reason:"BUSY"`('다른 요청이 지금 AI를 쓰고 있습니다')를 받는다. 사번별 쿨다운 60초는 시작 시각(`lastStart`) 기준이라 보통 이미 지나 있다.
  - 이어서 감사 `SCRIPT_AI_DRAFT`도 정상으로 남고, 응답을 쓰다가 끊긴 연결 오류가 서버 로그에 한 줄 남을 수 있다.
  - 화면은 BUSY 문구를 '다른 요청'으로 써서 모순되지 않게 했다.
  - **요청(개발: 백엔드)**: ⓐ 연결 끊김에서 LLM 호출을 취소하거나(비동기 요청의 끊김 감지로 HttpPost abort), ⓑ BUSY 응답에 `busy_since`·`retry_after_s` 같은 값을 실어 화면이 "몇 분 뒤"를 구체적으로 말하게 한다. 디자인의 이관 메모 1번(문구 '잠시 뒤 가능 — 최대 n분')이 이 답을 기다린다.
- **`basis`를 더 구체적으로**:
  - ⓐ `basis.inst: true/false`. 지금은 `fields.intro` 유무로 추정한다.
  - ⓑ `basis.draft`가 칸별이면 '확정 전 자료 기준' 칩을 정확한 칸에만 붙일 수 있다. 지금은 요청 단위라 intro가 이사회 자료를 안 썼어도 칩이 둘 다 붙는다.
  - ⓒ 공고 파일은 `notice_files:[{name, why}]`가 오면 5-5의 이름·사유 문구로 바꾼다(지금은 USED/NONE뿐이라 '쓰지 않았습니다' 한 줄).
- **`ai_fields` 수신**: `PrepScriptIn`에 `List<String> aiFields`(허용 키 intro·effect만)를 받고, `PrepController`에서 감사(시스템 기록)에 남기는 일. 이게 끝나면 3-D를 적용한다. 지금 실어도 `PrepScriptIn`의 `ignoreUnknown`과 Spring Boot 기본 설정 때문에 400이 아니라 무시된다(위 3-D 근거).
- **`status` 엔드포인트**: 사번별 자격은 안 돌려준다. 화면은 `viewer`로 먼저 걸러내므로 괜찮다. 별도 요청은 없다.
- **6e 세션 결합**: `ScriptDraftUiIds` 주석에 `BoardUiIds`에 `script_ai_draft`가 등록되면 이사회 자료 줄이 `BoardUiIds.of`로 바뀐다고 적혀 있다. 화면이 보내는 헤더 값(`prep_detail`/`script_ai_draft`)은 그대로 유지해야 한다.

## 7. 위험·되돌리는 법
- **`get dirtyText()` 접근자(게터)를 쓴 점**: `dlg.js`가 `e.opts.dirtyText`를 읽는 순간에 문장이 정해진다(`showConfirm`). 그 사이 상태가 바뀌어도 맞는 문장이 나온다. `dlg.open`이 옵션 객체를 복사하지 않고 그대로 들고 있어서 동작한다(dlg.js:79). dlg.js를 바꾸지 않았다.
- **창을 연 채 Word 성공을 유지**: 초안이 있는데 [Word 만들기]에 성공하면 창이 남는다. 규격 3절 결정이고, 되돌려도 안전하다(옛 동작으로 돌아간다).
- **마크가 바뀌는 시점**: `sc.aiMarks`는 `scriptRead` 같은 텍스트 칸 값과는 별도로 움직인다. 칸을 직접 고쳐도 `input` 핸들러가 표시를 맞춘다. 단, 프로그램으로 `ta.value`를 바꾸는 곳(넣기·되돌리기)은 `input` 이벤트가 나지 않아 글자 수·표시를 그 함수들이 직접 맞춘다.
- **지도·렌더 코드**: 건드리지 않았다.
- **되돌리는 법**: 패치는 모두 추가이거나 `onScriptOpen`의 분리다. `o.ai`를 안 넘기면(`aiGate` 거짓·상태 조회 실패) 화면이 이전과 같다. 완전히 되돌리려면 `prep.js`의 `onScriptOpen`·`onScriptGo` 두 곳을 원래대로 두고, 나머지 추가분(A-1·A-5·B-2·C-3)은 지우면 된다.

## 8. 이관 메모
- **[이관: 개발]** [그만두기]·창 닫기 뒤 서버의 동시 1건 자리가 마감(최대 480초)까지 풀리지 않는다. 연결 끊김 시 호출 취소를 넣을지, BUSY 응답에 대기 시간을 실을지 결정이 필요하다. 기다리는 것은 디자인의 '잠시 뒤 가능 — 최대 n분' 문구와 아래 보고의 BUSY 문구다. 근거는 `PrepScriptDraftController.draft`의 `try/finally`와 `OllamaLlmClient` watchdog 코드다.
- **[이관: 개발]** `basis.inst`, 칸별 `basis.draft`, `notice_files[{name, why}]`를 응답에 넣을지. 기다리는 것은 칩의 정확도와 5-5 공고 파일 문구다. 근거는 `ScriptDraftService.Result.toMap`의 `notice`가 USED/NONE뿐이라는 점이다.
- **[이관: 개발]** `PrepScriptIn.aiFields`·`PrepController` 감사 연결. 기다리는 것은 프런트 3-D 패치 적용이다. 근거는 `PrepScriptIn`이 ignoreUnknown이라 지금은 키가 조용히 버려진다는 점이다.
- **[이관: 디자인]** 규격 5-5에 칸 이름 목록이 2개뿐이라 일반화 문구가 없다. 다음 사례(요약서 기대효과)가 같은 컴포넌트를 쓸 때 칸 이름·도움말 문구만 갈아 끼우면 되는지 확인이 필요하다. 근거는 `pt.AI_FIELDS`·`aiSlotHtml`이 칸 정의(`ai:true`)에서 이름을 읽는다는 점이다.
- **[이관: 디자인]** 전/후 장(`docs/design_history/index.html`)은 규격 8절 문안으로 구현 커밋 때 메인이 쓴다. 근거는 PUSH-GATE다.
