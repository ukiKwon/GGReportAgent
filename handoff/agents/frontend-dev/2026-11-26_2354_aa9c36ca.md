각 항목은 "찾을 원문(파일에서 유일한 문자열) → 바꿀 문자열"이다. 파일은 모두 `c:\github\GGReportAgent\frontend\index.html`.

### 6-B-1 @font-face 4개 (`:root` 앞)
찾을 원문:
```
  :root { --bg:#e9ecf0;
```
바꿀 문자열(맨 끝 `:root { --bg:#e9ecf0;`는 그대로 이어서):
```
  /* KB금융 글꼴(사용자 11-26, NEXT 130) — 본문체 Light=400·Medium=600·Bold=700, 제목체 Bold. 설치돼 있으면 local()로 쓰고 없을 때만 fonts/를 내려받는다(preload 금지: local()을 무시한다). family 이름을 설치명(KBFG Text)과 다르게 둔 것은 400→Medium 기본 대응을 피하려는 것 — 400이 Light로 가야 맑은 고딕과 농도가 같다. */
  @font-face{font-family:"KB Text";font-weight:300 400;font-display:swap;
    src:local("KBFGText-Light"),local("KBFG Text Light"),local("KB금융 본문체 Light"),url("fonts/KBFGText-Light.woff2") format("woff2");}
  @font-face{font-family:"KB Text";font-weight:500 600;font-display:swap;
    src:local("KBFGText-Medium"),local("KBFG Text Medium"),local("KB금융 본문체 Medium"),url("fonts/KBFGText-Medium.woff2") format("woff2");}
  @font-face{font-family:"KB Text";font-weight:700 900;font-display:swap;
    src:local("KBFGText-Bold"),local("KBFG Text Bold"),local("KB금융 본문체 Bold"),url("fonts/KBFGText-Bold.woff2") format("woff2");}
  @font-face{font-family:"KB Display";font-weight:500 900;font-display:swap;
    src:local("KBFGDisplay-Bold"),local("KBFG Display Bold"),local("KB금융 제목체 Bold"),url("fonts/KBFGDisplay-Bold.woff2") format("woff2");}
  :root { --bg:#e9ecf0;
```
주의: 주석에 `:root {`(콜론 root 공백 중괄호)라는 글자를 쓰지 않았다. 테스트 두 개(`board-write-focus.test.js:240`, `kbboard-1119.test.js:159`)가 html에서 처음 나오는 `:root {`를 찾는다. 위 주석에는 그 글자가 없으니 그대로 둔다.

## 7. 새 테스트 초안 `frontend/test/kb-fonts.test.js`
읽기 방식은 `screen-97s2.test.js:6~18`과 같다. `index.html`을 읽어 `<style>`만 잘라 주석을 걷는다.
```js
'use strict';
const test = require('node:test');
const assert = require('node:assert');
const fs = require('fs');
const path = require('path');
const html = fs.readFileSync(path.join(__dirname, '..', 'index.html'), 'utf8');
const css = html.slice(html.indexOf('<style>'), html.indexOf('</style>')).replace(/\/\*[\s\S]*?\*\//g, '');
const faces = css.match(/@font-face\s*\{[^}]*\}/g) || [];

test('KB 글꼴: @font-face 4개 — family·굵기 범위·local() 뒤 url()·woff2 이름', function () {
  const want = [
    ['KB Text', '300 400', 'KBFGText-Light'],
    ['KB Text', '500 600', 'KBFGText-Medium'],
    ['KB Text', '700 900', 'KBFGText-Bold'],
    ['KB Display', '500 900', 'KBFGDisplay-Bold']
  ];
  assert.strictEqual(faces.length, 4);
  want.forEach(function (w, i) {
    const f = faces[i];
    assert.ok(f.indexOf('font-family:"' + w[0] + '"') >= 0, w[2] + ' family');
    assert.ok(f.indexOf('font-weight:' + w[1] + ';') >= 0, w[2] + ' weight');
    assert.ok(f.indexOf('font-display:swap') >= 0, w[2] + ' swap');
    assert.strictEqual((f.match(/local\(/g) || []).length, 3, w[2] + ' local 3개');
    assert.strictEqual((f.match(/url\(/g) || []).length, 1, w[2] + ' url 1개');
    assert.ok(f.lastIndexOf('local(') < f.indexOf('url('), w[2] + ' local이 url보다 앞');
    assert.ok(f.indexOf('local("' + w[2] + '")') >= 0, w[2] + ' PostScript 이름');
    assert.ok(f.indexOf('url("fonts/' + w[2] + '.woff2") format("woff2")') >= 0, w[2] + ' woff2');
    assert.ok(!/https?:|\/\//.test(f.replace(/url\("fonts\//, '')), w[2] + ' 외부 주소 없음');
  });
});

test('KB 글꼴: preload·외부 링크 없음, 글꼴 이름은 --… 선언에만 남는다', function () {
  assert.ok(!/rel=["']?preload/i.test(html));
  assert.ok(!/<link[^>]+font/i.test(html));
  const bad = css.split('\n').filter(function (l) { return /Malgun/.test(l) && !/^\s*--[\w-]+:/.test(l); });
  assert.deepStrictEqual(bad, []);   // font 단축 속성·body에 맑은 고딕을 박아 두지 않는다

test('KB 글꼴: 변수·body·--mono·--lg/--ld 변수', function () {
  assert.ok(css.indexOf('--font-text:"KB Text","Malgun Gothic","Segoe UI",sans-serif;') >= 0);
  assert.ok(css.indexOf('--font-display:"KB Display","KB Text","Malgun Gothic","Segoe UI",sans-serif;') >= 0);
  assert.ok(/(^|\n)\s*body \{[^}]*font-family:var\(--font-text\)/.test(css));
  assert.ok(css.indexOf('--mono:"KB Text",Consolas,') >= 0);
  assert.ok(css.indexOf('--ld-mono:var(--mono);') >= 0);
  assert.ok(css.indexOf('--lg-sans:var(--font-text);') >= 0 && css.indexOf('--ld-sans:var(--font-text);') >= 0);
  assert.ok(css.indexOf('--lg-thin:"KB Text",') >= 0 && css.indexOf('--ld-thin:"KB Text",') >= 0);

test('KB 글꼴: 제목체는 h1~h4, 13px 이하 제목은 그 뒤에서 본문체로 되돌린다', function () {
  const disp = css.indexOf('h1, h2, h3, h4 { font-family:var(--font-display); }');
  assert.ok(disp > 0);
  const back = /\.bd-m-return h3, \.mn h4, \.db-sub-h h4, \.hb-rec h4, \.bs-auto h3, \.rs-rm-h h3,[^{]*\{ font-family:var\(--font-text\); \}/.exec(css);
  assert.ok(back && back.index > disp);
```

## 9. 위험·되돌리는 법
- 위험 1(미설치 PC): `url("fonts/…woff2")`이 404면 글꼴이 맑은 고딕으로 돌아간다. 깨지지는 않는다.
- 위험 2(서버가 `.woff2`를 못 주는 경우): 정적 파일 서빙이 `fonts/` 폴더를 포함하는지, MIME `font/woff2`로 내는지는 확인이 필요하다(이관 메모 참고).
- 위험 3(고정 높이 칸): D+ 뱃지, `.stab-x`, `.ts-phase`, `.ts-mk`, `.ld-dday`, `.ld-sq`에서 글자가 약 0.8px 위로 간다(규격 5절). 적용 뒤 확대해서 보고 어긋나면 해당 칸에 `padding-top:1px`.
- 위험 4(`--mono`): KB 숫자가 Consolas보다 약 7% 넓다. 가로가 빠듯한 칸이 있으면 규격 7곳 확인에서 본다.
- 되돌리기: 6-B-1~7과 새 테스트 파일만 되돌리면 된다. 6-B-2의 `--mono` 값만 원래 문자열로 되돌려도 숫자 열은 Consolas로 돌아간다.
