    /** 같은 열(cl)의 다른 행을 참조하는 수식. */
    static String formula(R w, String cl) {
        int[] n = new int[w.src.length];
        for (int i = 0; i < n.length; i++) n[i] = 15 + idx.get(w.src[i]) + 1;
        switch (w.op) {
            case "SUM": { StringBuilder s = new StringBuilder("SUM("); for (int i = 0; i < n.length; i++) { if (i > 0) s.append(","); s.append(cl).append(n[i]); } return s.append(")").toString(); }
            case "MULT2": return cl + n[0] + "*" + cl + n[1] + "/100";
            case "MULT3": return cl + n[0] + "*" + cl + n[1] + "*" + cl + n[2] + "/100";
            case "COPY": return cl + n[0];
            case "SUB": return cl + n[0] + "-" + cl + n[1];
            case "SUB4": return cl + n[0] + "-" + cl + n[1] + "-" + cl + n[2] + "-" + cl + n[3];
            case "TAX": return "MAX(0," + cl + n[0] + ")*0.264";
            default: throw new IllegalStateException(w.op);
        }
    }
    static boolean same(String[] x, String[] y, int L) { for (int i = 0; i <= L; i++) if (!x[i].equals(y[i])) return false; return true; }
