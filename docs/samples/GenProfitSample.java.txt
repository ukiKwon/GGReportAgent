    static boolean samePrefix(String[] x, String[] y, int L) { for (int i = 0; i <= L; i++) if (!x[i].equals(y[i])) return false; return true; }

    static double irr(double[] cf) {
        double lo = -0.9, hi = 5; double flo = npv(cf, lo), fhi = npv(cf, hi);
        if (Double.isNaN(flo) || flo * fhi > 0) return Double.NaN;
        for (int k = 0; k < 100; k++) { double mid = (lo + hi) / 2, fm = npv(cf, mid); if (flo * fm <= 0) { hi = mid; } else { lo = mid; flo = fm; } }
        return (lo + hi) / 2;
    }
    static double npv(double[] cf, double r) { double s = 0; for (int i = 0; i < cf.length; i++) s += cf[i] / Math.pow(1 + r, i); return s; }
